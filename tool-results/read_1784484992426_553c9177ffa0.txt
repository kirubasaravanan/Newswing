    69→    69→
    70→    70→export function AutoTradeTab() {
    71→    71→  const { config } = useTradeStore();
    72→    72→  const [wallet, setWallet] = useState<WalletData | null>(null);
    73→    73→  const [rules, setRules] = useState<PositionRules>({
    74→    74→    maxPerStock: 50000, maxBuysPerMonth: 3, maxHoldingDays: 25,
    75→    75→    maxTotalPositions: 8, riskPerTradePct: 1.0,
    76→    76→    trailingStopR: 1.5, trailToR: 0.5, partialBookR: 2.0, partialBookPct: 30,
    77→    77→    cooldownDays: 3, maxSectorPct: 35, timeExitMins: 30,
    78→    78→    maxDrawdownPct: 8, dailyLossLimit: 5000, niftyRegimeFilter: true,
    79→    79→    atrTrailMultiplier: 2.0, adaptiveSizing: true, streakPenaltyPct: 20,
    80→    80→  });
    81→    81→  const [openPositions, setOpenPositions] = useState<OpenPosition[]>([]);
    82→    82→  const [logs, setLogs] = useState<AutoTradeLog[]>([]);
    83→    83→  const [scheduler, setScheduler] = useState<SchedulerState | null>(null);
    84→    84→  const [marketHours, setMarketHours] = useState(false);
    85→    85→  const [timeToClose, setTimeToClose] = useState(0);
    86→    86→  const [scanning, setScanning] = useState(false);
    87→    87→  const [checkingExits, setCheckingExits] = useState(false);
    88→    88→  const [editingWallet, setEditingWallet] = useState(false);
    89→    89→  const [walletInput, setWalletInput] = useState('200000');
    90→    90→  const [editingRules, setEditingRules] = useState(false);
    91→    91→  const [ruleEdits, setRuleEdits] = useState<PositionRules & { scanIntervalMin?: number; exitIntervalMin?: number }>({ ...rules });
    92→    92→  const [chartSymbol, setChartSymbol] = useState<string | null>(null);
    93→    93→  const [exitWarnings, setExitWarnings] = useState<any[]>([]);
    94→    94→  const [selectedLog, setSelectedLog] = useState<AutoTradeLog | null>(null);
    95→    95→  const [showAllLogs, setShowAllLogs] = useState(false);
    96→    96→  const [drawdown, setDrawdown] = useState<DrawdownData | null>(null);
    97→    97→
    98→    98→  // ── Data Fetching ─────────────────────────────────────
    99→    99→  const fetchStatus = useCallback(async () => {
   100→   100→    try {
   101→   101→      const res = await fetch('/api/auto-trade');
   102→   102→      const data = await res.json();
   103→   103→      if (data.success) {
   104→   104→        setWallet(data.wallet);
   105→   105→        setRules(data.rules);
   106→   106→        setOpenPositions(data.openTrades);
   107→   107→        setLogs(data.recentLogs || []);
   108→   108→        setScheduler(data.scheduler);
   109→   109→        setMarketHours(data.marketHours);
   110→   110→        setTimeToClose(data.timeToClose);
   111→   111→        if (data.drawdown) setDrawdown(data.drawdown);
   112→   112→      }
   113→   113→    } catch (err) { console.error('Fetch status error:', err); }
   114→   114→  }, []);
   115→   115→
   116→   116→  useEffect(() => {
   117→   117→    fetchStatus();
   118→   118→    const interval = setInterval(fetchStatus, 30000);
   119→   119→    return () => clearInterval(interval);
   120→   120→  }, [fetchStatus]);
   121→   121→
   122→   122→  // ── Scheduler Tick ─────────────────────────────────────
   123→   123→  const schedulerTickRef = useRef<NodeJS.Timeout | null>(null);
   124→   124→  useEffect(() => {
   125→   125→    if (scheduler?.enabled && marketHours && !scheduler?.circuitBreaker) {
   126→   126→      schedulerTickRef.current = setInterval(async () => {
   127→   127→        try {
   128→   128→          await fetch('/api/auto-trade', {
   129→   129→            method: 'POST', headers: { 'Content-Type': 'application/json' },
   130→   130→            body: JSON.stringify({ action: 'scheduler_tick' }),
   131→   131→          });
   132→   132→          fetchStatus();
   133→   133→        } catch { /* ignore */ }
   134→   134→      }, 60000);
   135→   135→    }
   136→   136→    return () => { if (schedulerTickRef.current) clearInterval(schedulerTickRef.current); };
   137→   137→  }, [scheduler?.enabled, marketHours, scheduler?.circuitBreaker, fetchStatus]);
   138→   138→
   139→   139→  // ── Actions ───────────────────────────────────────────
   140→   140→  const handleScanAndTrade = async () => {
   141→   141→    setScanning(true);
   142→   142→    try {
   143→   143→      const res = await fetch('/api/auto-trade', {
   144→   144→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   145→   145→        body: JSON.stringify({ action: 'scan_and_trade', config }),
   146→   146→      });
   147→   147→      const data = await res.json();
   148→   148→      if (data.success) {
   149→   149→        const extra = data.circuitBreaker ? ' [CIRCUIT BREAKER]' :
   150→   150→          data.niftyRegime === 'BEARISH' ? ' [BEARISH REGIME]' : '';
   151→   151→        toast.success(`Scan Complete${extra}`, {
   152→   152→          description: `${data.totalScanned} scanned | ${data.l1Passed} L1 | ${data.l2Signals} entries | ${data.skipped.length} skipped`,
   153→   153→        });
   154→   154→        if (data.adaptiveFactor && data.adaptiveFactor < 1) {
   155→   155→          toast.info('Adaptive Sizing Active', {
   156→   156→            description: `Factor: ${(data.adaptiveFactor * 100).toFixed(0)}% (${data.consecutiveLosses} consecutive losses)`,
   157→   157→          });
   158→   158→        }
   159→   159→        fetchStatus();
   160→   160→      } else {
   161→   161→        toast.error('Auto-trade failed', { description: data.error });
   162→   162→      }
   163→   163→    } catch { toast.error('Auto-trade failed'); }
   164→   164→    finally { setScanning(false); }
   165→   165→  };
   166→   166→
   167→   167→  const handleCheckExits = async () => {
   168→   168→    setCheckingExits(true);
   169→   169→    try {
   170→   170→      const res = await fetch('/api/auto-trade', {
   171→   171→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   172→   172→        body: JSON.stringify({ action: 'check_exits' }),
   173→   173→      });
   174→   174→      const data = await res.json();
   175→   175→      if (data.success) {
   176→   176→        if (data.exits.length > 0) {
   177→   177→          toast.success(`Exit Check: ${data.exits.length} closed`, {
   178→   178→            description: data.exits.map((e: any) => `${e.symbol} (${e.exitReason})`).join(', '),
   179→   179→          });
   180→   180→        }
   181→   181→        if (data.partialBooks?.length > 0) {
   182→   182→          toast.info(`Partial Book: ${data.partialBooks.length} positions`, {
   183→   183→            description: data.partialBooks.map((p: any) => `${p.symbol}: ${p.bookedQty}qty at ${p.rMultiple}R`).join(', '),
   184→   184→          });
   185→   185→        }
   186→   186→        if (data.exits.length === 0 && (!data.partialBooks || data.partialBooks.length === 0)) {
   187→   187→          toast.info('No exits triggered', { description: `${data.holding?.length || 0} positions holding` });
   188→   188→        }
   189→   189→        if (data.warnings?.length > 0) setExitWarnings(data.warnings);
   190→   190→        fetchStatus();
   191→   191→      }
   192→   192→    } catch { toast.error('Exit check failed'); }
   193→   193→    finally { setCheckingExits(false); }
   194→   194→  };
   195→   195→
   196→   196→  const handleSchedulerToggle = async (enabled: boolean) => {
   197→   197→    try {
   198→   198→      const res = await fetch('/api/auto-trade', {
   199→   199→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   200→   200→        body: JSON.stringify({
   201→   201→          action: 'scheduler_toggle', enabled,
   202→   202→          scanIntervalMin: ruleEdits.scanIntervalMin || 30,
   203→   203→          exitIntervalMin: ruleEdits.exitIntervalMin || 5,
   204→   204→        }),
   205→   205→      });
   206→   206→      const data = await res.json();
   207→   207→      if (data.success) {
   208→   208→        setScheduler(data.scheduler);
   209→   209→        toast.success(enabled ? 'Auto-Mode Armed' : 'Auto-Mode Disarmed');
   210→   210→        fetchStatus();
   211→   211→      }
   212→   212→    } catch { toast.error('Scheduler toggle failed'); }
   213→   213→  };
   214→   214→
   215→   215→  const handleResetCircuitBreaker = async () => {
   216→   216→    try {
   217→   217→      const res = await fetch('/api/auto-trade', {
   218→   218→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   219→   219→        body: JSON.stringify({ action: 'reset_circuit_breaker' }),
   220→   220→      });
   221→   221→      const data = await res.json();
   222→   222→      if (data.success) {
   223→   223→        setScheduler(data.scheduler);
   224→   224→        toast.success('Circuit Breaker Reset');
   225→   225→      }
   226→   226→    } catch { toast.error('Reset failed'); }
   227→   227→  };
   228→   228→
   229→   229→  const saveWallet = async () => {
   230→   230→    try {
   231→   231→      const res = await fetch('/api/auto-trade', {
   232→   232→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   233→   233→        body: JSON.stringify({ action: 'update_wallet', totalCapital: parseFloat(walletInput) }),
   234→   234→      });
   235→   235→      const data = await res.json();
   236→   236→      if (data.success) { setWallet(data.wallet); setEditingWallet(false); toast.success('Wallet updated'); }
   237→   237→    } catch { toast.error('Failed to update wallet'); }
   238→   238→  };
   239→   239→
   240→   240→  const saveRules = async () => {
   241→   241→    try {
   242→   242→      // Separate scheduler settings from rules
   243→   243→      const { scanIntervalMin, exitIntervalMin, ...rulesOnly } = ruleEdits;
   244→   244→      const res = await fetch('/api/auto-trade', {
   245→   245→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   246→   246→        body: JSON.stringify({ action: 'update_rules', rules: rulesOnly }),
   247→   247→      });
   248→   248→      const data = await res.json();
   249→   249→      if (data.success) { setRules(data.rules); setEditingRules(false); toast.success('Rules updated'); }
   250→   250→    } catch { toast.error('Failed to update rules'); }
   251→   251→  };
   252→   252→
   253→   253→  const totalPnl = (wallet?.realizedPnl || 0) + (wallet?.unrealizedPnl || 0);
   254→   254→  const pnlPct = wallet?.initialCapital ? (totalPnl / wallet.initialCapital) * 100 : 0;
   255→   255→
   256→   256→  return (
   257→   257→    <div className="space-y-4">
   258→   258→      {/* ── Top Bar ─────────────────────────────────────── */}
   259→   259→      <div className="flex items-center justify-between flex-wrap gap-3">
   260→   260→        <div className="flex items-center gap-3">
   261→   261→          <h2 className="text-lg font-bold flex items-center gap-2">
   262→   262→            <Bot className="h-5 w-5 text-primary" />
   263→   263→            Auto-Trade Engine
   264→   264→            <Badge variant="outline" className="text-[9px] h-4 text-muted-foreground">v2</Badge>
   265→   265→          </h2>
   266→   266→          {/* Market Status */}
   267→   267→          <div className={cn(
   268→   268→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   269→   269→            marketHours ? 'bg-emerald-500/10 text-emerald-400' : 'bg-muted text-muted-foreground'
   270→   270→          )}>
   271→   271→            <div className={cn('h-1.5 w-1.5 rounded-full', marketHours ? 'bg-emerald-400 animate-pulse' : 'bg-muted-foreground')} />
   272→   272→            {marketHours ? `${timeToClose}m to close` : 'Market Closed'}
   273→   273→          </div>
   274→   274→          {/* Circuit Breaker */}
   275→   275→          {scheduler?.circuitBreaker && (
   276→   276→            <div className="flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs bg-red-500/15 text-red-400 border border-red-500/20">
   277→   277→              <ShieldAlert className="h-3 w-3" />
   278→   278→              CIRCUIT BREAKER
   279→   279→              <Button variant="ghost" size="sm" onClick={handleResetCircuitBreaker} className="h-4 w-4 p-0 ml-1 text-red-400 hover:text-red-300">
   280→   280→                <RotateCcw className="h-2.5 w-2.5" />
   281→   281→              </Button>
   282→   282→            </div>
   283→   283→          )}
   284→   284→          {/* Nifty Regime */}
   285→   285→          <div className={cn(
   286→   286→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   287→   287→            scheduler?.niftyRegime === 'BULLISH' ? 'bg-emerald-500/10 text-emerald-400' :
   288→   288→            scheduler?.niftyRegime === 'BEARISH' ? 'bg-red-500/10 text-red-400' : 'bg-muted text-muted-foreground'
   289→   289→          )}>
   290→   290→            <TrendingUp className={cn('h-3 w-3', scheduler?.niftyRegime === 'BEARISH' && 'rotate-180')} />
   291→   291→            {scheduler?.niftyRegime || 'UNKNOWN'}
   292→   292→          </div>
   293→   293→          {/* Auto-Mode */}
   294→   294→          <div className={cn(
   295→   295→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   296→   296→            scheduler?.enabled ? 'bg-indigo-500/10 text-indigo-400' : 'bg-muted text-muted-foreground'
   297→   297→          )}>
   298→   298→            <Shield className="h-3 w-3" />
   299→   299→            {scheduler?.enabled ? 'Armed' : 'Manual'}
   300→   300→          </div>
   301→   301→        </div>
   302→   302→        <div className="flex items-center gap-2">
   303→   303→          <Button variant="outline" size="sm" onClick={handleCheckExits}
   304→   304→            disabled={checkingExits || openPositions.length === 0}
   305→   305→            className="h-8 gap-1.5 text-xs">
   306→   306→            {checkingExits ? <Loader2 className="h-3 w-3 animate-spin" /> : <ArrowRightLeft className="h-3 w-3" />}
   307→   307→            Check Exits
   308→   308→          </Button>
   309→   309→          <Button onClick={handleScanAndTrade} disabled={scanning || !marketHours || !!scheduler?.circuitBreaker}
   310→   310→            className={cn('gap-1.5', scheduler?.enabled ? 'bg-emerald-600 hover:bg-emerald-700' : '')}
   311→   311→            size="sm">
   312→   312→            {scanning ? <Loader2 className="h-3 w-3 animate-spin" /> : <Zap className="h-3 w-3" />}
   313→   313→            {scanning ? 'Scanning...' : 'Scan & Trade'}
   314→   314→          </Button>
   315→   315→        </div>
   316→   316→      </div>
   317→   317→
   318→   318→      <Tabs defaultValue="engine" className="space-y-4">
   319→   319→        <TabsList className="bg-secondary/50">
   320→   320→          <TabsTrigger value="engine" className="text-xs gap-1.5"><Bot className="h-3 w-3" /> Engine</TabsTrigger>
   321→   321→          <TabsTrigger value="scheduler" className="text-xs gap-1.5"><Timer className="h-3 w-3" /> Scheduler</TabsTrigger>
   322→   322→          <TabsTrigger value="positions" className="text-xs gap-1.5"><TrendingUp className="h-3 w-3" /> Positions</TabsTrigger>
   323→   323→          <TabsTrigger value="audit" className="text-xs gap-1.5"><FileText className="h-3 w-3" /> Audit</TabsTrigger>
   324→   324→          <TabsTrigger value="rules" className="text-xs gap-1.5"><Settings className="h-3 w-3" /> Rules</TabsTrigger>
   325→   325→        </TabsList>
   326→   326→
   327→   327→        {/* ═══ ENGINE TAB ═════════════════════════════════ */}
   328→   328→        <TabsContent value="engine" className="space-y-4">
   329→   329→          <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   330→   330→            {/* Wallet Card */}
   331→   331→            <Card className="lg:col-span-2 border-border">
   332→   332→              <CardContent className="p-4">
   333→   333→                <div className="flex items-center justify-between mb-4">
   334→   334→                  <h3 className="text-sm font-semibold flex items-center gap-2">
   335→   335→                    <Wallet className="h-4 w-4 text-primary" /> Capital Wallet
   336→   336→                  </h3>
   337→   337→                  {!editingWallet ? (
   338→   338→                    <Button variant="ghost" size="sm" onClick={() => { setWalletInput(String(wallet?.totalCapital || 200000)); setEditingWallet(true); }} className="h-7 text-xs">Edit</Button>
   339→   339→                  ) : (
   340→   340→                    <div className="flex items-center gap-2">
   341→   341→                      <Input value={walletInput} onChange={e => setWalletInput(e.target.value)} className="h-7 w-32 text-xs" />
   342→   342→                      <Button size="sm" onClick={saveWallet} className="h-7 text-xs">Save</Button>
   343→   343→                      <Button size="sm" variant="ghost" onClick={() => setEditingWallet(false)} className="h-7 text-xs">Cancel</Button>
   344→   344→                    </div>
   345→   345→                  )}
   346→   346→                </div>
   347→   347→                {wallet && (
   348→   348→                  <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   349→   349→                    <div className="rounded-lg bg-secondary/50 p-3">
   350→   350→                      <div className="text-[10px] text-muted-foreground uppercase">Total Capital</div>
   351→   351→                      <div className="text-lg font-bold font-mono mt-0.5">₹{wallet.totalCapital.toLocaleString()}</div>
   352→   352→                    </div>
   353→   353→                    <div className="rounded-lg bg-blue-500/10 p-3">
   354→   354→                      <div className="text-[10px] text-blue-400 uppercase">Deployed</div>
   355→   355→                      <div className="text-lg font-bold font-mono mt-0.5 text-blue-400">₹{wallet.deployed.toLocaleString()}</div>
   356→   356→                      <div className="text-[10px] text-muted-foreground">{wallet.totalCapital > 0 ? ((wallet.deployed / wallet.totalCapital) * 100).toFixed(1) : 0}%</div>
   357→   357→                    </div>
   358→   358→                    <div className="rounded-lg bg-emerald-500/10 p-3">
   359→   359→                      <div className="text-[10px] text-emerald-400 uppercase">Available</div>
   360→   360→                      <div className="text-lg font-bold font-mono mt-0.5 text-emerald-400">₹{wallet.available.toLocaleString()}</div>
   361→   361→                    </div>
   362→   362→                    <div className={cn('rounded-lg p-3', totalPnl >= 0 ? 'bg-emerald-500/10' : 'bg-red-500/10')}>
   363→   363→                      <div className={cn('text-[10px] uppercase', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>Total P&L</div>
   364→   364→                      <div className={cn('text-lg font-bold font-mono mt-0.5', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
