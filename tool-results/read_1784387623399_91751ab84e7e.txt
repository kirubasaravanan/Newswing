    29→// ── Risk Metrics Card ─────────────────────────────────
    30→function RiskMetricsCard() {
    31→  const [metrics, setMetrics] = useState<any>(null);
    32→  const [loading, setLoading] = useState(true);
    33→
    34→  const fetchMetrics = useCallback(async () => {
    35→    setLoading(true);
    36→    try {
    37→      const res = await fetch('/api/portfolio/risk-metrics');
    38→      const data = await res.json();
    39→      if (data.success) setMetrics(data);
    40→    } catch { /* ignore */ }
    41→    finally { setLoading(false); }
    42→  }, []);
    43→
    44→  useEffect(() => { fetchMetrics(); }, [fetchMetrics]);
    45→
    46→  if (loading) return <div className="text-center py-8 text-muted-foreground text-sm"><Loader2 className="h-4 w-4 mx-auto animate-spin" /></div>;
    47→  if (!metrics || metrics.tradeCount < 2) return <p className="text-center py-8 text-muted-foreground text-sm">Need at least 2 closed trades for risk metrics.</p>;
    48→
    49→  const m = metrics.metrics;
    50→  const items = [
    51→    { label: 'Sharpe Ratio', value: m.sharpe.toFixed(2), color: m.sharpe >= 1 ? 'text-emerald-400' : m.sharpe >= 0 ? 'text-amber-400' : 'text-red-400', desc: m.sharpe >= 1 ? 'Good' : m.sharpe >= 0 ? 'Moderate' : 'Poor' },
    52→    { label: 'Sortino Ratio', value: m.sortino.toFixed(2), color: m.sortino >= 1.5 ? 'text-emerald-400' : 'text-amber-400', desc: 'Downside risk adjusted' },
    53→    { label: 'Max Drawdown', value: `${m.maxDD.toFixed(1)}%`, color: 'text-red-400', desc: 'Worst peak-to-trough' },
    54→    { label: 'Calmar Ratio', value: m.calmar.toFixed(2), color: m.calmar >= 1 ? 'text-emerald-400' : 'text-amber-400', desc: 'Return / MaxDD' },
    55→    { label: 'VaR (95%)', value: `₹${Math.abs(m.var95).toLocaleString()}`, color: 'text-red-400', desc: 'Max daily loss (95% conf)' },
    56→    { label: 'CAGR', value: `${m.cagr.toFixed(1)}%`, color: m.cagr >= 15 ? 'text-emerald-400' : 'text-amber-400', desc: 'Annualized return' },
    57→    { label: 'Avg Holding', value: `${m.avgHoldingDays}d`, color: 'text-indigo-400', desc: 'Average trade duration' },
    58→    { label: 'Best Trade', value: `₹${m.bestTrade.toLocaleString()}`, color: 'text-emerald-400', desc: '' },
    59→    { label: 'Worst Trade', value: `₹${m.worstTrade.toLocaleString()}`, color: 'text-red-400', desc: '' },
    60→  ];
    61→
    62→  return (
    63→    <div className="grid grid-cols-3 lg:grid-cols-3 xl:grid-cols-3 gap-3">
    64→      {items.map(item => (
    65→        <div key={item.label} className="rounded-lg bg-secondary/50 p-3">
    66→          <div className="text-[10px] text-muted-foreground uppercase">{item.label}</div>
    67→          <div className={cn('text-lg font-bold font-mono mt-0.5', item.color)}>{item.value}</div>
    68→          {item.desc && <div className="text-[10px] text-muted-foreground mt-0.5">{item.desc}</div>}

    70→      ))}

   135→// ── Price Alerts ────────────────────────────────────────
   136→function AlertsPanel() {
   137→  const [alerts, setAlerts] = useState<any[]>([]);
   138→  const [open, setOpen] = useState(false);
   139→  const [form, setForm] = useState({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   140→  const [checking, setChecking] = useState(false);
   141→
   142→  const fetchAlerts = useCallback(async () => {
   143→    try {
   144→      const res = await fetch('/api/portfolio/alerts');
   145→      const data = await res.json();
   146→      if (data.success) setAlerts(data.alerts);
   147→    } catch { /* ignore */ }
   148→  }, []);
   149→
   150→  useEffect(() => { fetchAlerts(); }, [fetchAlerts]);
   151→
   152→  const createAlert = async () => {
   153→    if (!form.symbol || !form.targetPrice) return;
   154→    try {
   155→      const res = await fetch('/api/portfolio/alerts', {
   156→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify(form),
   158→      });
   159→      const data = await res.json();
   160→      if (data.success) {
   161→        toast.success(`Alert set: ${form.symbol} ${form.condition} ₹${form.targetPrice}`);
   162→        setForm({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   163→        setOpen(false);
   164→        fetchAlerts();
   165→      }
   166→    } catch { toast.error('Failed to create alert'); }
   167→  };
   168→
   169→  const deleteAlert = async (id: string) => {
   170→    await fetch(`/api/portfolio/alerts?id=${id}`, { method: 'DELETE' });
   171→    fetchAlerts();
   172→  };
   173→
   174→  const checkAlerts = async () => {
   175→    setChecking(true);
   176→    try {
   177→      const res = await fetch('/api/portfolio/alerts', { method: 'PUT' });
   178→      const data = await res.json();
   179→      if (data.success && data.triggered.length > 0) {
   180→        for (const t of data.triggered) {
   181→          toast.success(`🔔 ${t.symbol} hit ₹${t.triggeredPrice}!`, {
   182→            description: `Target: ₹${t.targetPrice} ${t.condition}`,
   183→            duration: 8000,
   184→          });
   185→        }
   186→        fetchAlerts();
   187→      } else {
   188→        toast.info('No alerts triggered');
   189→      }
   190→    } catch { toast.error('Failed to check alerts'); }
   191→    finally { setChecking(false); }
   192→  };
   193→
   194→  return (
   195→    <div>
   196→      <div className="flex items-center justify-between mb-3">
   197→        <h3 className="text-sm font-semibold flex items-center gap-2"><Bell className="h-4 w-4" /> Price Alerts</h3>
   198→        <div className="flex gap-2">
   199→          <Button variant="outline" size="sm" onClick={checkAlerts} disabled={checking} className="h-7 text-xs gap-1">
   200→            {checking ? <Loader2 className="h-3 w-3 animate-spin" /> : <Target className="h-3 w-3" />} Check
   201→          </Button>
   202→          <Dialog open={open} onOpenChange={setOpen}>
   203→            <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> New Alert</Button></DialogTrigger>
   204→            <DialogContent className="sm:max-w-sm">
   205→              <DialogHeader><DialogTitle>Create Price Alert</DialogTitle></DialogHeader>
   206→              <div className="grid gap-3 py-2">
   207→                <div><Label className="text-xs">Symbol</Label>
   208→                  <Input value={form.symbol} onChange={e => setForm({ ...form, symbol: e.target.value.toUpperCase() })} placeholder="RELIANCE" className="h-8 text-xs" /></div>
   209→                <div><Label className="text-xs">Condition</Label>
   210→                  <Select value={form.condition} onValueChange={v => setForm({ ...form, condition: v })}>
   211→                    <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   212→                    <SelectContent><SelectItem value="ABOVE">Price goes ABOVE</SelectItem><SelectItem value="BELOW">Price goes BELOW</SelectItem></SelectContent>
   213→                  </Select></div>
   214→                <div><Label className="text-xs">Target Price (₹)</Label>
   215→                  <Input type="number" value={form.targetPrice} onChange={e => setForm({ ...form, targetPrice: e.target.value })} placeholder="2500" className="h-8 text-xs" /></div>
   216→              </div>
   217→              <DialogFooter>
   218→                <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
   219→                <Button size="sm" onClick={createAlert}>Create</Button>
   220→              </DialogFooter>
   221→            </DialogContent>
   222→          </Dialog>
   223→        </div>
   224→      </div>
   225→      {alerts.length === 0 ? (
   226→        <p className="text-xs text-muted-foreground text-center py-4">No alerts set. Create one to get notified when a price target is hit.</p>
   227→      ) : (
   228→        <ScrollArea className="max-h-[200px]">
   229→          <div className="space-y-1.5 pr-2">
   230→            {alerts.slice(0, 20).map(a => (
   231→              <div key={a.id} className="flex items-center gap-2 text-xs py-1.5 border-b border-border/30">
   232→                <div className={cn('h-2 w-2 rounded-full shrink-0', a.triggered ? 'bg-amber-400' : 'bg-blue-400')} />
   233→                <span className="font-semibold">{a.symbol}</span>
   234→                <Badge variant="outline" className="text-[9px] h-4">{a.condition}</Badge>
   235→                <span className="font-mono">₹{a.targetPrice.toLocaleString()}</span>
   236→                {a.triggered && <Badge className="text-[9px] h-4 bg-amber-500/20 text-amber-400">HIT ₹{a.triggeredPrice}</Badge>}
   237→                <button onClick={() => deleteAlert(a.id)} className="ml-auto text-muted-foreground hover:text-red-400"><X className="h-3 w-3" /></button>
   238→              </div>
   239→            ))}
   240→          </div>
   241→        </ScrollArea>
   242→      )}
   243→    </div>
   244→  );
   245→}

   424→export function AnalyticsTab() {
   425→  const [trades, setTrades] = useState<any[]>([]);
   426→  const [analyticsData, setAnalyticsData] = useState<any>(null);
   427→  const [loading, setLoading] = useState(true);
   428→
   429→  const fetchAnalytics = useCallback(async () => {
   430→    setLoading(true);
   431→    try {
   432→      const res = await fetch('/api/trades');
   433→      const json = await res.json();
   434→      const allTrades = json.trades || [];
   435→      const closed = allTrades.filter((t: any) => t.status === 'CLOSED' && t.pnl != null);
   436→
   437→      const wins = closed.filter((t: any) => t.pnl > 0);
   438→      const totalPnL = closed.reduce((s: number, t: any) => s + t.pnl, 0);
   439→      const grossProfit = wins.reduce((s: number, t: any) => s + t.pnl, 0);
   440→      const grossLoss = Math.abs(closed.filter((t: any) => t.pnl <= 0).reduce((s: number, t: any) => s + t.pnl, 0));
   441→
   442→      const monthMap = new Map<string, { pnl: number; trades: number }>();
   443→      for (const t of closed) {
   444→        const d = new Date(t.exitDate);
   445→        const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   446→        const prev = monthMap.get(key) || { pnl: 0, trades: 0 };
   447→        prev.pnl += t.pnl;
   448→        prev.trades += 1;
   449→        monthMap.set(key, prev);
   450→      }
   451→      const byMonth = Array.from(monthMap.entries()).map(([month, v]) => ({
   452→        month, pnl: Math.round(v.pnl), trades: v.trades,
   453→      })).sort((a, b) => a.month.localeCompare(b.month));
   454→
   455→      const buckets = [
   456→        { range: '< -5%', min: -Infinity, max: -5 }, { range: '-5% to -2%', min: -5, max: -2 },
   457→        { range: '-2% to 0%', min: -2, max: 0 }, { range: '0% to 2%', min: 0, max: 2 },
   458→        { range: '2% to 5%', min: 2, max: 5 }, { range: '> 5%', min: 5, max: Infinity },
   459→      ];
   460→      const pnlDistribution = buckets.map(b => ({
   461→        range: b.range,
   462→        count: closed.filter((t: any) => t.pnlPercent >= b.min && t.pnlPercent < b.max).length,
   463→      }));
   464→
   465→      let peak = 0, running = 0, maxDD = 0;
   466→      for (const t of closed.sort((a: any, b: any) => new Date(a.exitDate).getTime() - new Date(b.exitDate).getTime())) {
   467→        running += t.pnl;
   468→        peak = Math.max(peak, running);
   469→        maxDD = Math.max(maxDD, (peak - running) / peak * 100);
   470→      }
   471→
   472→      setAnalyticsData({
   473→        trades: closed, totalPnL,
   474→        winRate: closed.length > 0 ? (wins.length / closed.length) * 100 : 0,
   475→        avgWin: wins.length > 0 ? grossProfit / wins.length : 0,
   476→        avgLoss: closed.length - wins.length > 0 ? grossLoss / (closed.length - wins.length) : 0,
   477→        profitFactor: grossLoss > 0 ? grossProfit / grossLoss : 0,
   478→        maxDD, byMonth, pnlDistribution,
   479→      });
   480→    } catch { /* ignore */ }
   481→    finally { setLoading(false); }
   482→  }, []);
   483→
   484→  useEffect(() => { fetchAnalytics(); }, [fetchAnalytics]);
   485→

   487→
   488→  const COLORS = ['#10b981', '#f59e0b', '#6366f1', '#ef4444', '#06b6d4', '#ec4899'];

   494→        <Button variant="outline" size="sm" onClick={fetchAnalytics} className="gap-2 h-8 text-xs"><RefreshCw className="h-3.5 w-3.5" /> Refresh</Button>
   495→      </div>
   496→
   497→      <Tabs defaultValue="overview" className="w-full">
   498→        <TabsList className="w-full justify-start mb-4">
   499→          <TabsTrigger value="overview" className="text-xs">Overview</TabsTrigger>
   500→          <TabsTrigger value="risk" className="text-xs">Risk Metrics</TabsTrigger>
   501→          <TabsTrigger value="benchmark" className="text-xs">Benchmark</TabsTrigger>
   502→          <TabsTrigger value="alerts" className="text-xs">Alerts</TabsTrigger>
