   136→function AlertsPanel() {
   137→  const [alerts, setAlerts] = useState<any[]>([]);
   138→  const [open, setOpen] = useState(false);
   139→  const [form, setForm] = useState({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   140→  const [checking, setChecking] = useState(false);
   141→
   142→  const fetchAlerts = useCallback(async () => {
   143→    try {
   144→      const res = await fetch('/api/portfolio/alerts');
   145→      const data = await res.json();
   146→      if (data.success) setAlerts(data.alerts);
   147→    } catch { /* ignore */ }
   148→  }, []);
   149→
   150→  useEffect(() => { fetchAlerts(); }, [fetchAlerts]);
   151→
   152→  const createAlert = async () => {
   153→    if (!form.symbol || !form.targetPrice) return;
   154→    try {
   155→      const res = await fetch('/api/portfolio/alerts', {
   156→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify(form),
   158→      });
   159→      const data = await res.json();
   160→      if (data.success) {
   161→        toast.success(`Alert set: ${form.symbol} ${form.condition} ₹${form.targetPrice}`);
   162→        setForm({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   163→        setOpen(false);
   164→        fetchAlerts();
   165→      }
   166→    } catch { toast.error('Failed to create alert'); }
   167→  };
   168→
   169→  const deleteAlert = async (id: string) => {
   170→    await fetch(`/api/portfolio/alerts?id=${id}`, { method: 'DELETE' });
   171→    fetchAlerts();
   172→  };
   173→
   174→  const checkAlerts = async () => {
   175→    setChecking(true);
   176→    try {
   177→      const res = await fetch('/api/portfolio/alerts', { method: 'PUT' });
   178→      const data = await res.json();
   179→      if (data.success && data.triggered.length > 0) {
   180→        for (const t of data.triggered) {
   181→          toast.success(`🔔 ${t.symbol} hit ₹${t.triggeredPrice}!`, {
   182→            description: `Target: ₹${t.targetPrice} ${t.condition}`,
   183→            duration: 8000,
   184→          });
   185→        }
   186→        fetchAlerts();
   187→      } else {
   188→        toast.info('No alerts triggered');
   189→      }
   190→    } catch { toast.error('Failed to check alerts'); }
   191→    finally { setChecking(false); }
   192→  };
   193→
   194→  return (
   195→    <div>
   196→      <div className="flex items-center justify-between mb-3">
   197→        <h3 className="text-sm font-semibold flex items-center gap-2"><Bell className="h-4 w-4" /> Price Alerts</h3>
   198→        <div className="flex gap-2">
   199→          <Button variant="outline" size="sm" onClick={checkAlerts} disabled={checking} className="h-7 text-xs gap-1">
   200→            {checking ? <Loader2 className="h-3 w-3 animate-spin" /> : <Target className="h-3 w-3" />} Check
   201→          </Button>
   202→          <Dialog open={open} onOpenChange={setOpen}>
   203→            <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> New Alert</Button></DialogTrigger>
   204→            <DialogContent className="sm:max-w-sm">
   205→              <DialogHeader><DialogTitle>Create Price Alert</DialogTitle></DialogHeader>
   206→              <div className="grid gap-3 py-2">
   207→                <div><Label className="text-xs">Symbol</Label>
   208→                  <Input value={form.symbol} onChange={e => setForm({ ...form, symbol: e.target.value.toUpperCase() })} placeholder="RELIANCE" className="h-8 text-xs" /></div>
   209→                <div><Label className="text-xs">Condition</Label>
   210→                  <Select value={form.condition} onValueChange={v => setForm({ ...form, condition: v })}>
   211→                    <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   212→                    <SelectContent><SelectItem value="ABOVE">Price goes ABOVE</SelectItem><SelectItem value="BELOW">Price goes BELOW</SelectItem></SelectContent>
   213→                  </Select></div>
   214→                <div><Label className="text-xs">Target Price (₹)</Label>
   215→                  <Input type="number" value={form.targetPrice} onChange={e => setForm({ ...form, targetPrice: e.target.value })} placeholder="2500" className="h-8 text-xs" /></div>
   216→              </div>
   217→              <DialogFooter>
   218→                <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
   219→                <Button size="sm" onClick={createAlert}>Create</Button>
   220→              </DialogFooter>
   221→            </DialogContent>
   222→          </Dialog>
   223→        </div>
   224→      </div>
   225→      {alerts.length === 0 ? (
   226→        <p className="text-xs text-muted-foreground text-center py-4">No alerts set. Create one to get notified when a price target is hit.</p>
   227→      ) : (
   228→        <ScrollArea className="max-h-[200px]">
   229→          <div className="space-y-1.5 pr-2">
   230→            {alerts.slice(0, 20).map(a => (
   231→              <div key={a.id} className="flex items-center gap-2 text-xs py-1.5 border-b border-border/30">
   232→                <div className={cn('h-2 w-2 rounded-full shrink-0', a.triggered ? 'bg-amber-400' : 'bg-blue-400')} />
   233→                <span className="font-semibold">{a.symbol}</span>

   236→                {a.triggered && <Badge className="text-[9px] h-4 bg-amber-500/20 text-amber-400">HIT ₹{a.triggeredPrice}</Badge>}
   237→                <button onClick={() => deleteAlert(a.id)} className="ml-auto text-muted-foreground hover:text-red-400"><X className="h-3 w-3" /></button>
   238→              </div>
   239→            ))}
   240→          </div>
   241→        </ScrollArea>

   243→    </div>

   531→// ── Main Analytics Tab ─────────────────────────────────
   532→export function AnalyticsTab() {
   533→  const [trades, setTrades] = useState<any[]>([]);
   534→  const [analyticsData, setAnalyticsData] = useState<any>(null);
   535→  const [loading, setLoading] = useState(true);
   536→
   537→  const fetchAnalytics = useCallback(async () => {
   538→    setLoading(true);
   539→    try {
   540→      const res = await fetch('/api/trades');
   541→      const json = await res.json();
   542→      const allTrades = json.trades || [];
   543→      const closed = allTrades.filter((t: any) => t.status === 'CLOSED' && t.pnl != null);
   544→
   545→      const wins = closed.filter((t: any) => t.pnl > 0);
   546→      const totalPnL = closed.reduce((s: number, t: any) => s + t.pnl, 0);
   547→      const grossProfit = wins.reduce((s: number, t: any) => s + t.pnl, 0);
   548→      const grossLoss = Math.abs(closed.filter((t: any) => t.pnl <= 0).reduce((s: number, t: any) => s + t.pnl, 0));
   549→
   550→      const monthMap = new Map<string, { pnl: number; trades: number }>();
   551→      for (const t of closed) {
   552→        const d = new Date(t.exitDate);
   553→        const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   554→        const prev = monthMap.get(key) || { pnl: 0, trades: 0 };
   555→        prev.pnl += t.pnl;
   556→        prev.trades += 1;
   557→        monthMap.set(key, prev);
   558→      }
   559→      const byMonth = Array.from(monthMap.entries()).map(([month, v]) => ({
   560→        month, pnl: Math.round(v.pnl), trades: v.trades,
   561→      })).sort((a, b) => a.month.localeCompare(b.month));
   562→
   563→      const buckets = [
   564→        { range: '< -5%', min: -Infinity, max: -5 }, { range: '-5% to -2%', min: -5, max: -2 },
   565→        { range: '-2% to 0%', min: -2, max: 0 }, { range: '0% to 2%', min: 0, max: 2 },
   566→        { range: '2% to 5%', min: 2, max: 5 }, { range: '> 5%', min: 5, max: Infinity },
   567→      ];
   568→      const pnlDistribution = buckets.map(b => ({
   569→        range: b.range,
   570→        count: closed.filter((t: any) => t.pnlPercent >= b.min && t.pnlPercent < b.max).length,
   571→      }));
   572→
   573→      let peak = 0, running = 0, maxDD = 0;
   574→      for (const t of closed.sort((a: any, b: any) => new Date(a.exitDate).getTime() - new Date(b.exitDate).getTime())) {
   575→        running += t.pnl;
   576→        peak = Math.max(peak, running);
   577→        maxDD = Math.max(maxDD, (peak - running) / peak * 100);
   578→      }
   579→
   580→      setAnalyticsData({
   581→        trades: closed, totalPnL,
   582→        winRate: closed.length > 0 ? (wins.length / closed.length) * 100 : 0,
   583→        avgWin: wins.length > 0 ? grossProfit / wins.length : 0,
   584→        avgLoss: closed.length - wins.length > 0 ? grossLoss / (closed.length - wins.length) : 0,
   585→        profitFactor: grossLoss > 0 ? grossProfit / grossLoss : 0,
   586→        maxDD, byMonth, pnlDistribution,
   587→      });
   588→    } catch { /* ignore */ }
   589→    finally { setLoading(false); }
   590→  }, []);
   591→
   592→  useEffect(() => { fetchAnalytics(); }, [fetchAnalytics]);
   593→
   594→  if (loading) return <div className="flex items-center justify-center py-16"><div className="text-center text-muted-foreground"><RefreshCw className="h-6 w-6 mx-auto mb-2 animate-spin" /><p className="text-sm">Loading analytics...</p></div></div>;
   595→
   596→  const COLORS = ['#10b981', '#f59e0b', '#6366f1', '#ef4444', '#06b6d4', '#ec4899'];
   597→
   598→  return (
   599→    <div className="space-y-4">
   600→      <div className="flex items-center justify-between">
   601→        <h2 className="text-lg font-bold flex items-center gap-2"><BarChart3 className="h-5 w-5 text-indigo-400" /> Analytics</h2>
   602→        <Button variant="outline" size="sm" onClick={fetchAnalytics} className="gap-2 h-8 text-xs"><RefreshCw className="h-3.5 w-3.5" /> Refresh</Button>
   603→      </div>
   604→
   605→      <Tabs defaultValue="overview" className="w-full">
   606→        <TabsList className="w-full justify-start mb-4">
   607→          <TabsTrigger value="overview" className="text-xs">Overview</TabsTrigger>
   608→          <TabsTrigger value="risk" className="text-xs">Risk Metrics</TabsTrigger>
   609→          <TabsTrigger value="benchmark" className="text-xs">Benchmark</TabsTrigger>
   610→          <TabsTrigger value="alerts" className="text-xs">Alerts</TabsTrigger>
   611→          <TabsTrigger value="tax" className="text-xs">Capital Gains</TabsTrigger>
   612→          <TabsTrigger value="rebalance" className="text-xs">Rebalance</TabsTrigger>

   617→        <TabsContent value="overview">
   618→          {analyticsData && analyticsData.trades.length > 0 ? (
   619→            <div className="space-y-4">
   620→              <div className="grid grid-cols-2 lg:grid-cols-3 xl:grid-cols-6 gap-3">
   621→                <Card><CardContent className="p-3">
