    94→    94→function scorePositionHealth(
    95→    95→  cp: number, entryPrice: number, sl: number, tp: number,
    96→    96→  holdingDays: number, maxHoldingDays: number, rMultiple: number
    97→    97→): HealthScore {
    98→    98→  const factors: { name: string; impact: number; detail: string }[] = [];
    99→    99→  let score = 50; // Base score
   100→   100→
   101→   101→  // Factor 1: R-multiple profit (±25 points)
   102→   102→  if (rMultiple >= 2) { score += 25; factors.push({ name: 'Profit', impact: 25, detail: `Strong ${rMultiple.toFixed(1)}R` }); }
   103→   103→  else if (rMultiple >= 1) { score += 15; factors.push({ name: 'Profit', impact: 15, detail: `Moderate ${rMultiple.toFixed(1)}R` }); }
   104→   104→  else if (rMultiple >= 0) { score += 5; factors.push({ name: 'Profit', impact: 5, detail: `Marginal ${rMultiple.toFixed(1)}R` }); }
   105→   105→  else if (rMultiple >= -0.5) { score -= 10; factors.push({ name: 'Loss', impact: -10, detail: `Small loss ${rMultiple.toFixed(1)}R` }); }
   106→   106→  else { score -= 25; factors.push({ name: 'Loss', impact: -25, detail: `Large loss ${rMultiple.toFixed(1)}R` }); }
   107→   107→
   108→   108→  // Factor 2: SL proximity (±20 points)
   109→   109→  const risk = entryPrice - sl;
   110→   110→  const slDistPct = risk > 0 ? ((cp - sl) / risk) * 100 : 100;
   111→   111→  if (slDistPct < 20) { score -= 20; factors.push({ name: 'SL Proximity', impact: -20, detail: `${slDistPct.toFixed(0)}% to SL` }); }
   112→   112→  else if (slDistPct < 40) { score -= 10; factors.push({ name: 'SL Proximity', impact: -10, detail: `${slDistPct.toFixed(0)}% to SL` }); }
   113→   113→  else { score += 5; factors.push({ name: 'SL Distance', impact: 5, detail: `Safe at ${slDistPct.toFixed(0)}%` }); }
   114→   114→
   115→   115→  // Factor 3: Age penalty (±15 points)
   116→   116→  const agePct = maxHoldingDays > 0 ? holdingDays / maxHoldingDays : 0;
   117→   117→  if (agePct >= 0.9) { score -= 15; factors.push({ name: 'Age', impact: -15, detail: `${holdingDays}/${maxHoldingDays}d (expired)` }); }
   118→   118→  else if (agePct >= 0.7) { score -= 8; factors.push({ name: 'Age', impact: -8, detail: `${holdingDays}/${maxHoldingDays}d (aging)` }); }
   119→   119→  else if (agePct < 0.3) { score += 5; factors.push({ name: 'Age', impact: 5, detail: `Fresh ${holdingDays}d` }); }
   120→   120→
   121→   121→  // Factor 4: Trend to TP (±10 points)
   122→   122→  if (tp > entryPrice && cp > entryPrice) {
   123→   123→    const tpProgress = ((cp - entryPrice) / (tp - entryPrice)) * 100;
   124→   124→    if (tpProgress > 75) { score += 10; factors.push({ name: 'TP Progress', impact: 10, detail: `${tpProgress.toFixed(0)}% to TP` }); }
   125→   125→    else if (tpProgress > 40) { score += 5; factors.push({ name: 'TP Progress', impact: 5, detail: `${tpProgress.toFixed(0)}% to TP` }); }
   126→   126→  }
   127→   127→
   128→   128→  // Clamp score
   129→   129→  score = Math.max(0, Math.min(100, score));
   130→   130→
   131→   131→  let urgency: HealthScore['urgency'] = 'OK';
   132→   132→  if (score < 30) urgency = 'CRITICAL';
   133→   133→  else if (score < 55) urgency = 'WARNING';
   134→   134→
   135→   135→  return { score, factors, urgency };
   136→   136→}

   139→   139→function isMarketHours(): boolean {
   140→   140→  const now = new Date();
   141→   141→  const ist = new Date(now.getTime() + (5.5 * 60 * 60 * 1000) + (now.getTimezoneOffset() * 60 * 1000));
   142→   142→  const day = ist.getDay();
   143→   143→  if (day === 0 || day === 6) return false;
   144→   144→  const h = ist.getHours();
   145→   145→  const m = ist.getMinutes();
   146→   146→  const mins = h * 60 + m;
   147→   147→  return mins >= 555 && mins <= 930;
   148→   148→}

   239→   239→async function recalcWallet() {
   240→   240→  const w = await getWallet();
   241→   241→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   242→   242→  const deployed = open.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   243→   243→  const total = w.totalCapital + w.realizedPnl;
   244→   244→  let unrealizedPnl = 0;
   245→   245→  const syms = [...new Set(open.map(t => t.symbol))];
   246→   246→  for (const sym of syms) {
   247→   247→    try {
   248→   248→      const { price } = await getCurrentPrice(sym);
   249→   249→      for (const t of open.filter(t => t.symbol === sym)) {
   250→   250→        unrealizedPnl += (price - t.entryPrice) * t.qty;
   251→   251→      }
   252→   252→    } catch { /* skip */ }
   253→   253→  }
   254→   254→  await db.capitalWallet.update({ where: { id: w.id }, data: { deployed, available: total - deployed, unrealizedPnl, totalCapital: total } });
   255→   255→  return { ...w, deployed, available: total - deployed, unrealizedPnl, totalCapital: total, initialCapital: w.initialCapital };
   256→   256→}

   275→   275→async function getPortfolioDrawdown(): Promise<{ drawdownPct: number; peakCapital: number; currentCapital: number }> {
   276→   276→  const w = await getWallet();
   277→   277→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   278→   278→  let unrealizedPnl = 0;
   279→   279→  const syms = [...new Set(open.map(t => t.symbol))];
   280→   280→  for (const sym of syms) {
   281→   281→    try {
   282→   282→      const { price } = await getCurrentPrice(sym);
   283→   283→      for (const t of open.filter(t => t.symbol === sym)) {
   284→   284→        unrealizedPnl += (price - t.entryPrice) * t.qty;
   285→   285→      }
   286→   286→    } catch { /* skip */ }
   287→   287→  }
   288→   288→
   289→   289→  const currentCapital = w.totalCapital + unrealizedPnl;
   290→   290→  // Peak = highest totalCapital value we've seen (initial + best realized)
   291→   291→  const peakCapital = Math.max(w.initialCapital, w.totalCapital + w.realizedPnl, currentCapital);
   292→   292→  const drawdownPct = peakCapital > 0 ? ((peakCapital - currentCapital) / peakCapital) * 100 : 0;
   293→   293→  return { drawdownPct, peakCapital, currentCapital };
   294→   294→}

   310→   310→function getAdaptiveFactor(consecutiveLosses: number, rules: PositionRules): number {
   311→   311→  if (!rules.adaptiveSizing) return 1.0;
   312→   312→  if (consecutiveLosses < 3) return 1.0;
   313→   313→  // Reduce by streakPenaltyPct for each loss beyond 2, min 20% of original
   314→   314→  const reductions = consecutiveLosses - 2;
   315→   315→  const factor = 1.0 - (reductions * (rules.streakPenaltyPct / 100));
   316→   316→  return Math.max(0.2, Math.min(1.0, factor));
   317→   317→}

   320→   320→async function getNiftyRegime(niftyData: any[]): Promise<{ regime: 'BULLISH' | 'BEARISH' | 'UNKNOWN'; ema200: number; currentClose: number }> {
   321→   321→  if (niftyData.length < 200) return { regime: 'UNKNOWN', ema200: 0, currentClose: 0 };
   322→   322→  const closes = niftyData.map(c => c.close);
   323→   323→  const ema200Arr = EMA.calculate({ period: 200, values: closes });
   324→   324→  if (ema200Arr.length === 0) return { regime: 'UNKNOWN', ema200: 0, currentClose: 0 };
   325→   325→  const ema200 = ema200Arr[ema200Arr.length - 1];
   326→   326→  const currentClose = closes[closes.length - 1];
   327→   327→  return {
   328→   328→    regime: currentClose > ema200 ? 'BULLISH' : 'BEARISH',
   329→   329→    ema200: Math.round(ema200 * 100) / 100,
   330→   330→    currentClose: Math.round(currentClose * 100) / 100,
   331→   331→  };
   332→   332→}

   335→   335→async function getATRTrailLevel(symbol: string, entryPrice: number, currentPrice: number, risk: number, rules: PositionRules): Promise<{ useATR: boolean; trailLevel: number; atr: number }> {
   336→   336→  if (rules.atrTrailMultiplier <= 0) {
   337→   337→    // Use fixed R-based trail (v1 behavior)
   338→   338→    const trailLevel = entryPrice + (risk * rules.trailToR);
   339→   339→    return { useATR: false, trailLevel, atr: 0 };
   340→   340→  }
   341→   341→
   342→   342→  try {
   343→   343→    const { data } = await getHistoricalData(symbol, 30);
   344→   344→    if (data.length < 14) return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   345→   345→
   346→   346→    const atrArr = ATR.calculate({
   347→   347→      period: 14,
   348→   348→      high: data.map(c => c.high),
   349→   349→      low: data.map(c => c.low),
   350→   350→      close: data.map(c => c.close),
   351→   351→    });
   352→   352→    if (atrArr.length === 0) return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   353→   353→
   354→   354→    const atr = atrArr[atrArr.length - 1];
   355→   355→    // Trail level = highest close seen - ATR * multiplier (stored in notes, or use current price as proxy)
   356→   356→    // Since we don't track the high-water mark in DB, use: max(current, entry) - ATR * multiplier
   357→   357→    const highWaterMark = Math.max(currentPrice, entryPrice);
   358→   358→    const trailLevel = highWaterMark - (atr * rules.atrTrailMultiplier);
   359→   359→    return { useATR: true, trailLevel: Math.round(trailLevel * 100) / 100, atr: Math.round(atr * 100) / 100 };
   360→   360→  } catch {
   361→   361→    return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   362→   362→  }
   363→   363→}

   366→   366→async function canOpen(symbol: string, entryPrice: number, qty: number, rules: PositionRules, sector?: string): Promise<{ ok: boolean; reason: string }> {
   367→   367→  const w = await getWallet();
   368→   368→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   369→   369→  const now = new Date();
   370→   370→
   371→   371→  // === v2: Circuit Breaker Check ===
   372→   372→  const { drawdownPct } = await getPortfolioDrawdown();
   373→   373→  if (drawdownPct >= rules.maxDrawdownPct) {
   374→   374→    return { ok: false, reason: `CIRCUIT BREAKER: Drawdown ${drawdownPct.toFixed(1)}% >= ${rules.maxDrawdownPct}%` };
   375→   375→  }
   376→   376→
   377→   377→  // === v2: Daily Loss Limit ===
   378→   378→  await resetDailyCounters();
   379→   379→  const st = await getSchedulerState();
   380→   380→  if (st.todayPnl < 0 && Math.abs(st.todayPnl) >= rules.dailyLossLimit) {
   381→   381→    return { ok: false, reason: `DAILY LOSS LIMIT: ₹${Math.abs(st.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}` };
   382→   382→  }
   383→   383→
   384→   384→  // Max positions check
   385→   385→  if (open.length >= rules.maxTotalPositions) return { ok: false, reason: `Max ${rules.maxTotalPositions} positions` };
   386→   386→
   387→   387→  // Per-stock cap
   388→   388→  const stockOpen = open.filter(t => t.symbol === symbol);
   389→   389→  const stockDep = stockOpen.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   390→   390→  if (stockDep + entryPrice * qty > rules.maxPerStock) return { ok: false, reason: `Max ₹${rules.maxPerStock}/stock` };
   391→   391→
   392→   392→  // Duplicate position
   393→   393→  if (stockOpen.length > 0) return { ok: false, reason: 'Open position exists' };
   394→   394→
   395→   395→  // Monthly cap
   396→   396→  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   397→   397→  const monthTrades = await db.paperTrade.count({ where: { symbol, entryDate: { gte: monthStart } } });
   398→   398→  if (monthTrades >= rules.maxBuysPerMonth) return { ok: false, reason: `Max ${rules.maxBuysPerMonth} buys/month` };
   399→   399→
   400→   400→  // Capital check
   401→   401→  if (entryPrice * qty > w.available) return { ok: false, reason: 'Insufficient capital' };
   402→   402→
   403→   403→  // Cooldown check
   404→   404→  if (rules.cooldownDays > 0) {
   405→   405→    const lastExit = await db.paperTrade.findFirst({
   406→   406→      where: { symbol, status: 'CLOSED', exitDate: { not: null } },
   407→   407→      orderBy: { exitDate: 'desc' },
   408→   408→    });
   409→   409→    if (lastExit?.exitDate) {
   410→   410→      const daysSinceExit = (now.getTime() - new Date(lastExit.exitDate).getTime()) / 86400000;
   411→   411→      if (daysSinceExit < rules.cooldownDays) {
   412→   412→        return { ok: false, reason: `Cooldown: ${Math.ceil(rules.cooldownDays - daysSinceExit)}d remaining` };
   413→   413→      }
   414→   414→    }
   415→   415→  }
   416→   416→
   417→   417→  // Sector concentration check
   418→   418→  if (sector && rules.maxSectorPct > 0) {
   419→   419→    const sectorAlloc = await getSectorAllocation();
   420→   420→    const currentPct = sectorAlloc[sector] || 0;
   421→   421→    const newPct = currentPct + ((entryPrice * qty) / w.totalCapital) * 100;
   422→   422→    if (newPct > rules.maxSectorPct) {
   423→   423→      return { ok: false, reason: `Sector cap: ${sector} at ${currentPct.toFixed(1)}% (max ${rules.maxSectorPct}%)` };
   424→   424→    }
   425→   425→  }
   426→   426→
   427→   427→  return { ok: true };
   428→   428→}

   561→   561→// ── Core Engine: Check Exits (v2 with health scoring) ─
   562→   562→async function autoCheckExits() {
   563→   563→  const rules = await getRules();
   564→   564→  const openTrades = await db.paperTrade.findMany({ where: { status: 'OPEN' }, orderBy: { entryDate: 'asc' } });
   565→   565→  const exits: any[] = [];
   566→   566→  const holding: any[] = [];
   567→   567→  const partialBooks: any[] = [];
   568→   568→  const warnings: any[] = [];
   569→   569→
   570→   570→  const minsToClose = timeToClose();
   571→   571→  const shouldTimeExit = minsToClose <= rules.timeExitMins && minsToClose > 0;
   572→   572→
   573→   573→  for (const trade of openTrades) {
   574→   574→    try {
   575→   575→      const { price: cp } = await getCurrentPrice(trade.symbol);
   576→   576→      if (cp <= 0) { holding.push({ symbol: trade.symbol, error: 'Price unavailable' }); continue; }
   577→   577→
   578→   578→      const days = Math.floor((Date.now() - new Date(trade.entryDate).getTime()) / 86400000);
   579→   579→      const risk = trade.entryPrice - trade.stopLoss;
   580→   580→      const rMultiple = risk > 0 ? (cp - trade.entryPrice) / risk : 0;
   581→   581→      let shouldExit = false;
   582→   582→      let reason = '';
   583→   583→      let exitPrice = cp;
   584→   584→      let exitQty = trade.qty;
   585→   585→
   586→   586→      // 1. Hard SL hit
   587→   587→      if (cp <= trade.stopLoss) {
   588→   588→        shouldExit = true; reason = 'SL_HIT'; exitPrice = trade.stopLoss;
   589→   589→      }
   590→   590→      // 2. Hard TP hit
   591→   591→      else if (cp >= trade.targetPrice) {
   592→   592→        shouldExit = true; reason = 'TP_HIT'; exitPrice = trade.targetPrice;
   593→   593→      }
   594→   594→      // 3. Trailing stop (v2: ATR-based if configured)
   595→   595→      else if (risk > 0 && rMultiple >= rules.trailingStopR) {
   596→   596→        const trail = await getATRTrailLevel(trade.symbol, trade.entryPrice, cp, risk, rules);
   597→   597→        if (trail.useATR) {
   598→   598→          // ATR trailing: only exit if price drops below ATR-based level
   599→   599→          if (cp <= trail.trailLevel) {
   600→   600→            shouldExit = true; reason = 'ATR_TRAIL_STOP'; exitPrice = trail.trailLevel;
   601→   601→          }
   602→   602→        } else {
   603→   603→          // Fixed R trailing (v1 behavior)
   604→   604→          const trailLevel = trade.entryPrice + (risk * rules.trailToR);
   605→   605→          if (cp <= trailLevel) {
   606→   606→            shouldExit = true; reason = 'TRAIL_STOP'; exitPrice = trailLevel;
   607→   607→          }
   608→   608→        }
   609→   609→      }
   610→   610→      // 4. Max holding days
   611→   611→      else if (days >= rules.maxHoldingDays) {
   612→   612→        shouldExit = true; reason = 'MAX_HOLDING_DAYS';
   613→   613→      }
   614→   614→      // 5. v2: Stale loser acceleration — exit aging losing positions faster
   615→   615→      else if (days >= rules.maxHoldingDays * 0.7 && rMultiple < -0.3) {
   616→   616→        shouldExit = true; reason = 'STALE_LOSER';
   617→   617→      }
   618→   618→      // 6. Time-based exit
   619→   619→      else if (shouldTimeExit) {
   620→   620→        if (rMultiple < 0.5) {
   621→   621→          shouldExit = true; reason = 'TIME_EXIT'; exitPrice = cp;
   622→   622→        }
   623→   623→      }
   624→   624→
   625→   625→      // 7. Partial booking (doesn't trigger full exit)
   626→   626→      if (!shouldExit && risk > 0 && rMultiple >= rules.partialBookR && trade.qty > 1) {
   627→   627→        const alreadyBooked = trade.tags?.includes('partial-booked');
   628→   628→        if (!alreadyBooked) {
   629→   629→          const bookQty = Math.max(1, Math.floor(trade.qty * (rules.partialBookPct / 100)));
   630→   630→          const remainingQty = trade.qty - bookQty;
   631→   631→          if (remainingQty >= 1) {
   632→   632→            const bookPnl = (cp - trade.entryPrice) * bookQty;
   633→   633→            await db.paperTrade.update({
   634→   634→              where: { id: trade.id },
   635→   635→              data: { qty: remainingQty, tags: (trade.tags ? trade.tags + ',' : '') + 'partial-booked' },
   636→   636→            });
   637→   637→            const w = await getWallet();
   638→   638→            await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + bookPnl } });
   639→   639→            await db.autoTradeLog.create({
   640→   640→              action: 'PARTIAL_BOOK', symbol: trade.symbol, tradeId: trade.id,
   641→   641→              executed: true, reason: `Booked ${bookQty}/${trade.qty + bookQty} at ${rMultiple.toFixed(1)}R | P&L: ₹${bookPnl.toLocaleString()}`,
   642→   642→            });
   643→   643→            partialBooks.push({ symbol: trade.symbol, bookedQty: bookQty, remainingQty, rMultiple, pnl: bookPnl, exitPrice: cp });
   644→   644→          }
   645→   645→        }
   646→   646→      }
   647→   647→
   648→   648→      // Execute full exit
   649→   649→      if (shouldExit) {
   650→   650→        const pnl = (exitPrice - trade.entryPrice) * exitQty;
   651→   651→        const pnlPct = ((exitPrice - trade.entryPrice) / trade.entryPrice) * 100;
   652→   652→        await db.paperTrade.update({
   653→   653→          where: { id: trade.id },
   654→   654→          data: { status: 'CLOSED', exitDate: new Date(), exitPrice, pnl, pnlPercent: pnlPct, exitReason: reason },
   655→   655→        });
   656→   656→        const w = await getWallet();
   657→   657→        await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + pnl } });
   658→   658→        await db.autoTradeLog.create({
   659→   659→          action: `AUTO_EXIT_${reason}`, symbol: trade.symbol, tradeId: trade.id,
   660→   660→          executed: true, reason: `${reason} | ₹${trade.entryPrice}→₹${exitPrice} | P&L: ₹${pnl.toLocaleString()} | ${rMultiple.toFixed(1)}R`,
   661→   661→        });
   662→   662→
   663→   663→        // v2: Track consecutive losses for adaptive sizing
   664→   664→        if (pnl < 0) {
   665→   665→          const newStreak = (await getSchedulerState()).consecutiveLosses + 1;
   666→   666→          await setSchedulerKV('sched_consecutiveLosses', String(newStreak));
   667→   667→        } else {
   668→   668→          await setSchedulerKV('sched_consecutiveLosses', '0');
   669→   669→          // Reset adaptive factor on win
   670→   670→          await setSchedulerKV('sched_lastAdaptiveFactor', '1');
   671→   671→        }
   672→   672→
   673→   673→        await resetDailyCounters();
   674→   674→        const st2 = await getSchedulerState();
   675→   675→        await setSchedulerKV('sched_todayExits', String(st2.todayExits + 1));
   676→   676→        await setSchedulerKV('sched_todayPnl', String(st2.todayPnl + pnl));
   677→   677→
   678→   678→        // v2: Check if we need to activate daily loss circuit breaker
   679→   679→        const updatedSt = await getSchedulerState();
   680→   680→        if (updatedSt.todayPnl < 0 && Math.abs(updatedSt.todayPnl) >= rules.dailyLossLimit) {
   681→   681→          await setSchedulerKV('sched_circuitBreaker', 'true');
   682→   682→          await setSchedulerKV('sched_circuitBreakerReason', `Daily loss ₹${Math.abs(updatedSt.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}`);
   683→   683→        }
   684→   684→
   685→   685→        exits.push({
   686→   686→          symbol: trade.symbol, exitReason: reason, entryPrice: trade.entryPrice, exitPrice,
   687→   687→          pnl, pnlPercent: pnlPct, holdingDays: days, rMultiple: rMultiple.toFixed(2),
   688→   688→        });
   689→   689→      } else {
   690→   690→        // v2: Position Health Score
   691→   691→        const health = scorePositionHealth(cp, trade.entryPrice, trade.stopLoss, trade.targetPrice, days, rules.maxHoldingDays, rMultiple);
   692→   692→        const healthData: any = {
   693→   693→          symbol: trade.symbol, entryPrice: trade.entryPrice, currentPrice: cp,
   694→   694→          pnl: (cp - trade.entryPrice) * trade.qty,
   695→   695→          pnlPercent: ((cp - trade.entryPrice) / trade.entryPrice) * 100,
   696→   696→          holdingDays: days, maxHoldingDays: rules.maxHoldingDays,
   697→   697→          sl: trade.stopLoss, tp: trade.targetPrice, rMultiple: rMultiple.toFixed(2),
   698→   698→          healthScore: health.score, healthUrgency: health.urgency, healthFactors: health.factors,
   699→   699→        };
   700→   700→
   701→   701→        if (health.urgency === 'CRITICAL') {
   702→   702→          healthData.slWarning = true;
   703→   703→          warnings.push({ ...healthData, type: health.score < 20 ? 'SL_PROXIMITY' : 'AGING', message: `${trade.symbol} health ${health.score}/100 (${health.urgency})` });
   704→   704→        } else if (health.urgency === 'WARNING') {
   705→   705→          warnings.push({ ...healthData, type: 'AGING', message: `${trade.symbol} health ${health.score}/100` });
   706→   706→        }
   707→   707→
   708→   708→        holding.push(healthData);
   709→   709→      }
   710→   710→    } catch { holding.push({ symbol: trade.symbol, error: 'Price fetch failed' }); }
   711→   711→  }
   712→   712→
   713→   713→  if (exits.length > 0 || partialBooks.length > 0) await recalcWallet();
   714→   714→  return { exits, holding, partialBooks, warnings, timeToClose: minsToClose, marketHours: isMarketHours() };
   715→   715→}

   717→   717→// ── Scheduler Control ──────────────────────────────────
   718→   718→async function runSchedulerTick() {
   719→   719→  if (!isMarketHours()) return { marketHours: false, message: 'Market closed' };
   720→   720→  const state = await getSchedulerState();
   721→   721→  if (!state.enabled) return { marketHours: true, enabled: false };
   722→   722→
   723→   723→  await resetDailyCounters();
   724→   724→  const now = new Date();
   725→   725→  const results: any = { scanResult: null, exitResult: null };
   726→   726→
   727→   727→  if (state.nextScanAt) {
   728→   728→    const nextScan = new Date(state.nextScanAt);
   729→   729→    if (now >= nextScan) {
   730→   730→      const config: ScreeningConfig = DEFAULT_CONFIG;
   731→   731→      results.scanResult = await autoScanAndTrade(config);
   732→   732→      await setSchedulerKV('sched_lastScanAt', now.toISOString());
   733→   733→      const next = new Date(now.getTime() + state.scanIntervalMin * 60000);
   734→   734→      await setSchedulerKV('sched_nextScanAt', next.toISOString());
   735→   735→    }
   736→   736→  }
   737→   737→
   738→   738→  if (state.nextExitAt) {
   739→   739→    const nextExit = new Date(state.nextExitAt);
   740→   740→    if (now >= nextExit) {
   741→   741→      results.exitResult = await autoCheckExits();
   742→   742→      await setSchedulerKV('sched_lastExitAt', now.toISOString());
   743→   743→      const next = new Date(now.getTime() + state.exitIntervalMin * 60000);
   744→   744→      await setSchedulerKV('sched_nextExitAt', next.toISOString());
   745→   745→    }
   746→   746→  }
   747→   747→
   748→   748→  return { ...results, marketHours: true, enabled: true, schedulerState: await getSchedulerState() };
   749→   749→}

   752→   752→export async function GET() {
   753→   753→  try {
   754→   754→    const wallet = await recalcWallet();
   755→   755→    const openTrades = await db.paperTrade.findMany({ where: { status: 'OPEN', autoTraded: true }, orderBy: { entryDate: 'desc' } });
   756→   756→    const recentLogs = await db.autoTradeLog.findMany({ orderBy: { createdAt: 'desc' }, take: 50 });
   757→   757→    const rules = await getRules();
   758→   758→    const scheduler = await getSchedulerState();
   759→   759→    const sectorAlloc = await getSectorAllocation();
   760→   760→    const { drawdownPct, peakCapital, currentCapital } = await getPortfolioDrawdown();
   761→   761→    return NextResponse.json({
   762→   762→      success: true, wallet, rules, openTrades, recentLogs, scheduler, sectorAllocation: sectorAlloc,
   763→   763→      marketHours: isMarketHours(), timeToClose: timeToClose(),
   764→   764→      // v2 data
   765→   765→      drawdown: { drawdownPct: Math.round(drawdownPct * 100) / 100, peakCapital, currentCapital },
   766→   766→      consecutiveLosses: scheduler.consecutiveLosses,
   767→   767→      adaptiveFactor: scheduler.lastAdaptiveFactor,
   768→   768→    });
   769→   769→  } catch (error) {
   770→   770→    return NextResponse.json({ success: false, error: String(error) }, { status: 500 });
   771→   771→  }

   774→   774→export async function POST(request: NextRequest) {
   775→   775→  try {
   776→   776→    const body = await request.json();
   777→   777→    const action: string = body.action || 'get_status';
   778→   778→    const config: ScreeningConfig = { ...DEFAULT_CONFIG, ...body.config };
   779→   779→
   780→   780→    switch (action) {
   781→   781→      case 'scan_and_trade': {
   782→   782→        const result = await autoScanAndTrade(config);
   783→   783→        return NextResponse.json({ success: true, action, ...result });
   784→   784→      }
   785→   785→      case 'check_exits': {
   786→   786→        const result = await autoCheckExits();
   787→   787→        return NextResponse.json({ success: true, action, ...result });
   788→   788→      }
   789→   789→      case 'scheduler_tick': {
   790→   790→        const result = await runSchedulerTick();
   791→   791→        return NextResponse.json({ success: true, action, ...result });
   792→   792→      }
   793→   793→      case 'scheduler_toggle': {
   794→   794→        const enabled = body.enabled;
   795→   795→        if (enabled) {
   796→   796→          const now = new Date();
   797→   797→          const state = await getSchedulerState();
   798→   798→          const scanMin = body.scanIntervalMin || state.scanIntervalMin;
   799→   799→          const exitMin = body.exitIntervalMin || state.exitIntervalMin;
   800→   800→          await setSchedulerKV('sched_enabled', 'true');
   801→   801→          await setSchedulerKV('sched_scanIntervalMin', String(scanMin));
   802→   802→          await setSchedulerKV('sched_exitIntervalMin', String(exitMin));
   803→   803→          await setSchedulerKV('sched_nextScanAt', now.toISOString());
   804→   804→          await setSchedulerKV('sched_nextExitAt', now.toISOString());
   805→   805→          await resetDailyCounters();
   806→   806→        } else {
   807→   807→          await setSchedulerKV('sched_enabled', 'false');
   808→   808→          await setSchedulerKV('sched_nextScanAt', '');
   809→   809→          await setSchedulerKV('sched_nextExitAt', '');
   810→   810→        }
   811→   811→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   812→   812→      }
   813→   813→      case 'update_wallet': {
   814→   814→        const { totalCapital } = body;
   815→   815→        if (totalCapital) {
   816→   816→          const w = await getWallet();
   817→   817→          const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   818→   818→          const deployed = open.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   819→   819→          await db.capitalWallet.update({ where: { id: w.id }, data: { totalCapital, available: totalCapital - deployed } });
   820→   820→          const updated = await recalcWallet();
   821→   821→          return NextResponse.json({ success: true, wallet: updated });
   822→   822→        }
   823→   823→        break;
   824→   824→      }
   825→   825→      case 'update_rules': {
   826→   826→        const rules = body.rules;
   827→   827→        if (rules) {
   828→   828→          for (const [key, value] of Object.entries(rules)) {
   829→   829→            await db.appSettings.upsert({
   830→   830→              where: { key: `rules_${key}` },
   831→   831→              create: { key: `rules_${key}`, value: String(value) },
   832→   832→              update: { value: String(value) },
   833→   833→            });
   834→   834→          }
   835→   835→          return NextResponse.json({ success: true, rules: await getRules() });
   836→   836→        }
   837→   837→        break;
   838→   838→      }
   839→   839→      case 'reset_circuit_breaker': {
   840→   840→        // v2: Manual circuit breaker reset
   841→   841→        await setSchedulerKV('sched_circuitBreaker', 'false');
   842→   842→        await setSchedulerKV('sched_circuitBreakerReason', '');
   843→   843→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   844→   844→      }
   845→   845→    }
   846→   846→    const wallet = await recalcWallet();
   847→   847→    return NextResponse.json({ success: true, wallet });
   848→   848→  } catch (error) {
   849→   849→    console.error('Auto-trade error:', error);
   850→   850→    return NextResponse.json({ success: false, error: String(error) }, { status: 500 });
   851→   851→  }
