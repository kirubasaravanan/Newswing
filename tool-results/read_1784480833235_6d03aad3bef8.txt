   524→/**
   525→ * Returns the full NSE universe, deduplicated by symbol.
   526→ * Total: ~600+ unique tradeable stocks across all segments.
   527→ */
   528→export function getFullNSEUniverse(): NSEStock[] {
   529→  const seen = new Set<string>();
   530→  const universe: NSEStock[] = [];
   531→
   532→  const all = [...NIFTY_50, ...NIFTY_100_EXTRA, ...FNO_EXTRA, ...MIDCAP_STOCKS, ...SMALLCAP_STOCKS];
   533→
   534→  for (const stock of all) {
   535→    // Normalize: uppercase, no spaces, no special chars
   536→    const key = stock.symbol.toUpperCase().replace(/[^A-Z0-9&]/g, '');
   537→    if (!seen.has(key) && key.length >= 2) {
   538→      seen.add(key);
   539→      universe.push({ ...stock, symbol: key });
   540→    }
   541→  }
   542→  return universe;
   543→}
   544→
   545→/**
   546→ * Get stocks by category
   547→ */
   548→export function getStocksByCategory(category?: NSEStock['category']): NSEStock[] {
   549→  const all = getFullNSEUniverse();
   550→  if (!category) return all;
   551→  return all.filter(s => s.category === category);
   552→}
   553→
   554→/**
   555→ * Get stock count by category
   556→ */
   557→export function getUniverseStats(): { total: number; byCategory: Record<string, number>; bySector: Record<string, number> } {
   558→  const all = getFullNSEUniverse();
   559→  const byCategory: Record<string, number> = {};
   560→  const bySector: Record<string, number> = {};
   561→
   562→  for (const s of all) {
   563→    byCategory[s.category] = (byCategory[s.category] || 0) + 1;
   564→    bySector[s.sector] = (bySector[s.sector] || 0) + 1;
   565→  }
   566→
   567→  return { total: all.length, byCategory, bySector };
   568→}
