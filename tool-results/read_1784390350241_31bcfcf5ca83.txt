    21→// ── Portfolio Manager ──────────────────────────────────
    22→function PortfolioManager() {
    23→  const [portfolios, setPortfolios] = useState<any[]>([]);
    24→  const [open, setOpen] = useState(false);
    25→  const [form, setForm] = useState({ name: '', description: '', totalCapital: '200000' });
    26→
    27→  const fetchPortfolios = useCallback(async () => {
    28→    try {
    29→      const res = await fetch('/api/portfolios');
    30→      const data = await res.json();
    31→      if (data.success) setPortfolios(data.portfolios);
    32→    } catch { /* ignore */ }
    33→  }, []);
    34→  useEffect(() => { fetchPortfolios(); }, [fetchPortfolios]);
    35→
    36→  const createPortfolio = async () => {
    37→    if (!form.name) return;
    38→    try {
    39→      const res = await fetch('/api/portfolios', {
    40→        method: 'POST', headers: { 'Content-Type': 'application/json' },
    41→        body: JSON.stringify({ name: form.name, description: form.description, totalCapital: parseFloat(form.totalCapital) }),
    42→      });
    43→      const data = await res.json();
    44→      if (data.success) {
    45→        toast.success(`Portfolio "${form.name}" created`);
    46→        setForm({ name: '', description: '', totalCapital: '200000' });
    47→        setOpen(false);
    48→        fetchPortfolios();
    49→      }
    50→    } catch { toast.error('Failed to create portfolio'); }
    51→  };
    52→
    53→  const deletePortfolio = async (id: string) => {
    54→    await fetch(`/api/portfolios?id=${id}`, { method: 'DELETE' });
    55→    fetchPortfolios();
    56→  };
    57→
    58→  return (
    59→    <div>
    60→      <div className="flex items-center justify-between mb-3">
    61→        <h3 className="text-sm font-semibold flex items-center gap-2"><Briefcase className="h-4 w-4" /> Portfolios</h3>
    62→        <Dialog open={open} onOpenChange={setOpen}>
    63→          <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> New Portfolio</Button></DialogTrigger>
    64→          <DialogContent className="sm:max-w-sm">
    65→            <DialogHeader><DialogTitle>Create Portfolio</DialogTitle></DialogHeader>
    66→            <div className="grid gap-3 py-2">
    67→              <div><Label className="text-xs">Name</Label><Input value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} placeholder="Aggressive Growth" className="h-8 text-xs" /></div>
    68→              <div><Label className="text-xs">Description</Label><Input value={form.description} onChange={e => setForm({ ...form, description: e.target.value })} placeholder="High risk, high reward" className="h-8 text-xs" /></div>
    69→              <div><Label className="text-xs">Capital (₹)</Label><Input type="number" value={form.totalCapital} onChange={e => setForm({ ...form, totalCapital: e.target.value })} className="h-8 text-xs" /></div>
    70→            </div>
    71→            <DialogFooter>
    72→              <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
    73→              <Button size="sm" onClick={createPortfolio}>Create</Button>
    74→            </DialogFooter>
    75→          </DialogContent>
    76→        </Dialog>
    77→      </div>
    78→      {portfolios.length === 0 ? (
    79→        <p className="text-xs text-muted-foreground text-center py-6">No portfolios. Create one to organize your trades.</p>
    80→      ) : (
    81→        <div className="grid gap-2">
    82→          {portfolios.map(p => (
    83→            <div key={p.id} className="flex items-center gap-3 rounded-lg bg-secondary/50 p-3">
    84→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-indigo-500/20 text-indigo-400 text-xs font-bold">
    85→                {p.name.charAt(0)}
    86→              </div>
    87→              <div className="flex-1 min-w-0">
    88→                <div className="text-sm font-semibold truncate">{p.name}</div>
    89→                <div className="text-[10px] text-muted-foreground">
    90→                  ₹{Math.round(p.totalCapital).toLocaleString()} &middot; {p.openPositions || 0} open
    91→                  {p.description && ` · ${p.description}`}
    92→                </div>
    93→              </div>
    94→              {p.isDefault && <Badge variant="outline" className="text-[9px] h-4 text-indigo-400">DEFAULT</Badge>}
    95→              <button onClick={() => deletePortfolio(p.id)} className="text-muted-foreground hover:text-red-400"><Trash2 className="h-3.5 w-3.5" /></button>
    96→            </div>
    97→          ))}
    98→        </div>
    99→      )}
   100→    </div>
   101→  );
   102→}

   311→// ── Capital & Config ───────────────────────────────────
   312→function CapitalConfig() {
   313→  const [wallet, setWallet] = useState<any>(null);
   314→  const [editing, setEditing] = useState(false);
   315→  const [capital, setCapital] = useState('200000');
   316→
   317→  const fetchWallet = useCallback(async () => {
   318→    try {
   319→      const res = await fetch('/api/portfolio/summary');
   320→      const data = await res.json();
   321→      if (data.success && data.summary?.wallet) {
   322→        setWallet(data.summary.wallet);
   323→        setCapital(String(data.summary.wallet.totalCapital));
   324→      }
   325→    } catch { /* ignore */ }
   326→  }, []);
   327→  useEffect(() => { fetchWallet(); }, [fetchWallet]);
   328→
   329→  const updateCapital = async () => {
   330→    try {
   331→      const res = await fetch('/api/portfolio/summary', {
   332→        method: 'PUT', headers: { 'Content-Type': 'application/json' },
   333→        body: JSON.stringify({ totalCapital: parseFloat(capital) }),
   334→      });
   335→      if (res.ok) { toast.success('Capital updated'); setEditing(false); fetchWallet(); }
   336→    } catch { toast.error('Failed to update capital'); }
   337→  };
   338→
   339→  if (!wallet) return <p className="text-xs text-muted-foreground text-center py-4">Loading...</p>;
   340→
   341→  return (
   342→    <div className="space-y-3">
   343→      <h3 className="text-sm font-semibold flex items-center gap-2"><Wallet className="h-4 w-4" /> Capital Configuration</h3>
   344→      <div className="grid grid-cols-2 gap-3">
   345→        <div className="rounded-lg bg-secondary/50 p-3">
   346→          <div className="text-[10px] text-muted-foreground uppercase">Total Capital</div>
   347→          {editing ? (
   348→            <div className="flex gap-1 mt-1">
   349→              <Input type="number" value={capital} onChange={e => setCapital(e.target.value)} className="h-7 text-xs" />
   350→              <Button size="sm" className="h-7 text-xs" onClick={updateCapital}>Save</Button>
   351→            </div>
   352→          ) : (
   353→            <div className="flex items-center gap-1 mt-1">
   354→              <span className="text-lg font-bold font-mono">₹{Math.round(wallet.totalCapital).toLocaleString()}</span>
   355→              <button onClick={() => setEditing(true)} className="text-muted-foreground hover:text-foreground"><RefreshCw className="h-3 w-3" /></button>
   356→            </div>
   357→          )}
   358→        </div>
   359→        <div className="rounded-lg bg-secondary/50 p-3">
   360→          <div className="text-[10px] text-muted-foreground uppercase">Available Cash</div>
   361→          <div className="text-lg font-bold font-mono text-emerald-400">₹{Math.round(wallet.available).toLocaleString()}</div>
   362→        </div>
   363→        <div className="rounded-lg bg-secondary/50 p-3">
   364→          <div className="text-[10px] text-muted-foreground uppercase">Deployed</div>
   365→          <div className="text-lg font-bold font-mono">₹{Math.round(wallet.deployed).toLocaleString()}</div>
   366→        </div>
   367→        <div className="rounded-lg bg-secondary/50 p-3">
   368→          <div className="text-[10px] text-muted-foreground uppercase">Realized P&L</div>
   369→          <div className={cn('text-lg font-bold font-mono', wallet.realizedPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   370→            {wallet.realizedPnl >= 0 ? '+' : ''}₹{Math.round(wallet.realizedPnl).toLocaleString()}
   371→          </div>
   372→        </div>
   373→      </div>
   374→    </div>
   375→  );
   376→}

   426→  const [form, setForm] = useState({ symbol: '', exDate: '', dividendPerShare: '', qty: '1', stockName: '' });
   427→
   428→  const fetchDividends = useCallback(async () => {
   429→    try {
   430→      const res = await fetch('/api/portfolio/dividends');
   431→      const d = await res.json();
   432→      if (d.success) setData(d);
   433→    } catch { /* ignore */ }
   434→  }, []);
   435→  useEffect(() => { fetchDividends(); }, [fetchDividends]);
   436→
   437→  const addDividend = async () => {
   438→    if (!form.symbol || !form.exDate || !form.dividendPerShare) return;
   439→    try {
   440→      const dps = parseFloat(form.dividendPerShare);
   441→      const qty = parseInt(form.qty) || 1;
   442→      const res = await fetch('/api/portfolio/dividends', {
   443→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   444→        body: JSON.stringify({ ...form, totalAmount: dps * qty }),
   445→      });
   446→      if ((await res.json()).success) {
   447→        toast.success(`Dividend recorded: ${form.symbol}`);
   448→        setForm({ symbol: '', exDate: '', dividendPerShare: '', qty: '1', stockName: '' });
   449→        setOpen(false);
   450→        fetchDividends();
   451→      }
   452→    } catch { toast.error('Failed to record dividend'); }
   453→  };
   454→
   455→  const deleteDiv = async (id: string) => {
   456→    await fetch(`/api/portfolio/dividends?id=${id}`, { method: 'DELETE' });
   457→    fetchDividends();
   458→  };
   459→
   460→  return (
   461→    <div>
   462→      <div className="flex items-center justify-between mb-3">
   463→        <h3 className="text-sm font-semibold flex items-center gap-2"><IndianRupee className="h-4 w-4" /> Dividend Tracker</h3>
   464→        <Dialog open={open} onOpenChange={setOpen}>
   465→          <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> Record Dividend</Button></DialogTrigger>
   466→          <DialogContent className="sm:max-w-sm">
   467→            <DialogHeader><DialogTitle>Record Dividend</DialogTitle></DialogHeader>
   468→            <div className="grid gap-3 py-2">
   469→              <div><Label className="text-xs">Symbol</Label><Input value={form.symbol} onChange={e => setForm({ ...form, symbol: e.target.value.toUpperCase() })} placeholder="ITC" className="h-8 text-xs" /></div>
   470→              <div className="grid grid-cols-2 gap-2">
   471→                <div><Label className="text-xs">Ex-Date</Label><Input type="date" value={form.exDate} onChange={e => setForm({ ...form, exDate: e.target.value })} className="h-8 text-xs" /></div>
   472→                <div><Label className="text-xs">DPS (₹)</Label><Input type="number" step="0.01" value={form.dividendPerShare} onChange={e => setForm({ ...form, dividendPerShare: e.target.value })} placeholder="6.75" className="h-8 text-xs" /></div>
   473→              </div>
   474→              <div><Label className="text-xs">Qty Held</Label><Input type="number" value={form.qty} onChange={e => setForm({ ...form, qty: e.target.value })} className="h-8 text-xs" /></div>
   475→            </div>
   476→            <DialogFooter>
   477→              <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
   478→              <Button size="sm" onClick={addDividend}>Record</Button>
   479→            </DialogFooter>
   480→          </DialogContent>
   481→        </Dialog>
   482→      </div>
   483→
   484→      {data?.summary && data.summary.totalDividends > 0 && (
   485→        <div className="grid grid-cols-2 gap-3 mb-3">
   486→          <div className="rounded-lg bg-emerald-500/10 p-3">
   487→            <div className="text-[10px] text-emerald-400 uppercase">Total Dividends</div>
   488→            <div className="text-lg font-bold font-mono text-emerald-400">₹{data.summary.totalDividends.toLocaleString()}</div>
   489→          </div>
   490→          <div className="rounded-lg bg-secondary/50 p-3">
   491→            <div className="text-[10px] text-muted-foreground uppercase">Stocks</div>
   492→            <div className="text-lg font-bold font-mono">{data.summary.uniqueStocks}</div>
   493→          </div>
   494→        </div>
   495→      )}
   496→
   497→      {(!data || data.dividends.length === 0) ? (
   498→        <p className="text-xs text-muted-foreground text-center py-4">No dividends recorded yet.</p>
   499→      ) : (
   500→        <ScrollArea className="max-h-[220px]">
   501→          <div className="space-y-1.5 pr-2">
   502→            {data.dividends.slice(0, 20).map((d: any) => (
   503→              <div key={d.id} className="flex items-center gap-2 text-xs py-1.5 border-b border-border/30">
   504→                <span className="font-semibold shrink-0">{d.symbol}</span>
   505→                <Badge variant="outline" className="text-[9px] h-4 text-emerald-400">₹{d.dividendPerShare}/share</Badge>
   506→                <span className="font-mono text-emerald-400">₹{Math.round(d.totalAmount).toLocaleString()}</span>
   507→                <span className="text-muted-foreground ml-auto">{new Date(d.exDate).toLocaleDateString()}</span>
   508→                <button onClick={() => deleteDiv(d.id)} className="text-muted-foreground hover:text-red-400"><Trash2 className="h-3 w-3" /></button>
   509→              </div>
   510→            ))}
   511→          </div>
   512→        </ScrollArea>
   513→      )}
   514→    </div>
   515→  );
   516→}
