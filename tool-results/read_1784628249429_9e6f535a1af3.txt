   200→   349→      low: data.map(c => c.low),
   201→   350→      close: data.map(c => c.close),
   202→   351→    });
   203→   352→    if (atrArr.length === 0) return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   204→   353→
   205→   354→    const atr = atrArr[atrArr.length - 1];
   206→   355→    // Trail level = highest close seen - ATR * multiplier (stored in notes, or use current price as proxy)
   207→   356→    // Since we don't track the high-water mark in DB, use: max(current, entry) - ATR * multiplier
   208→   357→    const highWaterMark = Math.max(currentPrice, entryPrice);
   209→   358→    const trailLevel = highWaterMark - (atr * rules.atrTrailMultiplier);
   210→   359→    return { useATR: true, trailLevel: Math.round(trailLevel * 100) / 100, atr: Math.round(atr * 100) / 100 };
   211→   360→  } catch {
   212→   361→    return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   213→   362→  }
   214→   363→}

   217→   366→async function canOpen(symbol: string, entryPrice: number, qty: number, rules: PositionRules, sector?: string): Promise<{ ok: boolean; reason: string }> {
   218→   367→  const w = await getWallet();
   219→   368→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   220→   369→  const now = new Date();
   221→   370→
   222→   371→  // === v2: Circuit Breaker Check ===
   223→   372→  const { drawdownPct } = await getPortfolioDrawdown();
   224→   373→  if (drawdownPct >= rules.maxDrawdownPct) {
   225→   374→    return { ok: false, reason: `CIRCUIT BREAKER: Drawdown ${drawdownPct.toFixed(1)}% >= ${rules.maxDrawdownPct}%` };
   226→   375→  }
   227→   376→
   228→   377→  // === v2: Daily Loss Limit ===
   229→   378→  await resetDailyCounters();
   230→   379→  const st = await getSchedulerState();
   231→   380→  if (st.todayPnl < 0 && Math.abs(st.todayPnl) >= rules.dailyLossLimit) {
   232→   381→    return { ok: false, reason: `DAILY LOSS LIMIT: ₹${Math.abs(st.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}` };
   233→   382→  }
   234→   383→
   235→   384→  // Max positions check
   236→   385→  if (open.length >= rules.maxTotalPositions) return { ok: false, reason: `Max ${rules.maxTotalPositions} positions` };
   237→   386→
   238→   387→  // Per-stock cap
   239→   388→  const stockOpen = open.filter(t => t.symbol === symbol);
   240→   389→  const stockDep = stockOpen.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   241→   390→  if (stockDep + entryPrice * qty > rules.maxPerStock) return { ok: false, reason: `Max ₹${rules.maxPerStock}/stock` };
   242→   391→
   243→   392→  // Duplicate position
   244→   393→  if (stockOpen.length > 0) return { ok: false, reason: 'Open position exists' };
   245→   394→
   246→   395→  // Monthly cap
   247→   396→  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   248→   397→  const monthTrades = await db.paperTrade.count({ where: { symbol, entryDate: { gte: monthStart } } });
   249→   398→  if (monthTrades >= rules.maxBuysPerMonth) return { ok: false, reason: `Max ${rules.maxBuysPerMonth} buys/month` };
   250→   399→
   251→   400→  // Capital check
   252→   401→  if (entryPrice * qty > w.available) return { ok: false, reason: 'Insufficient capital' };
   253→   402→
   254→   403→  // Cooldown check
   255→   404→  if (rules.cooldownDays > 0) {
   256→   405→    const lastExit = await db.paperTrade.findFirst({
   257→   406→      where: { symbol, status: 'CLOSED', exitDate: { not: null } },
   258→   407→      orderBy: { exitDate: 'desc' },
   259→   408→    });
   260→   409→    if (lastExit?.exitDate) {
   261→   410→      const daysSinceExit = (now.getTime() - new Date(lastExit.exitDate).getTime()) / 86400000;
   262→   411→      if (daysSinceExit < rules.cooldownDays) {
   263→   412→        return { ok: false, reason: `Cooldown: ${Math.ceil(rules.cooldownDays - daysSinceExit)}d remaining` };
   264→   413→      }
   265→   414→    }
   266→   415→  }
   267→   416→
   268→   417→  // Sector concentration check
   269→   418→  if (sector && rules.maxSectorPct > 0) {
   270→   419→    const sectorAlloc = await getSectorAllocation();
   271→   420→    const currentPct = sectorAlloc[sector] || 0;
   272→   421→    const newPct = currentPct + ((entryPrice * qty) / w.totalCapital) * 100;
   273→   422→    if (newPct > rules.maxSectorPct) {
   274→   423→      return { ok: false, reason: `Sector cap: ${sector} at ${currentPct.toFixed(1)}% (max ${rules.maxSectorPct}%)` };
   275→   424→    }
   276→   425→  }
   277→   426→
   278→   427→  return { ok: true };
   279→   428→}

   412→   561→// ── Core Engine: Check Exits (v2 with health scoring) ─
   413→   562→async function autoCheckExits() {
   414→   563→  const rules = await getRules();
   415→   564→  const openTrades = await db.paperTrade.findMany({ where: { status: 'OPEN' }, orderBy: { entryDate: 'asc' } });
   416→   565→  const exits: any[] = [];
   417→   566→  const holding: any[] = [];
   418→   567→  const partialBooks: any[] = [];
   419→   568→  const warnings: any[] = [];
   420→   569→
   421→   570→  const minsToClose = timeToClose();
   422→   571→  const shouldTimeExit = minsToClose <= rules.timeExitMins && minsToClose > 0;
   423→   572→
   424→   573→  for (const trade of openTrades) {
   425→   574→    try {
   426→   575→      const { price: cp } = await getCurrentPrice(trade.symbol);
   427→   576→      if (cp <= 0) { holding.push({ symbol: trade.symbol, error: 'Price unavailable' }); continue; }
   428→   577→
   429→   578→      const days = Math.floor((Date.now() - new Date(trade.entryDate).getTime()) / 86400000);
   430→   579→      const risk = trade.entryPrice - trade.stopLoss;
   431→   580→      const rMultiple = risk > 0 ? (cp - trade.entryPrice) / risk : 0;
   432→   581→      let shouldExit = false;
   433→   582→      let reason = '';
   434→   583→      let exitPrice = cp;
   435→   584→      let exitQty = trade.qty;
   436→   585→
   437→   586→      // 1. Hard SL hit
   438→   587→      if (cp <= trade.stopLoss) {
   439→   588→        shouldExit = true; reason = 'SL_HIT'; exitPrice = trade.stopLoss;
   440→   589→      }
   441→   590→      // 2. Hard TP hit
   442→   591→      else if (cp >= trade.targetPrice) {
   443→   592→        shouldExit = true; reason = 'TP_HIT'; exitPrice = trade.targetPrice;
   444→   593→      }
   445→   594→      // 3. Trailing stop (v2: ATR-based if configured)
   446→   595→      else if (risk > 0 && rMultiple >= rules.trailingStopR) {
   447→   596→        const trail = await getATRTrailLevel(trade.symbol, trade.entryPrice, cp, risk, rules);
   448→   597→        if (trail.useATR) {
   449→   598→          // ATR trailing: only exit if price drops below ATR-based level
   450→   599→          if (cp <= trail.trailLevel) {
   451→   600→            shouldExit = true; reason = 'ATR_TRAIL_STOP'; exitPrice = trail.trailLevel;
   452→   601→          }
   453→   602→        } else {
   454→   603→          // Fixed R trailing (v1 behavior)
   455→   604→          const trailLevel = trade.entryPrice + (risk * rules.trailToR);
   456→   605→          if (cp <= trailLevel) {
   457→   606→            shouldExit = true; reason = 'TRAIL_STOP'; exitPrice = trailLevel;
   458→   607→          }
   459→   608→        }
   460→   609→      }
   461→   610→      // 4. Max holding days
   462→   611→      else if (days >= rules.maxHoldingDays) {
   463→   612→        shouldExit = true; reason = 'MAX_HOLDING_DAYS';
   464→   613→      }
   465→   614→      // 5. v2: Stale loser acceleration — exit aging losing positions faster
   466→   615→      else if (days >= rules.maxHoldingDays * 0.7 && rMultiple < -0.3) {
   467→   616→        shouldExit = true; reason = 'STALE_LOSER';
   468→   617→      }
   469→   618→      // 6. Time-based exit
   470→   619→      else if (shouldTimeExit) {
   471→   620→        if (rMultiple < 0.5) {
   472→   621→          shouldExit = true; reason = 'TIME_EXIT'; exitPrice = cp;
   473→   622→        }
   474→   623→      }
   475→   624→
   476→   625→      // 7. Partial booking (doesn't trigger full exit)
   477→   626→      if (!shouldExit && risk > 0 && rMultiple >= rules.partialBookR && trade.qty > 1) {
   478→   627→        const alreadyBooked = trade.tags?.includes('partial-booked');
   479→   628→        if (!alreadyBooked) {
   480→   629→          const bookQty = Math.max(1, Math.floor(trade.qty * (rules.partialBookPct / 100)));
   481→   630→          const remainingQty = trade.qty - bookQty;
   482→   631→          if (remainingQty >= 1) {
   483→   632→            const bookPnl = (cp - trade.entryPrice) * bookQty;
   484→   633→            await db.paperTrade.update({
   485→   634→              where: { id: trade.id },
   486→   635→              data: { qty: remainingQty, tags: (trade.tags ? trade.tags + ',' : '') + 'partial-booked' },
   487→   636→            });
   488→   637→            const w = await getWallet();
   489→   638→            await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + bookPnl } });
   490→   639→            await db.autoTradeLog.create({
   491→   640→              action: 'PARTIAL_BOOK', symbol: trade.symbol, tradeId: trade.id,
   492→   641→              executed: true, reason: `Booked ${bookQty}/${trade.qty + bookQty} at ${rMultiple.toFixed(1)}R | P&L: ₹${bookPnl.toLocaleString()}`,
   493→   642→            });
   494→   643→            partialBooks.push({ symbol: trade.symbol, bookedQty: bookQty, remainingQty, rMultiple, pnl: bookPnl, exitPrice: cp });
   495→   644→          }
   496→   645→        }
   497→   646→      }
   498→   647→
   499→   648→      // Execute full exit
   500→   649→      if (shouldExit) {
   501→   650→        const pnl = (exitPrice - trade.entryPrice) * exitQty;
   502→   651→        const pnlPct = ((exitPrice - trade.entryPrice) / trade.entryPrice) * 100;
   503→   652→        await db.paperTrade.update({
   504→   653→          where: { id: trade.id },
   505→   654→          data: { status: 'CLOSED', exitDate: new Date(), exitPrice, pnl, pnlPercent: pnlPct, exitReason: reason },
   506→   655→        });
   507→   656→        const w = await getWallet();
   508→   657→        await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + pnl } });
   509→   658→        await db.autoTradeLog.create({
   510→   659→          action: `AUTO_EXIT_${reason}`, symbol: trade.symbol, tradeId: trade.id,
   511→   660→          executed: true, reason: `${reason} | ₹${trade.entryPrice}→₹${exitPrice} | P&L: ₹${pnl.toLocaleString()} | ${rMultiple.toFixed(1)}R`,
   512→   661→        });
   513→   662→
   514→   663→        // v2: Track consecutive losses for adaptive sizing
   515→   664→        if (pnl < 0) {
   516→   665→          const newStreak = (await getSchedulerState()).consecutiveLosses + 1;
   517→   666→          await setSchedulerKV('sched_consecutiveLosses', String(newStreak));
   518→   667→        } else {
   519→   668→          await setSchedulerKV('sched_consecutiveLosses', '0');
   520→   669→          // Reset adaptive factor on win
   521→   670→          await setSchedulerKV('sched_lastAdaptiveFactor', '1');
   522→   671→        }
   523→   672→
   524→   673→        await resetDailyCounters();
   525→   674→        const st2 = await getSchedulerState();
   526→   675→        await setSchedulerKV('sched_todayExits', String(st2.todayExits + 1));
   527→   676→        await setSchedulerKV('sched_todayPnl', String(st2.todayPnl + pnl));
   528→   677→
   529→   678→        // v2: Check if we need to activate daily loss circuit breaker
   530→   679→        const updatedSt = await getSchedulerState();
   531→   680→        if (updatedSt.todayPnl < 0 && Math.abs(updatedSt.todayPnl) >= rules.dailyLossLimit) {
   532→   681→          await setSchedulerKV('sched_circuitBreaker', 'true');
   533→   682→          await setSchedulerKV('sched_circuitBreakerReason', `Daily loss ₹${Math.abs(updatedSt.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}`);
   534→   683→        }
   535→   684→
   536→   685→        exits.push({
   537→   686→          symbol: trade.symbol, exitReason: reason, entryPrice: trade.entryPrice, exitPrice,
   538→   687→          pnl, pnlPercent: pnlPct, holdingDays: days, rMultiple: rMultiple.toFixed(2),
   539→   688→        });
   540→   689→      } else {
   541→   690→        // v2: Position Health Score
   542→   691→        const health = scorePositionHealth(cp, trade.entryPrice, trade.stopLoss, trade.targetPrice, days, rules.maxHoldingDays, rMultiple);
   543→   692→        const healthData: any = {
   544→   693→          symbol: trade.symbol, entryPrice: trade.entryPrice, currentPrice: cp,
   545→   694→          pnl: (cp - trade.entryPrice) * trade.qty,
   546→   695→          pnlPercent: ((cp - trade.entryPrice) / trade.entryPrice) * 100,
   547→   696→          holdingDays: days, maxHoldingDays: rules.maxHoldingDays,
   548→   697→          sl: trade.stopLoss, tp: trade.targetPrice, rMultiple: rMultiple.toFixed(2),
   549→   698→          healthScore: health.score, healthUrgency: health.urgency, healthFactors: health.factors,
   550→   699→        };
   551→   700→
   552→   701→        if (health.urgency === 'CRITICAL') {
   553→   702→          healthData.slWarning = true;
   554→   703→          warnings.push({ ...healthData, type: health.score < 20 ? 'SL_PROXIMITY' : 'AGING', message: `${trade.symbol} health ${health.score}/100 (${health.urgency})` });
   555→   704→        } else if (health.urgency === 'WARNING') {
   556→   705→          warnings.push({ ...healthData, type: 'AGING', message: `${trade.symbol} health ${health.score}/100` });
   557→   706→        }
   558→   707→
   559→   708→        holding.push(healthData);
   560→   709→      }
   561→   710→    } catch { holding.push({ symbol: trade.symbol, error: 'Price fetch failed' }); }
   562→   711→  }
   563→   712→
   564→   713→  if (exits.length > 0 || partialBooks.length > 0) await recalcWallet();
   565→   714→  return { exits, holding, partialBooks, warnings, timeToClose: minsToClose, marketHours: isMarketHours() };
   566→   715→}
   567→   716→
   568→   717→// ── Scheduler Control ──────────────────────────────────
   569→   718→async function runSchedulerTick() {
   570→   719→  if (!isMarketHours()) return { marketHours: false, message: 'Market closed' };
   571→   720→  const state = await getSchedulerState();
   572→   721→  if (!state.enabled) return { marketHours: true, enabled: false };
   573→   722→
   574→   723→  await resetDailyCounters();
   575→   724→  const now = new Date();
   576→   725→  const results: any = { scanResult: null, exitResult: null };
   577→   726→
   578→   727→  if (state.nextScanAt) {
   579→   728→    const nextScan = new Date(state.nextScanAt);
   580→   729→    if (now >= nextScan) {
   581→   730→      const config: ScreeningConfig = DEFAULT_CONFIG;
   582→   731→      results.scanResult = await autoScanAndTrade(config);
   583→   732→      await setSchedulerKV('sched_lastScanAt', now.toISOString());
   584→   733→      const next = new Date(now.getTime() + state.scanIntervalMin * 60000);
   585→   734→      await setSchedulerKV('sched_nextScanAt', next.toISOString());
   586→   735→    }
   587→   736→  }
   588→   737→
   589→   738→  if (state.nextExitAt) {
   590→   739→    const nextExit = new Date(state.nextExitAt);
   591→   740→    if (now >= nextExit) {
   592→   741→      results.exitResult = await autoCheckExits();
   593→   742→      await setSchedulerKV('sched_lastExitAt', now.toISOString());
   594→   743→      const next = new Date(now.getTime() + state.exitIntervalMin * 60000);
   595→   744→      await setSchedulerKV('sched_nextExitAt', next.toISOString());
   596→   745→    }
   597→   746→  }
   598→   747→
   599→   748→  return { ...results, marketHours: true, enabled: true, schedulerState: await getSchedulerState() };
   600→   749→}

   603→   752→export async function GET() {

   607→   756→    const recentLogs = await db.autoTradeLog.findMany({ orderBy: { createdAt: 'desc' }, take: 50 });

   625→   774→export async function POST(request: NextRequest) {
   626→   775→  try {
   627→   776→    const body = await request.json();
   628→   777→    const action: string = body.action || 'get_status';
   629→   778→    const config: ScreeningConfig = { ...DEFAULT_CONFIG, ...body.config };
   630→   779→
   631→   780→    switch (action) {
   632→   781→      case 'scan_and_trade': {
   633→   782→        const result = await autoScanAndTrade(config);
   634→   783→        return NextResponse.json({ success: true, action, ...result });
   635→   784→      }
   636→   785→      case 'check_exits': {
   637→   786→        const result = await autoCheckExits();
   638→   787→        return NextResponse.json({ success: true, action, ...result });
   639→   788→      }
   640→   789→      case 'scheduler_tick': {
   641→   790→        const result = await runSchedulerTick();
   642→   791→        return NextResponse.json({ success: true, action, ...result });
   643→   792→      }
   644→   793→      case 'scheduler_toggle': {
   645→   794→        const enabled = body.enabled;
   646→   795→        if (enabled) {
   647→   796→          const now = new Date();
   648→   797→          const state = await getSchedulerState();
   649→   798→          const scanMin = body.scanIntervalMin || state.scanIntervalMin;
   650→   799→          const exitMin = body.exitIntervalMin || state.exitIntervalMin;
   651→   800→          await setSchedulerKV('sched_enabled', 'true');
   652→   801→          await setSchedulerKV('sched_scanIntervalMin', String(scanMin));
   653→   802→          await setSchedulerKV('sched_exitIntervalMin', String(exitMin));
   654→   803→          await setSchedulerKV('sched_nextScanAt', now.toISOString());
   655→   804→          await setSchedulerKV('sched_nextExitAt', now.toISOString());
   656→   805→          await resetDailyCounters();
   657→   806→        } else {
   658→   807→          await setSchedulerKV('sched_enabled', 'false');
   659→   808→          await setSchedulerKV('sched_nextScanAt', '');
   660→   809→          await setSchedulerKV('sched_nextExitAt', '');
   661→   810→        }
   662→   811→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   663→   812→      }
   664→   813→      case 'update_wallet': {
   665→   814→        const { totalCapital } = body;
   666→   815→        if (totalCapital) {
   667→   816→          const w = await getWallet();
   668→   817→          const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   669→   818→          const deployed = open.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   670→   819→          await db.capitalWallet.update({ where: { id: w.id }, data: { totalCapital, available: totalCapital - deployed } });
   671→   820→          const updated = await recalcWallet();
   672→   821→          return NextResponse.json({ success: true, wallet: updated });
   673→   822→        }
   674→   823→        break;
   675→   824→      }
   676→   825→      case 'update_rules': {
   677→   826→        const rules = body.rules;
   678→   827→        if (rules) {
   679→   828→          for (const [key, value] of Object.entries(rules)) {
   680→   829→            await db.appSettings.upsert({
   681→   830→              where: { key: `rules_${key}` },
   682→   831→              create: { key: `rules_${key}`, value: String(value) },
   683→   832→              update: { value: String(value) },
   684→   833→            });
   685→   834→          }
   686→   835→          return NextResponse.json({ success: true, rules: await getRules() });
   687→   836→        }
   688→   837→        break;
   689→   838→      }
   690→   839→      case 'reset_circuit_breaker': {
   691→   840→        // v2: Manual circuit breaker reset
   692→   841→        await setSchedulerKV('sched_circuitBreaker', 'false');
   693→   842→        await setSchedulerKV('sched_circuitBreakerReason', '');
   694→   843→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   695→   844→      }
   696→   845→    }
   697→   846→    const wallet = await recalcWallet();
   698→   847→    return NextResponse.json({ success: true, wallet });
   699→   848→  } catch (error) {
   700→   849→    console.error('Auto-trade error:', error);
   701→   850→    return NextResponse.json({ success: false, error: String(error) }, { status: 500 });
   702→   851→  }
   703→   852→}
