    94→function scorePositionHealth(
    95→  cp: number, entryPrice: number, sl: number, tp: number,
    96→  holdingDays: number, maxHoldingDays: number, rMultiple: number
    97→): HealthScore {
    98→  const factors: { name: string; impact: number; detail: string }[] = [];
    99→  let score = 50; // Base score
   100→
   101→  // Factor 1: R-multiple profit (±25 points)
   102→  if (rMultiple >= 2) { score += 25; factors.push({ name: 'Profit', impact: 25, detail: `Strong ${rMultiple.toFixed(1)}R` }); }
   103→  else if (rMultiple >= 1) { score += 15; factors.push({ name: 'Profit', impact: 15, detail: `Moderate ${rMultiple.toFixed(1)}R` }); }
   104→  else if (rMultiple >= 0) { score += 5; factors.push({ name: 'Profit', impact: 5, detail: `Marginal ${rMultiple.toFixed(1)}R` }); }
   105→  else if (rMultiple >= -0.5) { score -= 10; factors.push({ name: 'Loss', impact: -10, detail: `Small loss ${rMultiple.toFixed(1)}R` }); }
   106→  else { score -= 25; factors.push({ name: 'Loss', impact: -25, detail: `Large loss ${rMultiple.toFixed(1)}R` }); }
   107→
   108→  // Factor 2: SL proximity (±20 points)
   109→  const risk = entryPrice - sl;
   110→  const slDistPct = risk > 0 ? ((cp - sl) / risk) * 100 : 100;
   111→  if (slDistPct < 20) { score -= 20; factors.push({ name: 'SL Proximity', impact: -20, detail: `${slDistPct.toFixed(0)}% to SL` }); }
   112→  else if (slDistPct < 40) { score -= 10; factors.push({ name: 'SL Proximity', impact: -10, detail: `${slDistPct.toFixed(0)}% to SL` }); }
   113→  else { score += 5; factors.push({ name: 'SL Distance', impact: 5, detail: `Safe at ${slDistPct.toFixed(0)}%` }); }
   114→
   115→  // Factor 3: Age penalty (±15 points)
   116→  const agePct = maxHoldingDays > 0 ? holdingDays / maxHoldingDays : 0;
   117→  if (agePct >= 0.9) { score -= 15; factors.push({ name: 'Age', impact: -15, detail: `${holdingDays}/${maxHoldingDays}d (expired)` }); }
   118→  else if (agePct >= 0.7) { score -= 8; factors.push({ name: 'Age', impact: -8, detail: `${holdingDays}/${maxHoldingDays}d (aging)` }); }
   119→  else if (agePct < 0.3) { score += 5; factors.push({ name: 'Age', impact: 5, detail: `Fresh ${holdingDays}d` }); }
   120→
   121→  // Factor 4: Trend to TP (±10 points)
   122→  if (tp > entryPrice && cp > entryPrice) {
   123→    const tpProgress = ((cp - entryPrice) / (tp - entryPrice)) * 100;
   124→    if (tpProgress > 75) { score += 10; factors.push({ name: 'TP Progress', impact: 10, detail: `${tpProgress.toFixed(0)}% to TP` }); }
   125→    else if (tpProgress > 40) { score += 5; factors.push({ name: 'TP Progress', impact: 5, detail: `${tpProgress.toFixed(0)}% to TP` }); }
   126→  }
   127→
   128→  // Clamp score
   129→  score = Math.max(0, Math.min(100, score));
   130→
   131→  let urgency: HealthScore['urgency'] = 'OK';
   132→  if (score < 30) urgency = 'CRITICAL';
   133→  else if (score < 55) urgency = 'WARNING';
   134→
   135→  return { score, factors, urgency };
   136→}

   139→function isMarketHours(): boolean {
   140→  const now = new Date();
   141→  const ist = new Date(now.getTime() + (5.5 * 60 * 60 * 1000) + (now.getTimezoneOffset() * 60 * 1000));
   142→  const day = ist.getDay();
   143→  if (day === 0 || day === 6) return false;
   144→  const h = ist.getHours();
   145→  const m = ist.getMinutes();
   146→  const mins = h * 60 + m;
   147→  return mins >= 555 && mins <= 930;
   148→}

   239→async function recalcWallet() {
   240→  const w = await getWallet();
   241→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   242→  const deployed = open.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   243→  const total = w.totalCapital + w.realizedPnl;
   244→  let unrealizedPnl = 0;
   245→  const syms = [...new Set(open.map(t => t.symbol))];
   246→  for (const sym of syms) {
   247→    try {
   248→      const { price } = await getCurrentPrice(sym);
   249→      for (const t of open.filter(t => t.symbol === sym)) {
   250→        unrealizedPnl += (price - t.entryPrice) * t.qty;
   251→      }
   252→    } catch { /* skip */ }
   253→  }
   254→  await db.capitalWallet.update({ where: { id: w.id }, data: { deployed, available: total - deployed, unrealizedPnl, totalCapital: total } });
   255→  return { ...w, deployed, available: total - deployed, unrealizedPnl, totalCapital: total, initialCapital: w.initialCapital };
   256→}

   258→async function getSectorAllocation(): Promise<Record<string, number>> {
   259→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   260→  const w = await getWallet();
   261→  const total = w.totalCapital;
   262→  if (total <= 0) return {};
   263→  const sectors: Record<string, number> = {};
   264→  for (const t of open) {
   265→    const ws = await db.watchlistStock.findUnique({ where: { symbol: t.symbol } });
   266→    const sector = ws?.sector || t.tags?.split(',').find((tag: string) => !['auto','aplus','b','partial-booked'].includes(tag)) || 'Unknown';
   267→    sectors[sector] = (sectors[sector] || 0) + (t.entryPrice * t.qty);
   268→  }
   269→  const pct: Record<string, number> = {};
   270→  for (const [k, v] of Object.entries(sectors)) pct[k] = (v / total) * 100;
   271→  return pct;
   272→}

   274→// ── v2: Drawdown & Circuit Breaker ─────────────────────
   275→async function getPortfolioDrawdown(): Promise<{ drawdownPct: number; peakCapital: number; currentCapital: number }> {
   276→  const w = await getWallet();
   277→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   278→  let unrealizedPnl = 0;
   279→  const syms = [...new Set(open.map(t => t.symbol))];
   280→  for (const sym of syms) {
   281→    try {
   282→      const { price } = await getCurrentPrice(sym);
   283→      for (const t of open.filter(t => t.symbol === sym)) {
   284→        unrealizedPnl += (price - t.entryPrice) * t.qty;
   285→      }
   286→    } catch { /* skip */ }
   287→  }
   288→
   289→  const currentCapital = w.totalCapital + unrealizedPnl;
   290→  // Peak = highest totalCapital value we've seen (initial + best realized)
   291→  const peakCapital = Math.max(w.initialCapital, w.totalCapital + w.realizedPnl, currentCapital);
   292→  const drawdownPct = peakCapital > 0 ? ((peakCapital - currentCapital) / peakCapital) * 100 : 0;
   293→  return { drawdownPct, peakCapital, currentCapital };
   294→}

   296→async function getConsecutiveLosses(): Promise<number> {
   297→  const recent = await db.paperTrade.findMany({
   298→    where: { status: 'CLOSED', autoTraded: true, pnl: { not: null } },
   299→    orderBy: { exitDate: 'desc' },
   300→    take: 10,
   301→  });
   302→  let streak = 0;
   303→  for (const t of recent) {
   304→    if ((t.pnl || 0) < 0) streak++;
   305→    else break;
   306→  }
   307→  return streak;
   308→}

   310→function getAdaptiveFactor(consecutiveLosses: number, rules: PositionRules): number {
   311→  if (!rules.adaptiveSizing) return 1.0;
   312→  if (consecutiveLosses < 3) return 1.0;
   313→  // Reduce by streakPenaltyPct for each loss beyond 2, min 20% of original
   314→  const reductions = consecutiveLosses - 2;
   315→  const factor = 1.0 - (reductions * (rules.streakPenaltyPct / 100));
   316→  return Math.max(0.2, Math.min(1.0, factor));
   317→}

   319→// ── v2: Nifty Regime Detection ──────────────────────────
   320→async function getNiftyRegime(niftyData: any[]): Promise<{ regime: 'BULLISH' | 'BEARISH' | 'UNKNOWN'; ema200: number; currentClose: number }> {
   321→  if (niftyData.length < 200) return { regime: 'UNKNOWN', ema200: 0, currentClose: 0 };
   322→  const closes = niftyData.map(c => c.close);
   323→  const ema200Arr = EMA.calculate({ period: 200, values: closes });
   324→  if (ema200Arr.length === 0) return { regime: 'UNKNOWN', ema200: 0, currentClose: 0 };
   325→  const ema200 = ema200Arr[ema200Arr.length - 1];
   326→  const currentClose = closes[closes.length - 1];
   327→  return {
   328→    regime: currentClose > ema200 ? 'BULLISH' : 'BEARISH',
   329→    ema200: Math.round(ema200 * 100) / 100,
   330→    currentClose: Math.round(currentClose * 100) / 100,
   331→  };
   332→}

   334→// ── v2: ATR-based Trailing Stop ─────────────────────────
   335→async function getATRTrailLevel(symbol: string, entryPrice: number, currentPrice: number, risk: number, rules: PositionRules): Promise<{ useATR: boolean; trailLevel: number; atr: number }> {
   336→  if (rules.atrTrailMultiplier <= 0) {
   337→    // Use fixed R-based trail (v1 behavior)
   338→    const trailLevel = entryPrice + (risk * rules.trailToR);
   339→    return { useATR: false, trailLevel, atr: 0 };
   340→  }
   341→
   342→  try {
   343→    const { data } = await getHistoricalData(symbol, 30);
   344→    if (data.length < 14) return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   345→
   346→    const atrArr = ATR.calculate({
   347→      period: 14,
   348→      high: data.map(c => c.high),
   349→      low: data.map(c => c.low),
   350→      close: data.map(c => c.close),
   351→    });
   352→    if (atrArr.length === 0) return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   353→
   354→    const atr = atrArr[atrArr.length - 1];
   355→    // Trail level = highest close seen - ATR * multiplier (stored in notes, or use current price as proxy)
   356→    // Since we don't track the high-water mark in DB, use: max(current, entry) - ATR * multiplier
   357→    const highWaterMark = Math.max(currentPrice, entryPrice);
   358→    const trailLevel = highWaterMark - (atr * rules.atrTrailMultiplier);
   359→    return { useATR: true, trailLevel: Math.round(trailLevel * 100) / 100, atr: Math.round(atr * 100) / 100 };
   360→  } catch {
   361→    return { useATR: false, trailLevel: entryPrice + (risk * rules.trailToR), atr: 0 };
   362→  }
   363→}

   365→// ── Gate Checks (v2 enhanced canOpen) ───────────────────
   366→async function canOpen(symbol: string, entryPrice: number, qty: number, rules: PositionRules, sector?: string): Promise<{ ok: boolean; reason: string }> {
   367→  const w = await getWallet();
   368→  const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   369→  const now = new Date();
   370→
   371→  // === v2: Circuit Breaker Check ===
   372→  const { drawdownPct } = await getPortfolioDrawdown();
   373→  if (drawdownPct >= rules.maxDrawdownPct) {
   374→    return { ok: false, reason: `CIRCUIT BREAKER: Drawdown ${drawdownPct.toFixed(1)}% >= ${rules.maxDrawdownPct}%` };
   375→  }
   376→
   377→  // === v2: Daily Loss Limit ===
   378→  await resetDailyCounters();
   379→  const st = await getSchedulerState();
   380→  if (st.todayPnl < 0 && Math.abs(st.todayPnl) >= rules.dailyLossLimit) {
   381→    return { ok: false, reason: `DAILY LOSS LIMIT: ₹${Math.abs(st.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}` };
   382→  }
   383→
   384→  // Max positions check
   385→  if (open.length >= rules.maxTotalPositions) return { ok: false, reason: `Max ${rules.maxTotalPositions} positions` };
   386→
   387→  // Per-stock cap
   388→  const stockOpen = open.filter(t => t.symbol === symbol);
   389→  const stockDep = stockOpen.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   390→  if (stockDep + entryPrice * qty > rules.maxPerStock) return { ok: false, reason: `Max ₹${rules.maxPerStock}/stock` };
   391→
   392→  // Duplicate position
   393→  if (stockOpen.length > 0) return { ok: false, reason: 'Open position exists' };
   394→
   395→  // Monthly cap
   396→  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
   397→  const monthTrades = await db.paperTrade.count({ where: { symbol, entryDate: { gte: monthStart } } });
   398→  if (monthTrades >= rules.maxBuysPerMonth) return { ok: false, reason: `Max ${rules.maxBuysPerMonth} buys/month` };
   399→
   400→  // Capital check
   401→  if (entryPrice * qty > w.available) return { ok: false, reason: 'Insufficient capital' };
   402→
   403→  // Cooldown check
   404→  if (rules.cooldownDays > 0) {
   405→    const lastExit = await db.paperTrade.findFirst({
   406→      where: { symbol, status: 'CLOSED', exitDate: { not: null } },
   407→      orderBy: { exitDate: 'desc' },
   408→    });
   409→    if (lastExit?.exitDate) {
   410→      const daysSinceExit = (now.getTime() - new Date(lastExit.exitDate).getTime()) / 86400000;
   411→      if (daysSinceExit < rules.cooldownDays) {
   412→        return { ok: false, reason: `Cooldown: ${Math.ceil(rules.cooldownDays - daysSinceExit)}d remaining` };
   413→      }
   414→    }
   415→  }
   416→
   417→  // Sector concentration check
   418→  if (sector && rules.maxSectorPct > 0) {
   419→    const sectorAlloc = await getSectorAllocation();
   420→    const currentPct = sectorAlloc[sector] || 0;
   421→    const newPct = currentPct + ((entryPrice * qty) / w.totalCapital) * 100;
   422→    if (newPct > rules.maxSectorPct) {
   423→      return { ok: false, reason: `Sector cap: ${sector} at ${currentPct.toFixed(1)}% (max ${rules.maxSectorPct}%)` };
   424→    }
   425→  }
   426→
   427→  return { ok: true };
   428→}

   430→// ── Core Engine: Scan & Auto-Enter (v2) ────────────────
   431→async function autoScanAndTrade(config: ScreeningConfig) {
   432→  const rules = await getRules();
   433→  const stocks = getFullUniverse();
   434→  const entries: any[] = [];
   435→  const skipped: any[] = [];
   436→  let niftyData: any[] = [];
   437→  let niftyRegime: 'BULLISH' | 'BEARISH' | 'UNKNOWN' = 'UNKNOWN';
   438→
   439→  try {
   440→    const result = await getHistoricalData('NIFTY50', 350);
   441→    niftyData = result.data;
   442→  } catch {
   443→    skipped.push({ symbol: 'NIFTY50', reason: 'Failed to fetch Nifty data for macro filter' });
   444→    return { entries, skipped, l1Passed: 0, l2Signals: 0, totalScanned: stocks.length, circuitBreaker: false, niftyRegime: 'UNKNOWN' };
   445→  }
   446→
   447→  // v2: Nifty Regime Filter
   448→  const regime = await getNiftyRegime(niftyData);
   449→  niftyRegime = regime.regime;
   450→  await setSchedulerKV('sched_niftyRegime', regime.regime);
   451→
   452→  if (rules.niftyRegimeFilter && regime.regime === 'BEARISH') {
   453→    return {
   454→      entries, skipped: [{ symbol: 'FILTER', reason: `Nifty regime BEARISH (Close ${regime.currentClose} < EMA200 ${regime.ema200})` }],
   455→      l1Passed: 0, l2Signals: 0, totalScanned: stocks.length, circuitBreaker: false, niftyRegime,
   456→    };
   457→  }
   458→
   459→  // v2: Circuit breaker & daily loss check
   460→  await resetDailyCounters();
   461→  const { drawdownPct } = await getPortfolioDrawdown();
   462→  const st = await getSchedulerState();
   463→  if (drawdownPct >= rules.maxDrawdownPct) {
   464→    await setSchedulerKV('sched_circuitBreaker', 'true');
   465→    await setSchedulerKV('sched_circuitBreakerReason', `Drawdown ${drawdownPct.toFixed(1)}% >= ${rules.maxDrawdownPct}%`);
   466→    return {
   467→      entries, skipped: [{ symbol: 'CIRCUIT_BREAKER', reason: `Portfolio drawdown ${drawdownPct.toFixed(1)}% exceeds ${rules.maxDrawdownPct}% limit` }],
   468→      l1Passed: 0, l2Signals: 0, totalScanned: stocks.length, circuitBreaker: true, niftyRegime,
   469→    };
   470→  }
   471→  if (st.todayPnl < 0 && Math.abs(st.todayPnl) >= rules.dailyLossLimit) {
   472→    await setSchedulerKV('sched_circuitBreaker', 'true');
   473→    await setSchedulerKV('sched_circuitBreakerReason', `Daily loss ₹${Math.abs(st.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}`);
   474→    return {
   475→      entries, skipped: [{ symbol: 'DAILY_LIMIT', reason: `Daily loss ₹${Math.abs(st.todayPnl).toFixed(0)} exceeds ₹${rules.dailyLossLimit} limit` }],
   476→      l1Passed: 0, l2Signals: 0, totalScanned: stocks.length, circuitBreaker: true, niftyRegime,
   477→    };
   478→  }
   479→
   480→  // v2: Adaptive sizing factor
   481→  const consLosses = await getConsecutiveLosses();
   482→  await setSchedulerKV('sched_consecutiveLosses', String(consLosses));
   483→  const adaptiveFactor = getAdaptiveFactor(consLosses, rules);
   484→  await setSchedulerKV('sched_lastAdaptiveFactor', String(adaptiveFactor));
   485→
   486→  // L1 filter in batches
   487→  const BATCH = 15;
   488→  const l1Pass: { stock: NSEStock; data: any[] }[] = [];
   489→  for (let i = 0; i < stocks.length; i += BATCH) {
   490→    const batch = stocks.slice(i, i + BATCH);
   491→    const results = await Promise.allSettled(batch.map(async (stock) => {
   492→      const { data } = await getHistoricalData(stock.symbol, 300);
   493→      return { stock, data };
   494→    }));
   495→    for (const r of results) {
   496→      if (r.status !== 'fulfilled') continue;
   497→      const l1 = runL1Filter(r.value.stock.symbol, r.value.stock.name, r.value.stock.sector, r.value.stock.category, r.value.data);
   498→      if (l1 && l1.passed) l1Pass.push(r.value);
   499→    }
   500→  }
   501→
   502→  // L2 V-Swing + auto-enter with v2 enhancements
   503→  for (const { stock, data } of l1Pass) {
   504→    try {
   505→      const signal = runScreening(stock.symbol, data, niftyData, config);
   506→      if (!signal) { skipped.push({ symbol: stock.symbol, reason: 'No V-Swing signal' }); continue; }
   507→
   508→      // v2: Apply adaptive sizing
   509→      let qty = signal.setupType === 'A+' ? signal.sizing.qtyA : signal.sizing.qtyB;
   510→      if (adaptiveFactor < 1.0) {
   511→        const adjustedQty = Math.max(1, Math.floor(qty * adaptiveFactor));
   512→        skipped.push({
   513→          symbol: stock.symbol, reason: `Adaptive sizing: ${qty} → ${adjustedQty} (${(adaptiveFactor * 100).toFixed(0)}% factor, ${consLosses} consecutive losses)`,
   514→        });
   515→        // Only show as info, still proceed with adjusted qty
   516→        qty = adjustedQty;
   517→      }
   518→      if (qty <= 0) { skipped.push({ symbol: stock.symbol, reason: 'Qty=0' }); continue; }
   519→
   520→      const check = await canOpen(stock.symbol, signal.entryPrice, qty, rules, stock.sector);
   521→      if (!check.ok) { skipped.push({ symbol: stock.symbol, reason: check.reason }); continue; }
   522→
   523→      const trade = await db.paperTrade.create({
   524→        data: {
   525→          symbol: stock.symbol, stockName: stock.name, direction: 'LONG',
   526→          entryDate: new Date(), entryPrice: signal.entryPrice, qty,
   527→          stopLoss: signal.stopLoss, targetPrice: signal.targetPrice,
   528→          autoTraded: true,
   529→          notes: `AUTO | ${signal.setupType} | Score:${signal.score}/6 | R:R:${signal.riskReward}x | Regime:${niftyRegime} | Factor:${(adaptiveFactor * 100).toFixed(0)}%`,
   530→          tags: `auto,${signal.setupType === 'A+' ? 'aplus' : 'b'},score-${signal.score},${stock.sector || ''}`.replace(/,$/, ''),
   531→        },
   532→      });
   533→      await db.autoTradeLog.create({
   534→        data: {
   535→          action: 'AUTO_ENTRY', symbol: stock.symbol, tradeId: trade.id,
   536→          signal: JSON.stringify(signal), executed: true,
   537→          reason: `${signal.setupType} | Score ${signal.score}/6 | Qty ${qty} | R:R ${signal.riskReward}x | Factor:${(adaptiveFactor * 100).toFixed(0)}%`,
   538→        },
   539→      });
   540→
   541→      await resetDailyCounters();
   542→      const st2 = await getSchedulerState();
   543→      await setSchedulerKV('sched_todayEntries', String(st2.todayEntries + 1));
   544→      await setSchedulerKV('sched_scanCount', String(st2.scanCount + 1));
   545→
   546→      entries.push({
   547→        symbol: stock.symbol, setupType: signal.setupType, score: signal.score,
   548→        entryPrice: signal.entryPrice, qty, tradeId: trade.id,
   549→        adaptiveFactor, niftyRegime,
   550→      });
   551→    } catch (err) { skipped.push({ symbol: stock.symbol, reason: String(err) }); }
   552→  }
   553→  await recalcWallet();
   554→  return {
   555→    entries, skipped, l1Passed: l1Pass.length, l2Signals: entries.length,
   556→    totalScanned: stocks.length, circuitBreaker: false, niftyRegime,
   557→    adaptiveFactor, consecutiveLosses: consLosses, drawdownPct,
   558→  };
   559→}

   561→// ── Core Engine: Check Exits (v2 with health scoring) ─
   562→async function autoCheckExits() {
   563→  const rules = await getRules();
   564→  const openTrades = await db.paperTrade.findMany({ where: { status: 'OPEN' }, orderBy: { entryDate: 'asc' } });
   565→  const exits: any[] = [];
   566→  const holding: any[] = [];
   567→  const partialBooks: any[] = [];
   568→  const warnings: any[] = [];
   569→
   570→  const minsToClose = timeToClose();
   571→  const shouldTimeExit = minsToClose <= rules.timeExitMins && minsToClose > 0;
   572→
   573→  for (const trade of openTrades) {
   574→    try {
   575→      const { price: cp } = await getCurrentPrice(trade.symbol);
   576→      if (cp <= 0) { holding.push({ symbol: trade.symbol, error: 'Price unavailable' }); continue; }
   577→
   578→      const days = Math.floor((Date.now() - new Date(trade.entryDate).getTime()) / 86400000);
   579→      const risk = trade.entryPrice - trade.stopLoss;
   580→      const rMultiple = risk > 0 ? (cp - trade.entryPrice) / risk : 0;
   581→      let shouldExit = false;
   582→      let reason = '';
   583→      let exitPrice = cp;
   584→      let exitQty = trade.qty;
   585→
   586→      // 1. Hard SL hit
   587→      if (cp <= trade.stopLoss) {
   588→        shouldExit = true; reason = 'SL_HIT'; exitPrice = trade.stopLoss;
   589→      }
   590→      // 2. Hard TP hit
   591→      else if (cp >= trade.targetPrice) {
   592→        shouldExit = true; reason = 'TP_HIT'; exitPrice = trade.targetPrice;
   593→      }
   594→      // 3. Trailing stop (v2: ATR-based if configured)
   595→      else if (risk > 0 && rMultiple >= rules.trailingStopR) {
   596→        const trail = await getATRTrailLevel(trade.symbol, trade.entryPrice, cp, risk, rules);
   597→        if (trail.useATR) {
   598→          // ATR trailing: only exit if price drops below ATR-based level
   599→          if (cp <= trail.trailLevel) {
   600→            shouldExit = true; reason = 'ATR_TRAIL_STOP'; exitPrice = trail.trailLevel;
   601→          }
   602→        } else {
   603→          // Fixed R trailing (v1 behavior)
   604→          const trailLevel = trade.entryPrice + (risk * rules.trailToR);
   605→          if (cp <= trailLevel) {
   606→            shouldExit = true; reason = 'TRAIL_STOP'; exitPrice = trailLevel;
   607→          }
   608→        }
   609→      }
   610→      // 4. Max holding days
   611→      else if (days >= rules.maxHoldingDays) {
   612→        shouldExit = true; reason = 'MAX_HOLDING_DAYS';
   613→      }
   614→      // 5. v2: Stale loser acceleration — exit aging losing positions faster
   615→      else if (days >= rules.maxHoldingDays * 0.7 && rMultiple < -0.3) {
   616→        shouldExit = true; reason = 'STALE_LOSER';
   617→      }
   618→      // 6. Time-based exit
   619→      else if (shouldTimeExit) {
   620→        if (rMultiple < 0.5) {
   621→          shouldExit = true; reason = 'TIME_EXIT'; exitPrice = cp;
   622→        }
   623→      }
   624→
   625→      // 7. Partial booking (doesn't trigger full exit)
   626→      if (!shouldExit && risk > 0 && rMultiple >= rules.partialBookR && trade.qty > 1) {
   627→        const alreadyBooked = trade.tags?.includes('partial-booked');
   628→        if (!alreadyBooked) {
   629→          const bookQty = Math.max(1, Math.floor(trade.qty * (rules.partialBookPct / 100)));
   630→          const remainingQty = trade.qty - bookQty;
   631→          if (remainingQty >= 1) {
   632→            const bookPnl = (cp - trade.entryPrice) * bookQty;
   633→            await db.paperTrade.update({
   634→              where: { id: trade.id },
   635→              data: { qty: remainingQty, tags: (trade.tags ? trade.tags + ',' : '') + 'partial-booked' },
   636→            });
   637→            const w = await getWallet();
   638→            await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + bookPnl } });
   639→            await db.autoTradeLog.create({
   640→              action: 'PARTIAL_BOOK', symbol: trade.symbol, tradeId: trade.id,
   641→              executed: true, reason: `Booked ${bookQty}/${trade.qty + bookQty} at ${rMultiple.toFixed(1)}R | P&L: ₹${bookPnl.toLocaleString()}`,
   642→            });
   643→            partialBooks.push({ symbol: trade.symbol, bookedQty: bookQty, remainingQty, rMultiple, pnl: bookPnl, exitPrice: cp });
   644→          }
   645→        }
   646→      }
   647→
   648→      // Execute full exit
   649→      if (shouldExit) {
   650→        const pnl = (exitPrice - trade.entryPrice) * exitQty;
   651→        const pnlPct = ((exitPrice - trade.entryPrice) / trade.entryPrice) * 100;
   652→        await db.paperTrade.update({
   653→          where: { id: trade.id },
   654→          data: { status: 'CLOSED', exitDate: new Date(), exitPrice, pnl, pnlPercent: pnlPct, exitReason: reason },
   655→        });
   656→        const w = await getWallet();
   657→        await db.capitalWallet.update({ where: { id: w.id }, data: { realizedPnl: w.realizedPnl + pnl } });
   658→        await db.autoTradeLog.create({
   659→          action: `AUTO_EXIT_${reason}`, symbol: trade.symbol, tradeId: trade.id,
   660→          executed: true, reason: `${reason} | ₹${trade.entryPrice}→₹${exitPrice} | P&L: ₹${pnl.toLocaleString()} | ${rMultiple.toFixed(1)}R`,
   661→        });
   662→
   663→        // v2: Track consecutive losses for adaptive sizing
   664→        if (pnl < 0) {
   665→          const newStreak = (await getSchedulerState()).consecutiveLosses + 1;
   666→          await setSchedulerKV('sched_consecutiveLosses', String(newStreak));
   667→        } else {
   668→          await setSchedulerKV('sched_consecutiveLosses', '0');
   669→          // Reset adaptive factor on win
   670→          await setSchedulerKV('sched_lastAdaptiveFactor', '1');
   671→        }
   672→
   673→        await resetDailyCounters();
   674→        const st2 = await getSchedulerState();
   675→        await setSchedulerKV('sched_todayExits', String(st2.todayExits + 1));
   676→        await setSchedulerKV('sched_todayPnl', String(st2.todayPnl + pnl));
   677→
   678→        // v2: Check if we need to activate daily loss circuit breaker
   679→        const updatedSt = await getSchedulerState();
   680→        if (updatedSt.todayPnl < 0 && Math.abs(updatedSt.todayPnl) >= rules.dailyLossLimit) {
   681→          await setSchedulerKV('sched_circuitBreaker', 'true');
   682→          await setSchedulerKV('sched_circuitBreakerReason', `Daily loss ₹${Math.abs(updatedSt.todayPnl).toFixed(0)} >= ₹${rules.dailyLossLimit}`);
   683→        }
   684→
   685→        exits.push({
   686→          symbol: trade.symbol, exitReason: reason, entryPrice: trade.entryPrice, exitPrice,
   687→          pnl, pnlPercent: pnlPct, holdingDays: days, rMultiple: rMultiple.toFixed(2),
   688→        });
   689→      } else {
   690→        // v2: Position Health Score
   691→        const health = scorePositionHealth(cp, trade.entryPrice, trade.stopLoss, trade.targetPrice, days, rules.maxHoldingDays, rMultiple);
   692→        const healthData: any = {
   693→          symbol: trade.symbol, entryPrice: trade.entryPrice, currentPrice: cp,
   694→          pnl: (cp - trade.entryPrice) * trade.qty,
   695→          pnlPercent: ((cp - trade.entryPrice) / trade.entryPrice) * 100,
   696→          holdingDays: days, maxHoldingDays: rules.maxHoldingDays,
   697→          sl: trade.stopLoss, tp: trade.targetPrice, rMultiple: rMultiple.toFixed(2),
   698→          healthScore: health.score, healthUrgency: health.urgency, healthFactors: health.factors,
   699→        };
   700→
   701→        if (health.urgency === 'CRITICAL') {
   702→          healthData.slWarning = true;
   703→          warnings.push({ ...healthData, type: health.score < 20 ? 'SL_PROXIMITY' : 'AGING', message: `${trade.symbol} health ${health.score}/100 (${health.urgency})` });
   704→        } else if (health.urgency === 'WARNING') {
   705→          warnings.push({ ...healthData, type: 'AGING', message: `${trade.symbol} health ${health.score}/100` });
   706→        }
   707→
   708→        holding.push(healthData);
   709→      }
   710→    } catch { holding.push({ symbol: trade.symbol, error: 'Price fetch failed' }); }
   711→  }
   712→
   713→  if (exits.length > 0 || partialBooks.length > 0) await recalcWallet();
   714→  return { exits, holding, partialBooks, warnings, timeToClose: minsToClose, marketHours: isMarketHours() };
   715→}

   717→// ── Scheduler Control ──────────────────────────────────
   718→async function runSchedulerTick() {
   719→  if (!isMarketHours()) return { marketHours: false, message: 'Market closed' };
   720→  const state = await getSchedulerState();
   721→  if (!state.enabled) return { marketHours: true, enabled: false };
   722→
   723→  await resetDailyCounters();
   724→  const now = new Date();
   725→  const results: any = { scanResult: null, exitResult: null };
   726→
   727→  if (state.nextScanAt) {
   728→    const nextScan = new Date(state.nextScanAt);
   729→    if (now >= nextScan) {
   730→      const config: ScreeningConfig = DEFAULT_CONFIG;
   731→      results.scanResult = await autoScanAndTrade(config);
   732→      await setSchedulerKV('sched_lastScanAt', now.toISOString());
   733→      const next = new Date(now.getTime() + state.scanIntervalMin * 60000);
   734→      await setSchedulerKV('sched_nextScanAt', next.toISOString());
   735→    }
   736→  }
   737→
   738→  if (state.nextExitAt) {
   739→    const nextExit = new Date(state.nextExitAt);
   740→    if (now >= nextExit) {
   741→      results.exitResult = await autoCheckExits();
   742→      await setSchedulerKV('sched_lastExitAt', now.toISOString());
   743→      const next = new Date(now.getTime() + state.exitIntervalMin * 60000);
   744→      await setSchedulerKV('sched_nextExitAt', next.toISOString());
   745→    }
   746→  }
   747→
   748→  return { ...results, marketHours: true, enabled: true, schedulerState: await getSchedulerState() };
   749→}

   774→export async function POST(request: NextRequest) {
   775→  try {
   776→    const body = await request.json();
   777→    const action: string = body.action || 'get_status';
   778→    const config: ScreeningConfig = { ...DEFAULT_CONFIG, ...body.config };
   779→
   780→    switch (action) {
   781→      case 'scan_and_trade': {
   782→        const result = await autoScanAndTrade(config);
   783→        return NextResponse.json({ success: true, action, ...result });
   784→      }
   785→      case 'check_exits': {
   786→        const result = await autoCheckExits();
   787→        return NextResponse.json({ success: true, action, ...result });
   788→      }
   789→      case 'scheduler_tick': {
   790→        const result = await runSchedulerTick();
   791→        return NextResponse.json({ success: true, action, ...result });
   792→      }
   793→      case 'scheduler_toggle': {
   794→        const enabled = body.enabled;
   795→        if (enabled) {
   796→          const now = new Date();
   797→          const state = await getSchedulerState();
   798→          const scanMin = body.scanIntervalMin || state.scanIntervalMin;
   799→          const exitMin = body.exitIntervalMin || state.exitIntervalMin;
   800→          await setSchedulerKV('sched_enabled', 'true');
   801→          await setSchedulerKV('sched_scanIntervalMin', String(scanMin));
   802→          await setSchedulerKV('sched_exitIntervalMin', String(exitMin));
   803→          await setSchedulerKV('sched_nextScanAt', now.toISOString());
   804→          await setSchedulerKV('sched_nextExitAt', now.toISOString());
   805→          await resetDailyCounters();
   806→        } else {
   807→          await setSchedulerKV('sched_enabled', 'false');
   808→          await setSchedulerKV('sched_nextScanAt', '');
   809→          await setSchedulerKV('sched_nextExitAt', '');
   810→        }
   811→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   812→      }
   813→      case 'update_wallet': {
   814→        const { totalCapital } = body;
   815→        if (totalCapital) {
   816→          const w = await getWallet();
   817→          const open = await db.paperTrade.findMany({ where: { status: 'OPEN' } });
   818→          const deployed = open.reduce((s, t) => s + t.entryPrice * t.qty, 0);
   819→          await db.capitalWallet.update({ where: { id: w.id }, data: { totalCapital, available: totalCapital - deployed } });
   820→          const updated = await recalcWallet();
   821→          return NextResponse.json({ success: true, wallet: updated });
   822→        }
   823→        break;
   824→      }
   825→      case 'update_rules': {
   826→        const rules = body.rules;
   827→        if (rules) {
   828→          for (const [key, value] of Object.entries(rules)) {
   829→            await db.appSettings.upsert({
   830→              where: { key: `rules_${key}` },
   831→              create: { key: `rules_${key}`, value: String(value) },
   832→              update: { value: String(value) },
   833→            });
   834→          }
   835→          return NextResponse.json({ success: true, rules: await getRules() });
   836→        }
   837→        break;
   838→      }
   839→      case 'reset_circuit_breaker': {
   840→        // v2: Manual circuit breaker reset
   841→        await setSchedulerKV('sched_circuitBreaker', 'false');
   842→        await setSchedulerKV('sched_circuitBreakerReason', '');
   843→        return NextResponse.json({ success: true, scheduler: await getSchedulerState() });
   844→      }
   845→    }
   846→    const wallet = await recalcWallet();
   847→    return NextResponse.json({ success: true, wallet });
