   157→   157→const safeJson = async (res: Response | null) => {
   158→   158→  if (!res) return null;
   159→   159→  try { return await res.json(); } catch { return null; }
   160→   160→};
   161→   161→
   162→   162→const fetchSafe = async (url: string) => {
   163→   163→  const res = await fetch(url).catch(() => null);
   164→   164→  return res ? await safeJson(res) : null;
   165→   165→};
   166→   166→
   167→   167→const fmt = (n: number | null | undefined, decimals = 2) => {
   168→   168→  if (n == null) return '—';
   169→   169→  return n.toFixed(decimals);
   170→   170→};
   171→   171→
   172→   172→const fmtRupee = (n: number | null | undefined) => {
   173→   173→  if (n == null) return '—';
   174→   174→  const abs = Math.abs(n);
   175→   175→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)}L`;
   176→   176→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
   177→   177→  return `₹${n.toFixed(0)}`;
   178→   178→};

   228→   228→function OptionChainTab() {

   233→   233→
   234→   234→  const fetchChain = useCallback(async () => {
   235→   235→    setLoading(true);
   236→   236→    const expiry = chainData?.expiryDate || '';
   237→   237→    const url = `/api/options/chain?symbol=${symbol}${expiry ? `&expiry=${expiry}` : ''}`;
   238→   238→    const data = await fetchSafe(url);
   239→   239→    if (data?.success) {
   240→   240→      setChainData(data);
   241→   241→    }
   242→   242→    setLoading(false);
   243→   243→  }, [symbol, chainData?.expiryDate]);
   244→   244→
   245→   245→  useEffect(() => { fetchChain(); }, [symbol]);
