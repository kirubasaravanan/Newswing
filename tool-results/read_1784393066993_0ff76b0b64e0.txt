   272→   531→// ── Main Analytics Tab ─────────────────────────────────
   273→   532→export function AnalyticsTab() {
   274→   533→  const [trades, setTrades] = useState<any[]>([]);
   275→   534→  const [analyticsData, setAnalyticsData] = useState<any>(null);
   276→   535→  const [loading, setLoading] = useState(true);
   277→   536→
   278→   537→  const fetchAnalytics = useCallback(async () => {
   279→   538→    setLoading(true);
   280→   539→    try {
   281→   540→      const res = await fetch('/api/trades');
   282→   541→      const json = await res.json();
   283→   542→      const allTrades = json.trades || [];
   284→   543→      const closed = allTrades.filter((t: any) => t.status === 'CLOSED' && t.pnl != null);
   285→   544→
   286→   545→      const wins = closed.filter((t: any) => t.pnl > 0);
   287→   546→      const totalPnL = closed.reduce((s: number, t: any) => s + t.pnl, 0);
   288→   547→      const grossProfit = wins.reduce((s: number, t: any) => s + t.pnl, 0);
   289→   548→      const grossLoss = Math.abs(closed.filter((t: any) => t.pnl <= 0).reduce((s: number, t: any) => s + t.pnl, 0));
   290→   549→
   291→   550→      const monthMap = new Map<string, { pnl: number; trades: number }>();
   292→   551→      for (const t of closed) {
   293→   552→        const d = new Date(t.exitDate);
   294→   553→        const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   295→   554→        const prev = monthMap.get(key) || { pnl: 0, trades: 0 };
   296→   555→        prev.pnl += t.pnl;
   297→   556→        prev.trades += 1;
   298→   557→        monthMap.set(key, prev);
   299→   558→      }
   300→   559→      const byMonth = Array.from(monthMap.entries()).map(([month, v]) => ({
   301→   560→        month, pnl: Math.round(v.pnl), trades: v.trades,
   302→   561→      })).sort((a, b) => a.month.localeCompare(b.month));
   303→   562→
   304→   563→      const buckets = [
   305→   564→        { range: '< -5%', min: -Infinity, max: -5 }, { range: '-5% to -2%', min: -5, max: -2 },
   306→   565→        { range: '-2% to 0%', min: -2, max: 0 }, { range: '0% to 2%', min: 0, max: 2 },
   307→   566→        { range: '2% to 5%', min: 2, max: 5 }, { range: '> 5%', min: 5, max: Infinity },
   308→   567→      ];
   309→   568→      const pnlDistribution = buckets.map(b => ({
   310→   569→        range: b.range,
   311→   570→        count: closed.filter((t: any) => t.pnlPercent >= b.min && t.pnlPercent < b.max).length,
   312→   571→      }));
   313→   572→
   314→   573→      let peak = 0, running = 0, maxDD = 0;
   315→   574→      for (const t of closed.sort((a: any, b: any) => new Date(a.exitDate).getTime() - new Date(b.exitDate).getTime())) {
   316→   575→        running += t.pnl;
   317→   576→        peak = Math.max(peak, running);
   318→   577→        maxDD = Math.max(maxDD, (peak - running) / peak * 100);
   319→   578→      }
   320→   579→
   321→   580→      setAnalyticsData({
   322→   581→        trades: closed, totalPnL,
   323→   582→        winRate: closed.length > 0 ? (wins.length / closed.length) * 100 : 0,
   324→   583→        avgWin: wins.length > 0 ? grossProfit / wins.length : 0,
   325→   584→        avgLoss: closed.length - wins.length > 0 ? grossLoss / (closed.length - wins.length) : 0,
   326→   585→        profitFactor: grossLoss > 0 ? grossProfit / grossLoss : 0,
   327→   586→        maxDD, byMonth, pnlDistribution,
   328→   587→      });
   329→   588→    } catch { /* ignore */ }
   330→   589→    finally { setLoading(false); }
   331→   590→  }, []);
   332→   591→
