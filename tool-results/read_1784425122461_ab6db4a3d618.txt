     2→

     4→import { Button } from '@/components/ui/button';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Input } from '@/components/ui/input';
     7→import { Label } from '@/components/ui/label';
     8→import { Badge } from '@/components/ui/badge';
     9→import { ScrollArea } from '@/components/ui/scroll-area';
    10→import { Switch } from '@/components/ui/switch';
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    12→import {

    14→} from '@/components/ui/dialog';
    15→import {
    16→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    17→} from '@/components/ui/select';
    18→import {
    19→  Bot, Play, Wallet, Settings, TrendingUp, TrendingDown, Clock,
    20→  ArrowRightLeft, Shield, AlertTriangle, CheckCircle2, XCircle,
    21→  Wifi, WifiOff, Loader2, Plus, Minus, RefreshCw, Zap, Timer,
    22→  ChevronDown, ChevronUp, Eye, FileText, Activity,
    23→} from 'lucide-react';
    24→import { cn } from '@/lib/utils';
    25→import { toast } from 'sonner';
    26→import { useTradeStore } from '@/store/trade-store';
    27→import { TradingViewChart } from './tradingview-chart';
    28→
    29→// ── Types ───────────────────────────────────────────────
    30→interface WalletData {
    31→  id: string; totalCapital: number; deployed: number; available: number;
    32→  realizedPnl: number; unrealizedPnl: number;
    33→}
    34→
    35→interface PositionRules {
    36→  maxPerStock: number; maxBuysPerMonth: number; maxHoldingDays: number;
    37→  maxTotalPositions: number; riskPerTradePct: number;
    38→  trailingStopR: number; trailToR: number; partialBookR: number; partialBookPct: number;
    39→  cooldownDays: number; maxSectorPct: number; timeExitMins: number;
    40→}
    41→
    42→interface SchedulerState {
    43→  enabled: boolean; scanIntervalMin: number; exitIntervalMin: number;
    44→  lastScanAt: string | null; lastExitAt: string | null;
    45→  todayEntries: number; todayExits: number; todayPnl: number; scanCount: number;
    46→}
    47→
    48→interface AutoTradeLog {
    49→  id: string; action: string; symbol: string; executed: boolean;
    50→  reason: string; createdAt: string; signal?: string;
    51→}
    52→
    53→interface OpenPosition {
    54→  id: string; symbol: string; stockName: string | null;
    55→  entryPrice: number; qty: number; stopLoss: number; targetPrice: number;
    56→  entryDate: string; autoTraded: boolean; tags?: string;
    57→}
    58→
    59→export function AutoTradeTab() {
    60→  const { config } = useTradeStore();
    61→  const [wallet, setWallet] = useState<WalletData | null>(null);
    62→  const [rules, setRules] = useState<PositionRules>({
    63→    maxPerStock: 50000, maxBuysPerMonth: 3, maxHoldingDays: 25,
    64→    maxTotalPositions: 8, riskPerTradePct: 1.0,
    65→    trailingStopR: 1.5, trailToR: 0.5, partialBookR: 2.0, partialBookPct: 30,
    66→    cooldownDays: 3, maxSectorPct: 35, timeExitMins: 30,
    67→  });
    68→  const [openPositions, setOpenPositions] = useState<OpenPosition[]>([]);
    69→  const [logs, setLogs] = useState<AutoTradeLog[]>([]);
    70→  const [scheduler, setScheduler] = useState<SchedulerState | null>(null);
    71→  const [marketHours, setMarketHours] = useState(false);
    72→  const [timeToClose, setTimeToClose] = useState(0);
    73→  const [scanning, setScanning] = useState(false);
    74→  const [checkingExits, setCheckingExits] = useState(false);
    75→  const [editingWallet, setEditingWallet] = useState(false);
    76→  const [walletInput, setWalletInput] = useState('200000');
    77→  const [editingRules, setEditingRules] = useState(false);
    78→  const [ruleEdits, setRuleEdits] = useState<PositionRules>({ ...rules });
    79→  const [chartSymbol, setChartSymbol] = useState<string | null>(null);
    80→  const [exitWarnings, setExitWarnings] = useState<any[]>([]);
    81→  const [selectedLog, setSelectedLog] = useState<AutoTradeLog | null>(null);
    82→  const [showAllLogs, setShowAllLogs] = useState(false);
    83→
    84→  // ── Data Fetching ─────────────────────────────────────
    85→  const fetchStatus = useCallback(async () => {
    86→    try {
    87→      const res = await fetch('/api/auto-trade');
    88→      const data = await res.json();
    89→      if (data.success) {
    90→        setWallet(data.wallet);
    91→        setRules(data.rules);
    92→        setOpenPositions(data.openTrades);
    93→        setLogs(data.recentLogs || []);
    94→        setScheduler(data.scheduler);
    95→        setMarketHours(data.marketHours);
    96→        setTimeToClose(data.timeToClose);
    97→      }
    98→    } catch (err) { console.error('Fetch status error:', err); }
    99→  }, []);
   100→
   101→  useEffect(() => {
   102→    fetchStatus();
   103→    const interval = setInterval(fetchStatus, 30000);
   104→    return () => clearInterval(interval);
   105→  }, [fetchStatus]);
   106→
   107→  // ── Scheduler Tick (called periodically) ──────────────
   108→  const schedulerTickRef = useRef<NodeJS.Timeout | null>(null);
   109→  useEffect(() => {
   110→    if (scheduler?.enabled && marketHours) {
   111→      schedulerTickRef.current = setInterval(async () => {
   112→        try {
   113→          await fetch('/api/auto-trade', {
   114→            method: 'POST', headers: { 'Content-Type': 'application/json' },
   115→            body: JSON.stringify({ action: 'scheduler_tick' }),
   116→          });
   117→          fetchStatus(); // Refresh UI
   118→        } catch { /* ignore */ }
   119→      }, 60000); // Check every minute
   120→    }
   121→    return () => { if (schedulerTickRef.current) clearInterval(schedulerTickRef.current); };
   122→  }, [scheduler?.enabled, marketHours, fetchStatus]);
   123→
   124→  // ── Actions ───────────────────────────────────────────
   125→  const handleScanAndTrade = async () => {
   126→    setScanning(true);
   127→    try {
   128→      const res = await fetch('/api/auto-trade', {
   129→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   130→        body: JSON.stringify({ action: 'scan_and_trade', config }),
   131→      });
   132→      const data = await res.json();
   133→      if (data.success) {
   134→        toast.success(`Scan Complete`, {
   135→          description: `${data.totalScanned} scanned → ${data.l1Passed} L1 pass → ${data.l2Signals} entries, ${data.skipped.length} skipped`,
   136→        });
   137→        fetchStatus();
   138→      } else {
   139→        toast.error('Auto-trade failed', { description: data.error });
   140→      }
   141→    } catch { toast.error('Auto-trade failed'); }
   142→    finally { setScanning(false); }
   143→  };
   144→
   145→  const handleCheckExits = async () => {
   146→    setCheckingExits(true);
   147→    try {
   148→      const res = await fetch('/api/auto-trade', {
   149→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   150→        body: JSON.stringify({ action: 'check_exits' }),
   151→      });
   152→      const data = await res.json();
   153→      if (data.success) {
   154→        if (data.exits.length > 0) {
   155→          toast.success(`Exit Check: ${data.exits.length} closed`, {
   156→            description: data.exits.map((e: any) => `${e.symbol} (${e.exitReason})`).join(', '),
   157→          });
   158→        }
   159→        if (data.partialBooks?.length > 0) {
   160→          toast.info(`Partial Book: ${data.partialBooks.length} positions`, {
   161→            description: data.partialBooks.map((p: any) => `${p.symbol}: ${p.bookedQty}qty at ${p.rMultiple}R`).join(', '),
   162→          });
   163→        }
   164→        if (data.exits.length === 0 && (!data.partialBooks || data.partialBooks.length === 0)) {
   165→          toast.info('No exits triggered', { description: `${data.holding?.length || 0} positions holding` });
   166→        }
   167→        if (data.warnings?.length > 0) setExitWarnings(data.warnings);
   168→        fetchStatus();
   169→      }
   170→    } catch { toast.error('Exit check failed'); }
   171→    finally { setCheckingExits(false); }
   172→  };
   173→
   174→  const handleSchedulerToggle = async (enabled: boolean) => {
   175→    try {
   176→      const res = await fetch('/api/auto-trade', {
   177→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   178→        body: JSON.stringify({
   179→          action: 'scheduler_toggle', enabled,
   180→          scanIntervalMin: ruleEdits.scanIntervalMin || 30,
   181→          exitIntervalMin: ruleEdits.exitIntervalMin || 5,
   182→        }),
   183→      });
   184→      const data = await res.json();
   185→      if (data.success) {
   186→        setScheduler(data.scheduler);
   187→        toast.success(enabled ? 'Auto-Mode Armed' : 'Auto-Mode Disarmed');
   188→        fetchStatus();
   189→      }
   190→    } catch { toast.error('Scheduler toggle failed'); }
   191→  };
   192→
   193→  const saveWallet = async () => {
   194→    try {
   195→      const res = await fetch('/api/auto-trade', {
   196→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   197→        body: JSON.stringify({ action: 'update_wallet', totalCapital: parseFloat(walletInput) }),
   198→      });
   199→      const data = await res.json();
   200→      if (data.success) { setWallet(data.wallet); setEditingWallet(false); toast.success('Wallet updated'); }
   201→    } catch { toast.error('Failed to update wallet'); }
   202→  };
   203→
   204→  const saveRules = async () => {
   205→    try {
   206→      const res = await fetch('/api/auto-trade', {
   207→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ action: 'update_rules', rules: ruleEdits }),
   209→      });
   210→      const data = await res.json();
   211→      if (data.success) { setRules(data.rules); setEditingRules(false); toast.success('Rules updated'); }
   212→    } catch { toast.error('Failed to update rules'); }
   213→  };
   214→
   215→  const totalPnl = (wallet?.realizedPnl || 0) + (wallet?.unrealizedPnl || 0);
   216→  const pnlPct = wallet?.totalCapital ? (totalPnl / (wallet.totalCapital - wallet.realizedPnl)) * 100 : 0;
   217→
   218→  return (
   219→    <div className="space-y-4">
   220→      {/* ── Top Bar ─────────────────────────────────────── */}
   221→      <div className="flex items-center justify-between flex-wrap gap-3">
   222→        <div className="flex items-center gap-3">
   223→          <h2 className="text-lg font-bold flex items-center gap-2">
   224→            <Bot className="h-5 w-5 text-primary" />
   225→            Auto-Trade Engine
   226→          </h2>
   227→          {/* Market Status */}
   228→          <div className={cn(
   229→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   230→            marketHours ? 'bg-emerald-500/10 text-emerald-400' : 'bg-muted text-muted-foreground'
   231→          )}>
   232→            <div className={cn('h-1.5 w-1.5 rounded-full', marketHours ? 'bg-emerald-400 animate-pulse' : 'bg-muted-foreground')} />
   233→            {marketHours ? `Market Open — ${timeToClose}m to close` : 'Market Closed'}
   234→          </div>
   235→          {/* Scheduler Armed Status */}
   236→          <div className={cn(
   237→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   238→            scheduler?.enabled ? 'bg-indigo-500/10 text-indigo-400' : 'bg-muted text-muted-foreground'
   239→          )}>
   240→            <Shield className="h-3 w-3" />
   241→            {scheduler?.enabled ? 'Auto-Mode Armed' : 'Manual'}
   242→          </div>
   243→        </div>
   244→        <div className="flex items-center gap-2">
   245→          <Button variant="outline" size="sm" onClick={handleCheckExits}
   246→            disabled={checkingExits || openPositions.length === 0}
   247→            className="h-8 gap-1.5 text-xs">
   248→            {checkingExits ? <Loader2 className="h-3 w-3 animate-spin" /> : <ArrowRightLeft className="h-3 w-3" />}
   249→            Check Exits
   250→          </Button>
   251→          <Button onClick={handleScanAndTrade} disabled={scanning || !marketHours}
   252→            className={cn('gap-1.5', scheduler?.enabled ? 'bg-emerald-600 hover:bg-emerald-700' : '')}
   253→            size="sm">
   254→            {scanning ? <Loader2 className="h-3 w-3 animate-spin" /> : <Zap className="h-3 w-3" />}
   255→            {scanning ? 'Scanning Universe...' : 'Scan & Auto-Trade'}
   256→          </Button>
   257→        </div>
   258→      </div>
   259→
   260→      <Tabs defaultValue="engine" className="space-y-4">
   261→        <TabsList className="bg-secondary/50">
   262→          <TabsTrigger value="engine" className="text-xs gap-1.5"><Bot className="h-3 w-3" /> Engine</TabsTrigger>
   263→          <TabsTrigger value="scheduler" className="text-xs gap-1.5"><Timer className="h-3 w-3" /> Scheduler</TabsTrigger>
   264→          <TabsTrigger value="positions" className="text-xs gap-1.5"><TrendingUp className="h-3 w-3" /> Positions</TabsTrigger>
   265→          <TabsTrigger value="audit" className="text-xs gap-1.5"><FileText className="h-3 w-3" /> Audit Log</TabsTrigger>
   266→          <TabsTrigger value="rules" className="text-xs gap-1.5"><Settings className="h-3 w-3" /> Rules</TabsTrigger>
   267→        </TabsList>
   268→
   269→        {/* ── ENGINE TAB ──────────────────────────────────── */}
   270→        <TabsContent value="engine" className="space-y-4">
   271→          <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   272→            {/* Wallet Card */}
   273→            <Card className="lg:col-span-2 border-border">
   274→              <CardContent className="p-4">
   275→                <div className="flex items-center justify-between mb-4">
   276→                  <h3 className="text-sm font-semibold flex items-center gap-2">
   277→                    <Wallet className="h-4 w-4 text-primary" /> Capital Wallet
   278→                  </h3>
   279→                  {!editingWallet ? (
   280→                    <Button variant="ghost" size="sm" onClick={() => { setWalletInput(String(wallet?.totalCapital || 200000)); setEditingWallet(true); }} className="h-7 text-xs">Edit</Button>
   281→                  ) : (
   282→                    <div className="flex items-center gap-2">
   283→                      <Input value={walletInput} onChange={e => setWalletInput(e.target.value)} className="h-7 w-32 text-xs" />
   284→                      <Button size="sm" onClick={saveWallet} className="h-7 text-xs">Save</Button>
   285→                      <Button size="sm" variant="ghost" onClick={() => setEditingWallet(false)} className="h-7 text-xs">Cancel</Button>
   286→                    </div>
   287→                  )}
   288→                </div>
   289→                {wallet && (
   290→                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   291→                    <div className="rounded-lg bg-secondary/50 p-3">
   292→                      <div className="text-[10px] text-muted-foreground uppercase">Total Capital</div>
   293→                      <div className="text-lg font-bold font-mono mt-0.5">₹{wallet.totalCapital.toLocaleString()}</div>
   294→                    </div>
   295→                    <div className="rounded-lg bg-blue-500/10 p-3">
   296→                      <div className="text-[10px] text-blue-400 uppercase">Deployed</div>
   297→                      <div className="text-lg font-bold font-mono mt-0.5 text-blue-400">₹{wallet.deployed.toLocaleString()}</div>
   298→                      <div className="text-[10px] text-muted-foreground">{wallet.totalCapital > 0 ? ((wallet.deployed / wallet.totalCapital) * 100).toFixed(1) : 0}%</div>
   299→                    </div>
   300→                    <div className="rounded-lg bg-emerald-500/10 p-3">
   301→                      <div className="text-[10px] text-emerald-400 uppercase">Available</div>
   302→                      <div className="text-lg font-bold font-mono mt-0.5 text-emerald-400">₹{wallet.available.toLocaleString()}</div>
   303→                    </div>
   304→                    <div className={cn('rounded-lg p-3', totalPnl >= 0 ? 'bg-emerald-500/10' : 'bg-red-500/10')}>
   305→                      <div className={cn('text-[10px] uppercase', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>Total P&L</div>
   306→                      <div className={cn('text-lg font-bold font-mono mt-0.5', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   307→                        {totalPnl >= 0 ? '+' : ''}₹{totalPnl.toLocaleString()}
   308→                      </div>
   309→                      <div className="text-[10px] text-muted-foreground">
   310→                        R: {wallet.realizedPnl.toLocaleString()} | U: {wallet.unrealizedPnl.toLocaleString()}
   311→                      </div>
   312→                    </div>
   313→                  </div>
   314→                )}
   315→              </CardContent>
   316→            </Card>
   317→
   318→            {/* Today's Summary */}
   319→            <Card className="border-border">
   320→              <CardContent className="p-4">
   321→                <h3 className="text-sm font-semibold flex items-center gap-2 mb-3">
   322→                  <Activity className="h-4 w-4 text-primary" /> Today&apos;s Summary
   323→                </h3>
   324→                {scheduler && (
   325→                  <div className="space-y-2">
   326→                    <div className="flex justify-between text-xs">
   327→                      <span className="text-muted-foreground">Total Scans</span>
   328→                      <span className="font-mono font-bold">{scheduler.scanCount}</span>
   329→                    </div>
   330→                    <div className="flex justify-between text-xs">
   331→                      <span className="text-muted-foreground">Entries Today</span>
   332→                      <span className="font-mono font-bold text-emerald-400">{scheduler.todayEntries}</span>
   333→                    </div>
   334→                    <div className="flex justify-between text-xs">
   335→                      <span className="text-muted-foreground">Exits Today</span>
   336→                      <span className="font-mono font-bold text-red-400">{scheduler.todayExits}</span>
   337→                    </div>
   338→                    <div className="flex justify-between text-xs pt-2 border-t border-border/50">
   339→                      <span className="text-muted-foreground">Today P&L</span>
   340→                      <span className={cn('font-mono font-bold', scheduler.todayPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   341→                        {scheduler.todayPnl >= 0 ? '+' : ''}₹{Math.round(scheduler.todayPnl).toLocaleString()}
   342→                      </span>
   343→                    </div>
   344→                    {scheduler.lastScanAt && (
   345→                      <div className="text-[10px] text-muted-foreground pt-1">
   346→                        Last scan: {new Date(scheduler.lastScanAt).toLocaleTimeString()}
   347→                      </div>
   348→                    )}
   349→                    {scheduler.lastExitAt && (
   350→                      <div className="text-[10px] text-muted-foreground">
   351→                        Last exit check: {new Date(scheduler.lastExitAt).toLocaleTimeString()}
   352→                      </div>
   353→                    )}
   354→                  </div>
   355→                )}
   356→              </CardContent>
   357→            </Card>
   358→          </div>
   359→
   360→          {/* Position Health Warnings */}
   361→          {exitWarnings.length > 0 && (
   362→            <Card className="border-amber-500/30 bg-amber-500/5">
   363→              <CardContent className="p-4">
   364→                <h3 className="text-sm font-semibold flex items-center gap-2 mb-2">
   365→                  <AlertTriangle className="h-4 w-4 text-amber-400" />
   366→                  Position Warnings ({exitWarnings.length})
   367→                </h3>
   368→                <div className="space-y-1.5">
   369→                  {exitWarnings.map((w: any, i: number) => (
   370→                    <div key={i} className={cn(
   371→                      'flex items-center gap-2 text-xs px-3 py-2 rounded-lg',
   372→                      w.type === 'SL_PROXIMITY' ? 'bg-red-500/10' : 'bg-amber-500/10'
   373→                    )}>
   374→                      <Badge variant="outline" className={cn('text-[9px] h-4',
   375→                        w.type === 'SL_PROXIMITY' ? 'text-red-400 border-red-500/30' : 'text-amber-400 border-amber-500/30'
   376→                      )}>{w.type === 'SL_PROXIMITY' ? 'SL RISK' : 'AGING'}</Badge>
   377→                      <span className="font-semibold">{w.symbol}</span>
   378→                      <span className="text-muted-foreground">{w.message}</span>
   379→                      <span className={cn('ml-auto font-mono', (w.pnlPercent || 0) >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   380→                        {w.pnlPercent >= 0 ? '+' : ''}{w.pnlPercent?.toFixed(2)}%
   381→                      </span>
   382→                    </div>
   383→                  ))}
   384→                </div>
   385→              </CardContent>
   386→            </Card>
   387→          )}
   388→        </TabsContent>
   389→
   390→        {/* ── SCHEDULER TAB ──────────────────────────────── */}
   391→        <TabsContent value="scheduler" className="space-y-4">
   392→          <Card className="border-border">
   393→            <CardContent className="p-4">
   394→              <div className="flex items-center justify-between mb-4">
   395→                <div>
   396→                  <h3 className="text-sm font-semibold flex items-center gap-2">
   397→                    <Timer className="h-4 w-4 text-primary" /> Automation Scheduler
   398→                  </h3>
   399→                  <p className="text-[10px] text-muted-foreground mt-1">
   400→                    Automatically scans for entries and manages exits during market hours (9:15 AM - 3:30 PM IST)
   401→                  </p>
   402→                </div>
   403→                <div className="flex items-center gap-3">
   404→                  <span className={cn('text-xs', marketHours ? 'text-emerald-400' : 'text-muted-foreground')}>
   405→                    {marketHours ? 'Market is OPEN' : 'Market is CLOSED'}
   406→                  </span>
   407→                  <Switch
   408→                    checked={scheduler?.enabled || false}
   409→                    onCheckedChange={handleSchedulerToggle}
   410→                  />
   411→                </div>
   412→              </div>
   413→
   414→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   415→                <div className="rounded-lg bg-secondary/50 p-4 space-y-3">
   416→                  <h4 className="text-xs font-semibold flex items-center gap-2">
   417→                    <Zap className="h-3 w-3 text-emerald-400" /> Entry Scan Settings
   418→                  </h4>
   419→                  <div className="flex items-center justify-between text-xs">
   420→                    <Label className="text-muted-foreground">Scan every (minutes)</Label>
   421→                    <Input type="number" value={ruleEdits.scanIntervalMin || 30}
   422→                      onChange={e => setRuleEdits({ ...ruleEdits, scanIntervalMin: parseInt(e.target.value) || 30 })}
   423→                      className="h-7 w-20 text-xs text-right" />
   424→                  </div>
   425→                  <p className="text-[10px] text-muted-foreground">
   426→                    V-Swing universe scan will run automatically. New A+/B signals will be entered if rules permit.
   427→                  </p>
   428→                </div>
   429→
   430→                <div className="rounded-lg bg-secondary/50 p-4 space-y-3">
   431→                  <h4 className="text-xs font-semibold flex items-center gap-2">
   432→                    <ArrowRightLeft className="h-3 w-3 text-red-400" /> Exit Check Settings
   433→                  </h4>
   434→                  <div className="flex items-center justify-between text-xs">
