   100→    diMinus: number;

   104→// ── Helper: lookback with index safety ──────────────────────
   105→
   106→function get<T>(arr: T[], index: number, fallback: T): T {
   107→  if (index < 0 || index >= arr.length) return fallback;
   108→  return arr[index];
   109→}

   111→// ── Core Engine ────────────────────────────────────────────
   112→
   113→export function runScreening(
   114→  symbol: string,
   115→  candles: OHLCV[],
   116→  niftyCandles: OHLCV[],
   117→  config: ScreeningConfig = DEFAULT_CONFIG
   118→): ScreeningResult | null {
   119→  const len = candles.length;
   120→  if (len < 250) return null; // Need enough data for SMA 200
   121→
   122→  const closes = candles.map(c => c.close);
   123→  const highs = candles.map(c => c.high);
   124→  const lows = candles.map(c => c.low);
   125→  const volumes = candles.map(c => c.volume);
   126→
   127→  // ── Indicators ─────────────────────────────────────────
   128→  const sma200Arr = SMA.calculate({ period: 200, values: closes });
   129→  const ema20Arr = EMA.calculate({ period: 20, values: closes });
   130→  const ema10Arr = EMA.calculate({ period: 10, values: closes });
   131→  const rsi14Arr = RSI.calculate({ period: 14, values: closes });
   132→  const atr14Arr = ATR.calculate({ period: 14, high: highs, low: lows, close: closes });
   133→  const dmiResult = ADX.calculate({ period: 14, high: highs, low: lows, close: closes });
   134→
   135→  // Pad arrays to match candle length
   136→  const sma200 = padArray(sma200Arr, len);
   137→  const ema20 = padArray(ema20Arr, len);
   138→  const ema10 = padArray(ema10Arr, len);
   139→  const rsi14 = padArray(rsi14Arr, len);
   140→  const atr14 = padArray(atr14Arr, len);
   141→  const adxArr = dmiResult.map(d => d.adx);
   142→  const diPlusArr = dmiResult.map(d => d.pdi);
   143→  const diMinusArr = dmiResult.map(d => d.mdi);
   144→  const adx = padArray(adxArr, len);
   145→  const diPlus = padArray(diPlusArr, len);
   146→  const diMinus = padArray(diMinusArr, len);
   147→
   148→  // ── Current bar (last bar) ─────────────────────────────
   149→  const i = len - 1; // current bar index
   150→  const i1 = len - 2; // previous bar
   151→
   152→  // Need at least 2 bars of valid indicator data
   153→  if (!sma200[i] || !ema20[i] || !atr14[i] || !rsi14[i] || !adx[i]) return null;
   154→
   155→  const currentClose = closes[i];
   156→  const currentOpen = candles[i].open;
   157→  const currentHigh = highs[i];
   158→  const currentLow = lows[i];
   159→  const currentVolume = volumes[i];
   160→  const prevHigh = get(highs, i1, currentHigh);
   161→
   162→  // ── Adaptive range (ATR-based lookback) ────────────────
   163→  const atrSma20 = calcSMA(atr14.slice(0, i + 1).filter(v => v != null), 20);
   164→  const currentAtr = atr14[i]!;
   165→  const rangeLen = currentAtr > atrSma20 ? 30 : 60;
   166→
   167→  const structHigh = Math.max(...highs.slice(Math.max(0, i - rangeLen), i));
   168→  const structLow = Math.min(...lows.slice(Math.max(0, i - rangeLen), i));
   169→
   170→  // ── ADX / Trending check ──────────────────────────────
   171→  const prevAdx = get(adx, i1, 0);
   172→  const adxRising = adx[i]! > prevAdx;
   173→  const adxHealthy = adx[i]! > 15;
   174→  const marketTrending = (adx[i]! > 18) || (adxRising && adxHealthy);
   175→
   176→  // ── Volatility check ───────────────────────────────────
   177→  const volLo = marketTrending ? 0.6 : 0.8;
   178→  const volHi = marketTrending ? 1.8 : 1.3;
   179→  const validVolatility = currentAtr > (atrSma20 * volLo) && currentAtr < (atrSma20 * volHi);
   180→
   181→  // ── Macro / Nifty check ────────────────────────────────
   182→  let regimeSafe = true;
   183→  if (config.useMacro && niftyCandles.length >= 200) {
   184→    const niftyCloses = niftyCandles.map(c => c.close);
   185→    const nifty200Arr = SMA.calculate({ period: 200, values: niftyCloses });
   186→    const nifty200 = nifty200Arr[nifty200Arr.length - 1];
   187→    regimeSafe = nifty200 != null && currentClose > nifty200;
   188→  }
   189→
   190→  // ── Liquidity check ────────────────────────────────────
   191→  const turnoverCr = (currentClose * currentVolume) / 10000000;
   192→  const turnoverMA20 = calcSMA(
   193→    candles.slice(Math.max(0, i - 20), i).map(c => (c.close * c.volume) / 10000000),
   194→    20
   195→  );
   196→  const liquid = turnoverMA20 >= config.minTurnoverCr;
   197→
   198→  // ── Volume MA ─────────────────────────────────────────
   199→  const volMA20 = calcSMA(volumes.slice(Math.max(0, i - 20), i + 1), 20);
   200→  const minLiquidityCheck = currentVolume > volMA20;
   201→
   202→  // ── Not extended check ─────────────────────────────────
   203→  const notExtended = currentClose < (ema20[i]! * 1.06);
   204→
   205→  // ── Confluence Scores (exact Pine Script logic) ────────
   206→
   207→  // Score 1: Trend
   208→  const scoreTrend = (currentClose > sma200[i]! && rsi14[i]! > 52) ? 1 : 0;
   209→
   210→  // Score 2: Pullback
   211→  const lowest4 = Math.min(...lows.slice(Math.max(0, i - 4), i + 1));
   212→  const scorePullback = (lowest4 < ema20[i]! * 1.02 && currentClose > ema20[i]!) ? 1 : 0;
   213→
   214→  // Score 3: Trigger
   215→  const scoreTrigger = currentClose > prevHigh ? 1 : 0;
   216→
   217→  // Score 4: Volume
   218→  const prevVolMA20 = calcSMA(volumes.slice(Math.max(0, i1 - 20), i1 + 1), 20);
   219→  const scoreVolume = (currentVolume > volMA20 * 1.05 || get(volumes, i1, 0) > prevVolMA20 * 1.1) ? 1 : 0;
   220→
   221→  // Score 5: Relative Strength (vs Nifty)
   222→  let scoreRS = 0;
   223→  if (niftyCandles.length >= 20) {
   224→    const niftyClose = niftyCandles[niftyCandles.length - 1]?.close ?? 0;
   225→    if (niftyClose > 0) {
   226→      // Calculate rolling RS for last 25 bars
   227→      const rsWindow = Math.min(25, candles.length, niftyCandles.length);
   228→      const rsValues: number[] = [];
   229→      for (let j = len - rsWindow; j < len; j++) {
   230→        const nj = niftyCandles.length - rsWindow + (j - (len - rsWindow));
   231→        if (nj >= 0 && niftyCandles[nj]?.close > 0) {
   232→          rsValues.push(closes[j] / niftyCandles[nj].close);
   233→        }
   234→      }
   235→      const rsMA = calcSMA(rsValues, 20);
   236→      const currentRS = closes[i] / niftyClose;
   237→      const rs5ago = rsValues.length > 5 ? rsValues[rsValues.length - 6] : 0;
   238→      scoreRS = (currentRS > rsMA || currentRS > rs5ago) ? 1 : 0;
   239→    }
   240→  }
   241→
   242→  // Score 6: Gap
   243→  const prevClose = get(closes, i1, currentClose);
   244→  const gapPct = Math.abs(currentOpen - prevClose) / prevClose * 100;
   245→  const scoreGap = gapPct < 3.5 ? 1 : 0;
   246→
   247→  const totalScore = scoreTrend + scorePullback + scoreTrigger + scoreVolume + scoreRS + scoreGap;
   248→
   249→  // ── Regime Clear check ─────────────────────────────────
   250→  const regimeClear = regimeSafe && liquid && minLiquidityCheck && marketTrending && validVolatility && notExtended;
   251→
   252→  // ── Entry / SL / TP calculation ────────────────────────
   253→  const theoreticalEP = Math.max(currentOpen, prevHigh);
   254→  const stopLow5 = Math.min(...lows.slice(Math.max(0, i - 5), i));
   255→  const rawSL = stopLow5 - (get(atr14, i1, currentAtr) * 0.8);
   256→  const riskPerShare = theoreticalEP - rawSL;
   257→
   258→  if (riskPerShare <= 0) return null;
   259→
   260→  const structRange = structHigh - structLow;
   261→  const expectedReward = structRange;
   262→  const rr = expectedReward / riskPerShare;
   263→
   264→  if (rr < config.minRR && totalScore < 6) return null;
   265→
   266→  // ── Position Sizing ────────────────────────────────────
   267→  const tp1 = theoreticalEP + (riskPerShare * 1.5);
   268→
   269→  // A+ sizing
   270→  const riskAmtA = config.liveCapital * (config.riskPct / 100);
   271→  const qtyRiskA = Math.floor(riskAmtA / (riskPerShare * 1)); // pointvalue = 1 for NSE cash
   272→  const qtyCapA = Math.floor((config.liveCapital / config.maxSlots) / theoreticalEP);
   273→  const qtyA = Math.max(0, Math.min(qtyRiskA, qtyCapA));
   274→
   275→  // B sizing (half risk, 60% capital)
   276→  const riskAmtB = config.liveCapital * ((config.riskPct * 0.5) / 100);
   277→  const qtyRiskB = Math.floor(riskAmtB / (riskPerShare * 1));
   278→  const qtyCapB = Math.floor(((config.liveCapital / config.maxSlots) * 0.6) / theoreticalEP);
   279→  const qtyB = Math.max(0, Math.min(qtyRiskB, qtyCapB));
   280→
   281→  // ── Final assembly ─────────────────────────────────────
   282→  if (totalScore < config.minScore) return null;
   283→  if (!regimeClear) return null;
   284→
   285→  // Strong momentum check (gap < 5% for score 6)
   286→  const strongMomentum = totalScore === 6 && gapPct < 5.0;
   287→  const withinGap = currentOpen <= (prevHigh * 1.03);
   288→  if (!withinGap && !strongMomentum) return null;
   289→
   290→  return {
   291→    symbol,
   292→    date: candles[i].date,
   293→    score: totalScore,
   294→    setupType: totalScore === 6 ? 'A+' : 'B',
   295→    entryPrice: Math.round(theoreticalEP * 100) / 100,
   296→    stopLoss: Math.round(rawSL * 100) / 100,
   297→    targetPrice: Math.round(tp1 * 100) / 100,
   298→    riskReward: Math.round(rr * 100) / 100,
   299→    atr: Math.round(currentAtr * 100) / 100,
   300→    rsi: Math.round(rsi14[i]! * 100) / 100,
   301→    scores: {
   302→      scoreTrend,
   303→      scorePullback,
   304→      scoreTrigger,
   305→      scoreVolume,
   306→      scoreRS,
   307→      scoreGap,
   308→      totalScore,
   309→    },
   310→    checks: {
   311→      trendAbove: scoreTrend === 1,
   312→      pullbackOk: scorePullback === 1,
   313→      triggerOk: scoreTrigger === 1,
   314→      volumeOk: scoreVolume === 1,
   315→      rsOk: scoreRS === 1,
   316→      gapOk: scoreGap === 1,
   317→      regimeSafe,
   318→      liquid,
   319→      trending: marketTrending,
   320→      validVol: validVolatility,
   321→      notExtended,
   322→    },
   323→    sizing: {
   324→      qtyA,
   325→      qtyB,
   326→      riskPerShareA: Math.round(riskPerShare * 100) / 100,
   327→      riskPerShareB: Math.round(riskPerShare * 100) / 100,
   328→      riskAmtA: Math.round(riskAmtA),
   329→      riskAmtB: Math.round(riskAmtB),
   330→    },
   331→    indicators: {
   332→      sma200: Math.round(sma200[i]! * 100) / 100,
   333→      ema20: Math.round(ema20[i]! * 100) / 100,
   334→      ema10: Math.round(get(ema10, i, 0) * 100) / 100,
   335→      adx: Math.round(adx[i]! * 100) / 100,
   336→      diPlus: Math.round(get(diPlus, i, 0) * 100) / 100,
   337→      diMinus: Math.round(get(diMinus, i, 0) * 100) / 100,
   338→    },
   339→  };
   340→}

   344→export interface BacktestTradeResult {
   345→  symbol: string;
   346→  entryDate: string;
   347→  entryPrice: number;
   348→  exitDate: string;
   349→  exitPrice: number;
   350→  qty: number;
   351→  pnl: number;
   352→  pnlPercent: number;
   353→  score: number;
   354→  setupType: string;
   355→  exitReason: string;
   356→}

   358→export interface BacktestResult {
   359→  trades: BacktestTradeResult[];
   360→  equityCurve: { date: string; equity: number }[];
   361→  stats: {
   362→    totalTrades: number;
   363→    winTrades: number;
   364→    lossTrades: number;
   365→    winRate: number;
   366→    profitFactor: number;
   367→    maxDrawdown: number;
   368→    avgWin: number;
   369→    avgLoss: number;
   370→    bestTrade: number;
   371→    worstTrade: number;
   372→    finalCapital: number;
   373→    sharpeRatio: number;
   374→  };
   375→}
   376→
   377→export function runBacktest(
   378→  symbol: string,
   379→  candles: OHLCV[],
   380→  niftyCandles: OHLCV[],
   381→  config: ScreeningConfig = DEFAULT_CONFIG
   382→): BacktestResult {
   383→  const { maxSlots, maxOpenTrades, maxHoldBars, riskPct, liveCapital } = config;
   384→  const trades: BacktestTradeResult[] = [];
   385→  const equityCurve: { date: string; equity: number }[] = [];
   386→  
   387→  let capital = liveCapital;
   388→  let peakEquity = capital;
   389→  let maxDD = 0;
   390→  let lastTradeBar = -999;
   391→
   392→  // Track open positions for pyramiding
   393→  interface OpenPosition {
   394→    entryBar: number;
   395→    entryPrice: number;
   396→    qty: number;
   397→    stopLoss: number;
   398→    tp1: number;
   399→    score: number;
   400→    setupType: string;
   401→    partialTaken: boolean;
   402→    trailActive: boolean;
   403→    totalQty: number;
   404→    avgPrice: number;
   405→    blendedSL: number;
   406→  }
   407→  let openPositions: OpenPosition[] = [];
   408→
   409→  // Pre-compute all indicators for the full dataset
   410→  const closes = candles.map(c => c.close);
   411→  const highs = candles.map(c => c.high);
   412→  const lows = candles.map(c => c.low);
   413→  const volumes = candles.map(c => c.volume);
   414→
   415→  const sma200Full = padArray(SMA.calculate({ period: 200, values: closes }), candles.length);
   416→  const ema20Full = padArray(EMA.calculate({ period: 20, values: closes }), candles.length);
   417→  const ema10Full = padArray(EMA.calculate({ period: 10, values: closes }), candles.length);
   418→  const rsi14Full = padArray(RSI.calculate({ period: 14, values: closes }), candles.length);
   419→  const atr14Full = padArray(ATR.calculate({ period: 14, high: highs, low: lows, close: closes }), candles.length);
   420→  const adxResultFull = ADX.calculate({ period: 14, high: highs, low: lows, close: closes });
   421→  const adxFull = padArray(adxResultFull.map(d => d.adx), candles.length);
   422→
   423→  // Nifty data
   424→  let nifty200Full: (number | null)[] = [];
   425→  if (niftyCandles.length >= 200) {
   426→    const nCloses = niftyCandles.map(c => c.close);
   427→    nifty200Full = padArray(SMA.calculate({ period: 200, values: nCloses }), niftyCandles.length);
   428→  }
   429→
   430→  // Walk through each bar
   431→  for (let barIdx = 201; barIdx < candles.length; barIdx++) {
   432→    const bar = candles[barIdx];
   433→    
   434→    // Check exits for all open positions
   435→    const closedIndices: number[] = [];
   436→    for (let p = 0; p < openPositions.length; p++) {
   437→      const pos = openPositions[p];
   438→      const barsHeld = barIdx - pos.entryBar;
   439→      
   440→      // Partial TP check
   441→      if (!pos.partialTaken && bar.high >= pos.tp1) {
   442→        const closeQty = Math.ceil(pos.totalQty * 0.3);
   443→        const pnl = closeQty * (pos.tp1 - pos.avgPrice);
   444→        capital += pnl + closeQty * pos.avgPrice;
   445→        pos.totalQty -= closeQty;
   446→        pos.trailActive = true;
   447→        pos.partialTaken = true;
   448→      }
   449→
   450→      // Trailing stop
   451→      if (pos.trailActive) {
   452→        const trailLow5 = Math.min(...lows.slice(Math.max(0, barIdx - 5), barIdx + 1));
   453→        const dynamicTrail = Math.max(
   454→          (ema10Full[barIdx] ?? pos.avgPrice) - (atr14Full[barIdx] ?? 0),
   455→          trailLow5
   456→        );
   457→        pos.blendedSL = Math.max(pos.blendedSL, pos.avgPrice, dynamicTrail);
   458→      }
   459→
   460→      // SL hit
   461→      if (bar.low <= pos.blendedSL && pos.totalQty > 0) {
   462→        const pnl = pos.totalQty * (pos.blendedSL - pos.avgPrice);
   463→        capital += pnl + pos.totalQty * pos.avgPrice;
   464→        trades.push({
   465→          symbol,
   466→          entryDate: candles[pos.entryBar].date,
   467→          entryPrice: pos.avgPrice,
   468→          exitDate: bar.date,
   469→          exitPrice: pos.blendedSL,
   470→          qty: pos.totalQty,
   471→          pnl: Math.round(pnl * 100) / 100,
   472→          pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   473→          score: pos.score,
   474→          setupType: pos.setupType,
   475→          exitReason: pos.trailActive ? 'TRAIL_STOP' : 'SL_HIT',
   476→        });
   477→        closedIndices.push(p);
   478→        continue;
   479→      }
   480→
   481→      // Dead capital exit
   482→      if (barsHeld > maxHoldBars && !pos.trailActive && pos.totalQty > 0) {
   483→        const pnl = pos.totalQty * (bar.close - pos.avgPrice);
   484→        capital += pnl + pos.totalQty * pos.avgPrice;
   485→        trades.push({
   486→          symbol,
   487→          entryDate: candles[pos.entryBar].date,
   488→          entryPrice: pos.avgPrice,
   489→          exitDate: bar.date,
   490→          exitPrice: bar.close,
   491→          qty: pos.totalQty,
   492→          pnl: Math.round(pnl * 100) / 100,
   493→          pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   494→          score: pos.score,
   495→          setupType: pos.setupType,
   496→          exitReason: 'DEAD_CAPITAL',
   497→        });
   498→        closedIndices.push(p);
   499→      }
   500→    }
   501→    // Remove closed positions (and fully exited ones)
   502→    openPositions = openPositions.filter((_, idx) => !closedIndices.includes(idx)).filter(p => p.totalQty > 0);
   503→
   504→    // Track equity
   505→    peakEquity = Math.max(peakEquity, capital);
   506→    const dd = ((peakEquity - capital) / peakEquity) * 100;
   507→    maxDD = Math.max(maxDD, dd);
   508→    equityCurve.push({ date: bar.date, equity: Math.round(capital * 100) / 100 });
   509→
   510→    // Check for new entry (simplified - using previous bar's signal)
   511→    if (barIdx < 2 || barIdx - lastTradeBar <= config.cooldownBars) continue;
   512→    if (openPositions.length >= maxOpenTrades) continue;
   513→
   514→    const i = barIdx;
   515→    const i1 = barIdx - 1;
   516→    
   517→    // Quick signal check on previous bar
   518→    const prevSMA200 = sma200Full[i1];
   519→    const prevEMA20 = ema20Full[i1];
   520→    const prevRSI = rsi14Full[i1];
   521→    const prevATR = atr14Full[i1];
   522→    const prevADX = adxFull[i1];
   523→    const prevCandle = candles[i1];
   524→
   525→    if (!prevSMA200 || !prevEMA20 || !prevRSI || !prevATR || !prevADX) continue;
   526→
   527→    // Simplified confluence check (mirroring the screening logic)
   528→    const prevADX1 = adxFull[i1 - 1] ?? 0;
   529→    const trending = prevADX > 18 || (prevADX > prevADX1 && prevADX > 15);
   530→    const atrSma20 = calcSMA(atr14Full.slice(Math.max(0, i - 20), i).filter(v => v != null), 20);
   531→    const validVol = prevATR > atrSma20 * 0.6 && prevATR < atrSma20 * 1.8;
   532→    const notExt = prevCandle.close < prevEMA20 * 1.06;
   533→
   534→    // Nifty check
   535→    let regimeSafe = true;
   536→    if (niftyCandles.length >= 200) {
   537→      const niftyIdx = Math.min(niftyCandles.length - 1, barIdx);
   538→      const nifty200 = nifty200Full[niftyIdx];
   539→      if (nifty200) regimeSafe = niftyCandles[niftyIdx].close > nifty200;
   540→    }
   541→
   542→    if (!trending || !validVol || !notExt || !regimeSafe) continue;
   543→
   544→    // Score calculation on prev bar
   545→    let score = 0;
   546→    if (prevCandle.close > prevSMA200 && prevRSI > 52) score++;
   547→    const low4 = Math.min(...lows.slice(Math.max(0, i1 - 4), i1 + 1));
   548→    if (low4 < prevEMA20 * 1.02 && prevCandle.close > prevEMA20) score++;
   549→    if (prevCandle.close > candles[i1 - 1]?.high) score++;
   550→    const vm = calcSMA(volumes.slice(Math.max(0, i1 - 20), i1 + 1), 20);
   551→    if (prevCandle.volume > vm * 1.05) score++;
   552→    
   553→    const gapPct = Math.abs(prevCandle.open - candles[i1 - 1]?.close) / (candles[i1 - 1]?.close || 1) * 100;
   554→    if (gapPct < 3.5) score++;
   555→
   556→    if (score >= config.minScore) {
   557→      const theoreticalEP = Math.max(bar.open, prevCandle.high);
   558→      const stopLow5 = Math.min(...lows.slice(Math.max(0, i - 5), i));
   559→      const rawSL = stopLow5 - (prevATR * 0.8);
   560→      const riskPerShare = theoreticalEP - rawSL;
   561→
   562→      if (riskPerShare <= 0) continue;
   563→
   564→      const structRange = Math.max(...highs.slice(Math.max(0, i - 60), i)) - Math.min(...lows.slice(Math.max(0, i - 60), i));
   565→      const rr = structRange / riskPerShare;
   566→      if (rr < config.minRR && score < 6) continue;
   567→
   568→      // Size calculation
   569→      const ddPenalty = Math.max(0.25, Math.min(1.0, 1.0 - (maxDD / 20.0)));
   570→      const baseRisk = score === 6 ? riskPct : riskPct * 0.5;
   571→      const appliedRisk = baseRisk * ddPenalty;
   572→      const maxRisk = capital * (appliedRisk / 100);
   573→      
   574→      const qtyRisk = Math.floor(maxRisk / riskPerShare);
   575→      const slotMultiplier = score === 6 ? 1.0 : 0.6;
   576→      const targetAlloc = (capital / maxSlots) * slotMultiplier;
   577→      const qtyCap = Math.floor(targetAlloc / theoreticalEP);
   578→      const qty = Math.max(1, Math.min(qtyRisk, qtyCap));
   579→
   580→      const tp1 = theoreticalEP + (riskPerShare * 1.5);
   581→
   582→      if (capital < qty * theoreticalEP) continue;
   583→
   584→      openPositions.push({
   585→        entryBar: barIdx,
   586→        entryPrice: theoreticalEP,
   587→        qty,
   588→        stopLoss: rawSL,
   589→        tp1,
   590→        score,
   591→        setupType: score === 6 ? 'A+' : 'B',
   592→        partialTaken: false,
   593→        trailActive: false,
   594→        totalQty: qty,
   595→        avgPrice: theoreticalEP,
   596→        blendedSL: rawSL,
   597→      });
   598→
   599→      capital -= qty * theoreticalEP;
   600→      lastTradeBar = barIdx;
   601→    }
   602→  }
   603→
   604→  // Close remaining open positions at last bar price
   605→  for (const pos of openPositions) {
   606→    const lastPrice = candles[candles.length - 1].close;
   607→    const pnl = pos.totalQty * (lastPrice - pos.avgPrice);
   608→    capital += pnl + pos.totalQty * pos.avgPrice;
   609→    trades.push({
   610→      symbol,
   611→      entryDate: candles[pos.entryBar].date,
   612→      entryPrice: pos.avgPrice,
   613→      exitDate: candles[candles.length - 1].date,
   614→      exitPrice: lastPrice,
   615→      qty: pos.totalQty,
   616→      pnl: Math.round(pnl * 100) / 100,
   617→      pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   618→      score: pos.score,
   619→      setupType: pos.setupType,
   620→      exitReason: 'BACKTEST_END',
   621→    });
   622→  }
   623→
   624→  // Calculate stats
   625→  const wins = trades.filter(t => t.pnl > 0);
   626→  const losses = trades.filter(t => t.pnl <= 0);
   627→  const grossProfit = wins.reduce((s, t) => s + t.pnl, 0);
   628→  const grossLoss = Math.abs(losses.reduce((s, t) => s + t.pnl, 0));
   629→
   630→  // Sharpe ratio (simplified)
   631→  const returns = trades.map(t => t.pnlPercent);
   632→  const avgReturn = returns.length > 0 ? returns.reduce((a, b) => a + b, 0) / returns.length : 0;
   633→  const stdReturn = returns.length > 1
   634→    ? Math.sqrt(returns.reduce((s, r) => s + Math.pow(r - avgReturn, 2), 0) / (returns.length - 1))
   635→    : 1;
   636→  const sharpeRatio = stdReturn > 0 ? (avgReturn / stdReturn) * Math.sqrt(252) : 0;
   637→
   638→  return {
   639→    trades,
   640→    equityCurve,
   641→    stats: {
   642→      totalTrades: trades.length,
   643→      winTrades: wins.length,
   644→      lossTrades: losses.length,
   645→      winRate: trades.length > 0 ? Math.round((wins.length / trades.length) * 10000) / 100 : 0,
   646→      profitFactor: grossLoss > 0 ? Math.round((grossProfit / grossLoss) * 100) / 100 : 0,
   647→      maxDrawdown: Math.round(maxDD * 100) / 100,
   648→      avgWin: wins.length > 0 ? Math.round((grossProfit / wins.length) * 100) / 100 : 0,
   649→      avgLoss: losses.length > 0 ? Math.round((grossLoss / losses.length) * 100) / 100 : 0,
   650→      bestTrade: trades.length > 0 ? Math.round(Math.max(...trades.map(t => t.pnl)) * 100) / 100 : 0,
   651→      worstTrade: trades.length > 0 ? Math.round(Math.min(...trades.map(t => t.pnl)) * 100) / 100 : 0,
   652→      finalCapital: Math.round(capital * 100) / 100,
   653→      sharpeRatio: Math.round(sharpeRatio * 100) / 100,
   654→    },
   655→  };
   656→}

   660→function padArray(arr: (number | undefined)[], targetLen: number): (number | null)[] {
   661→  const padding = targetLen - arr.length;
   662→  const padded: (number | null)[] = new Array(padding).fill(null);
   663→  return [...padded, ...arr.map(v => v ?? null)];
   664→}
   665→
   666→function calcSMA(values: (number | null | undefined)[], period: number): number {
   667→  const valid = values.filter(v => v != null) as number[];
   668→  const slice = valid.slice(-period);
   669→  if (slice.length === 0) return 0;
   670→  return slice.reduce((a, b) => a + b, 0) / slice.length;
   671→}
