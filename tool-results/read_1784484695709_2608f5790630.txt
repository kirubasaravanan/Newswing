     4→import { Button } from '@/components/ui/button';
     5→import { Card, CardContent } from '@/components/ui/card';
     6→import { Input } from '@/components/ui/input';
     7→import { Label } from '@/components/ui/label';
     8→import { Badge } from '@/components/ui/badge';
     9→import { ScrollArea } from '@/components/ui/scroll-area';
    10→import { Switch } from '@/components/ui/switch';
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs';
    12→import {
    13→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogClose,
    14→} from '@/components/ui/dialog';
    15→import {
    16→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    17→} from '@/components/ui/select';
    18→import {
    19→  Bot, Play, Wallet, Settings, TrendingUp, TrendingDown, Clock,

    29→
    30→// ── Types ───────────────────────────────────────────────
    31→interface WalletData {
    32→  id: string; totalCapital: number; deployed: number; available: number;
    33→  realizedPnl: number; unrealizedPnl: number;
    34→}
    35→
    36→interface PositionRules {
    37→  maxPerStock: number; maxBuysPerMonth: number; maxHoldingDays: number;
    38→  maxTotalPositions: number; riskPerTradePct: number;
    39→  trailingStopR: number; trailToR: number; partialBookR: number; partialBookPct: number;
    40→  cooldownDays: number; maxSectorPct: number; timeExitMins: number;
    41→  // v2
    42→  maxDrawdownPct: number; dailyLossLimit: number; niftyRegimeFilter: boolean;
    43→  atrTrailMultiplier: number; adaptiveSizing: boolean; streakPenaltyPct: number;
    44→}
    45→
    46→interface SchedulerState {
    47→  enabled: boolean; scanIntervalMin: number; exitIntervalMin: number;
    48→  lastScanAt: string | null; lastExitAt: string | null;
    49→  todayEntries: number; todayExits: number; todayPnl: number; scanCount: number;
    50→  // v2
    51→  circuitBreaker: boolean; circuitBreakerReason: string;
    52→  niftyRegime: string; consecutiveLosses: number; lastAdaptiveFactor: number;
    53→}
    54→
    55→interface AutoTradeLog {
    56→  id: string; action: string; symbol: string; executed: boolean;
    57→  reason: string; createdAt: string; signal?: string;
    58→}
    59→
    60→interface OpenPosition {
    61→  id: string; symbol: string; stockName: string | null;
    62→  entryPrice: number; qty: number; stopLoss: number; targetPrice: number;
    63→  entryDate: string; autoTraded: boolean; tags?: string;
    64→}
    65→
    66→interface DrawdownData {
    67→  drawdownPct: number; peakCapital: number; currentCapital: number;
    68→}
    69→
    70→export function AutoTradeTab() {
    71→  const { config } = useTradeStore();
    72→  const [wallet, setWallet] = useState<WalletData | null>(null);
    73→  const [rules, setRules] = useState<PositionRules>({
    74→    maxPerStock: 50000, maxBuysPerMonth: 3, maxHoldingDays: 25,
    75→    maxTotalPositions: 8, riskPerTradePct: 1.0,
    76→    trailingStopR: 1.5, trailToR: 0.5, partialBookR: 2.0, partialBookPct: 30,
    77→    cooldownDays: 3, maxSectorPct: 35, timeExitMins: 30,
    78→    maxDrawdownPct: 8, dailyLossLimit: 5000, niftyRegimeFilter: true,
    79→    atrTrailMultiplier: 2.0, adaptiveSizing: true, streakPenaltyPct: 20,
    80→  });
    81→  const [openPositions, setOpenPositions] = useState<OpenPosition[]>([]);
    82→  const [logs, setLogs] = useState<AutoTradeLog[]>([]);
    83→  const [scheduler, setScheduler] = useState<SchedulerState | null>(null);
    84→  const [marketHours, setMarketHours] = useState(false);
    85→  const [timeToClose, setTimeToClose] = useState(0);
    86→  const [scanning, setScanning] = useState(false);
    87→  const [checkingExits, setCheckingExits] = useState(false);
    88→  const [editingWallet, setEditingWallet] = useState(false);
    89→  const [walletInput, setWalletInput] = useState('200000');
    90→  const [editingRules, setEditingRules] = useState(false);
    91→  const [ruleEdits, setRuleEdits] = useState<PositionRules & { scanIntervalMin?: number; exitIntervalMin?: number }>({ ...rules });
    92→  const [chartSymbol, setChartSymbol] = useState<string | null>(null);
    93→  const [exitWarnings, setExitWarnings] = useState<any[]>([]);
    94→  const [selectedLog, setSelectedLog] = useState<AutoTradeLog | null>(null);
    95→  const [showAllLogs, setShowAllLogs] = useState(false);
    96→  const [drawdown, setDrawdown] = useState<DrawdownData | null>(null);
    97→
    98→  // ── Data Fetching ─────────────────────────────────────
    99→  const fetchStatus = useCallback(async () => {
   100→    try {
   101→      const res = await fetch('/api/auto-trade');
   102→      const data = await res.json();
   103→      if (data.success) {
   104→        setWallet(data.wallet);
   105→        setRules(data.rules);
   106→        setOpenPositions(data.openTrades);
   107→        setLogs(data.recentLogs || []);
   108→        setScheduler(data.scheduler);
   109→        setMarketHours(data.marketHours);
   110→        setTimeToClose(data.timeToClose);
   111→        if (data.drawdown) setDrawdown(data.drawdown);
   112→      }
   113→    } catch (err) { console.error('Fetch status error:', err); }
   114→  }, []);
   115→
   116→  useEffect(() => {
   117→    fetchStatus();
   118→    const interval = setInterval(fetchStatus, 30000);
   119→    return () => clearInterval(interval);
   120→  }, [fetchStatus]);
   121→
   122→  // ── Scheduler Tick ─────────────────────────────────────
   123→  const schedulerTickRef = useRef<NodeJS.Timeout | null>(null);
   124→  useEffect(() => {
   125→    if (scheduler?.enabled && marketHours && !scheduler?.circuitBreaker) {
   126→      schedulerTickRef.current = setInterval(async () => {
   127→        try {
   128→          await fetch('/api/auto-trade', {
   129→            method: 'POST', headers: { 'Content-Type': 'application/json' },
   130→            body: JSON.stringify({ action: 'scheduler_tick' }),
   131→          });
   132→          fetchStatus();
   133→        } catch { /* ignore */ }
   134→      }, 60000);
   135→    }
   136→    return () => { if (schedulerTickRef.current) clearInterval(schedulerTickRef.current); };
   137→  }, [scheduler?.enabled, marketHours, scheduler?.circuitBreaker, fetchStatus]);
   138→
   139→  // ── Actions ───────────────────────────────────────────
   140→  const handleScanAndTrade = async () => {
   141→    setScanning(true);
   142→    try {
   143→      const res = await fetch('/api/auto-trade', {
   144→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   145→        body: JSON.stringify({ action: 'scan_and_trade', config }),
   146→      });
   147→      const data = await res.json();
   148→      if (data.success) {
   149→        const extra = data.circuitBreaker ? ' [CIRCUIT BREAKER]' :
   150→          data.niftyRegime === 'BEARISH' ? ' [BEARISH REGIME]' : '';
   151→        toast.success(`Scan Complete${extra}`, {
   152→          description: `${data.totalScanned} scanned | ${data.l1Passed} L1 | ${data.l2Signals} entries | ${data.skipped.length} skipped`,
   153→        });
   154→        if (data.adaptiveFactor && data.adaptiveFactor < 1) {
   155→          toast.info('Adaptive Sizing Active', {
   156→            description: `Factor: ${(data.adaptiveFactor * 100).toFixed(0)}% (${data.consecutiveLosses} consecutive losses)`,
   157→          });
   158→        }
   159→        fetchStatus();
   160→      } else {
   161→        toast.error('Auto-trade failed', { description: data.error });
   162→      }
   163→    } catch { toast.error('Auto-trade failed'); }
   164→    finally { setScanning(false); }
   165→  };
   166→
   167→  const handleCheckExits = async () => {
   168→    setCheckingExits(true);
   169→    try {
   170→      const res = await fetch('/api/auto-trade', {
   171→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({ action: 'check_exits' }),
   173→      });
   174→      const data = await res.json();
   175→      if (data.success) {
   176→        if (data.exits.length > 0) {
   177→          toast.success(`Exit Check: ${data.exits.length} closed`, {
   178→            description: data.exits.map((e: any) => `${e.symbol} (${e.exitReason})`).join(', '),
   179→          });
   180→        }
   181→        if (data.partialBooks?.length > 0) {
   182→          toast.info(`Partial Book: ${data.partialBooks.length} positions`, {
   183→            description: data.partialBooks.map((p: any) => `${p.symbol}: ${p.bookedQty}qty at ${p.rMultiple}R`).join(', '),
   184→          });
   185→        }
   186→        if (data.exits.length === 0 && (!data.partialBooks || data.partialBooks.length === 0)) {
   187→          toast.info('No exits triggered', { description: `${data.holding?.length || 0} positions holding` });
   188→        }
   189→        if (data.warnings?.length > 0) setExitWarnings(data.warnings);
   190→        fetchStatus();
   191→      }
   192→    } catch { toast.error('Exit check failed'); }
   193→    finally { setCheckingExits(false); }
   194→  };
   195→
   196→  const handleSchedulerToggle = async (enabled: boolean) => {
   197→    try {
   198→      const res = await fetch('/api/auto-trade', {
   199→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   200→        body: JSON.stringify({
   201→          action: 'scheduler_toggle', enabled,
   202→          scanIntervalMin: ruleEdits.scanIntervalMin || 30,
   203→          exitIntervalMin: ruleEdits.exitIntervalMin || 5,
   204→        }),
   205→      });
   206→      const data = await res.json();
   207→      if (data.success) {
   208→        setScheduler(data.scheduler);
   209→        toast.success(enabled ? 'Auto-Mode Armed' : 'Auto-Mode Disarmed');
   210→        fetchStatus();
   211→      }
   212→    } catch { toast.error('Scheduler toggle failed'); }
   213→  };
   214→
   215→  const handleResetCircuitBreaker = async () => {
   216→    try {
   217→      const res = await fetch('/api/auto-trade', {
   218→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({ action: 'reset_circuit_breaker' }),
   220→      });
   221→      const data = await res.json();
   222→      if (data.success) {
   223→        setScheduler(data.scheduler);
   224→        toast.success('Circuit Breaker Reset');
   225→      }
   226→    } catch { toast.error('Reset failed'); }
   227→  };
   228→
   229→  const saveWallet = async () => {
   230→    try {
   231→      const res = await fetch('/api/auto-trade', {
   232→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify({ action: 'update_wallet', totalCapital: parseFloat(walletInput) }),
   234→      });
   235→      const data = await res.json();
   236→      if (data.success) { setWallet(data.wallet); setEditingWallet(false); toast.success('Wallet updated'); }
   237→    } catch { toast.error('Failed to update wallet'); }
   238→  };
   239→
   240→  const saveRules = async () => {
   241→    try {
   242→      // Separate scheduler settings from rules
   243→      const { scanIntervalMin, exitIntervalMin, ...rulesOnly } = ruleEdits;
   244→      const res = await fetch('/api/auto-trade', {
   245→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   246→        body: JSON.stringify({ action: 'update_rules', rules: rulesOnly }),
   247→      });
   248→      const data = await res.json();
   249→      if (data.success) { setRules(data.rules); setEditingRules(false); toast.success('Rules updated'); }
   250→    } catch { toast.error('Failed to update rules'); }
   251→  };
   252→
   253→  const totalPnl = (wallet?.realizedPnl || 0) + (wallet?.unrealizedPnl || 0);
   254→  const pnlPct = wallet?.initialCapital ? (totalPnl / wallet.initialCapital) * 100 : 0;
   255→
   256→  return (
   257→    <div className="space-y-4">
   258→      {/* ── Top Bar ─────────────────────────────────────── */}
   259→      <div className="flex items-center justify-between flex-wrap gap-3">
   260→        <div className="flex items-center gap-3">
   261→          <h2 className="text-lg font-bold flex items-center gap-2">
   262→            <Bot className="h-5 w-5 text-primary" />
   263→            Auto-Trade Engine
   264→            <Badge variant="outline" className="text-[9px] h-4 text-muted-foreground">v2</Badge>
   265→          </h2>
   266→          {/* Market Status */}
   267→          <div className={cn(
   268→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   269→            marketHours ? 'bg-emerald-500/10 text-emerald-400' : 'bg-muted text-muted-foreground'
   270→          )}>
   271→            <div className={cn('h-1.5 w-1.5 rounded-full', marketHours ? 'bg-emerald-400 animate-pulse' : 'bg-muted-foreground')} />
   272→            {marketHours ? `${timeToClose}m to close` : 'Market Closed'}
   273→          </div>
   274→          {/* Circuit Breaker */}
   275→          {scheduler?.circuitBreaker && (
   276→            <div className="flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs bg-red-500/15 text-red-400 border border-red-500/20">
   277→              <ShieldAlert className="h-3 w-3" />
   278→              CIRCUIT BREAKER
   279→              <Button variant="ghost" size="sm" onClick={handleResetCircuitBreaker} className="h-4 w-4 p-0 ml-1 text-red-400 hover:text-red-300">
   280→                <RotateCcw className="h-2.5 w-2.5" />
   281→              </Button>
   282→            </div>
   283→          )}
   284→          {/* Nifty Regime */}
   285→          <div className={cn(
   286→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   287→            scheduler?.niftyRegime === 'BULLISH' ? 'bg-emerald-500/10 text-emerald-400' :
   288→            scheduler?.niftyRegime === 'BEARISH' ? 'bg-red-500/10 text-red-400' : 'bg-muted text-muted-foreground'
   289→          )}>
   290→            <TrendingUp className={cn('h-3 w-3', scheduler?.niftyRegime === 'BEARISH' && 'rotate-180')} />
   291→            {scheduler?.niftyRegime || 'UNKNOWN'}
   292→          </div>
   293→          {/* Auto-Mode */}
   294→          <div className={cn(
   295→            'flex items-center gap-1.5 rounded-full px-2.5 py-1 text-xs',
   296→            scheduler?.enabled ? 'bg-indigo-500/10 text-indigo-400' : 'bg-muted text-muted-foreground'
   297→          )}>
   298→            <Shield className="h-3 w-3" />
   299→            {scheduler?.enabled ? 'Armed' : 'Manual'}
   300→          </div>
   301→        </div>
   302→        <div className="flex items-center gap-2">
   303→          <Button variant="outline" size="sm" onClick={handleCheckExits}
   304→            disabled={checkingExits || openPositions.length === 0}
   305→            className="h-8 gap-1.5 text-xs">
   306→            {checkingExits ? <Loader2 className="h-3 w-3 animate-spin" /> : <ArrowRightLeft className="h-3 w-3" />}
   307→            Check Exits
   308→          </Button>
   309→          <Button onClick={handleScanAndTrade} disabled={scanning || !marketHours || !!scheduler?.circuitBreaker}
   310→            className={cn('gap-1.5', scheduler?.enabled ? 'bg-emerald-600 hover:bg-emerald-700' : '')}
   311→            size="sm">
   312→            {scanning ? <Loader2 className="h-3 w-3 animate-spin" /> : <Zap className="h-3 w-3" />}
   313→            {scanning ? 'Scanning...' : 'Scan & Trade'}
   314→          </Button>
   315→        </div>
   316→      </div>
   317→
   318→      <Tabs defaultValue="engine" className="space-y-4">
   319→        <TabsList className="bg-secondary/50">
   320→          <TabsTrigger value="engine" className="text-xs gap-1.5"><Bot className="h-3 w-3" /> Engine</TabsTrigger>
   321→          <TabsTrigger value="scheduler" className="text-xs gap-1.5"><Timer className="h-3 w-3" /> Scheduler</TabsTrigger>
   322→          <TabsTrigger value="positions" className="text-xs gap-1.5"><TrendingUp className="h-3 w-3" /> Positions</TabsTrigger>
   323→          <TabsTrigger value="audit" className="text-xs gap-1.5"><FileText className="h-3 w-3" /> Audit</TabsTrigger>
   324→          <TabsTrigger value="rules" className="text-xs gap-1.5"><Settings className="h-3 w-3" /> Rules</TabsTrigger>
   325→        </TabsList>
   326→
   327→        {/* ═══ ENGINE TAB ═════════════════════════════════ */}
   328→        <TabsContent value="engine" className="space-y-4">
   329→          <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   330→            {/* Wallet Card */}
   331→            <Card className="lg:col-span-2 border-border">
   332→              <CardContent className="p-4">
   333→                <div className="flex items-center justify-between mb-4">
   334→                  <h3 className="text-sm font-semibold flex items-center gap-2">
   335→                    <Wallet className="h-4 w-4 text-primary" /> Capital Wallet
   336→                  </h3>
   337→                  {!editingWallet ? (
   338→                    <Button variant="ghost" size="sm" onClick={() => { setWalletInput(String(wallet?.totalCapital || 200000)); setEditingWallet(true); }} className="h-7 text-xs">Edit</Button>
   339→                  ) : (
   340→                    <div className="flex items-center gap-2">
   341→                      <Input value={walletInput} onChange={e => setWalletInput(e.target.value)} className="h-7 w-32 text-xs" />
   342→                      <Button size="sm" onClick={saveWallet} className="h-7 text-xs">Save</Button>
   343→                      <Button size="sm" variant="ghost" onClick={() => setEditingWallet(false)} className="h-7 text-xs">Cancel</Button>
   344→                    </div>
   345→                  )}
   346→                </div>
   347→                {wallet && (
   348→                  <div className="grid grid-cols-2 md:grid-cols-5 gap-3">
   349→                    <div className="rounded-lg bg-secondary/50 p-3">
   350→                      <div className="text-[10px] text-muted-foreground uppercase">Total Capital</div>
   351→                      <div className="text-lg font-bold font-mono mt-0.5">₹{wallet.totalCapital.toLocaleString()}</div>
   352→                    </div>
   353→                    <div className="rounded-lg bg-blue-500/10 p-3">
   354→                      <div className="text-[10px] text-blue-400 uppercase">Deployed</div>
   355→                      <div className="text-lg font-bold font-mono mt-0.5 text-blue-400">₹{wallet.deployed.toLocaleString()}</div>
   356→                      <div className="text-[10px] text-muted-foreground">{wallet.totalCapital > 0 ? ((wallet.deployed / wallet.totalCapital) * 100).toFixed(1) : 0}%</div>
   357→                    </div>
   358→                    <div className="rounded-lg bg-emerald-500/10 p-3">
   359→                      <div className="text-[10px] text-emerald-400 uppercase">Available</div>
   360→                      <div className="text-lg font-bold font-mono mt-0.5 text-emerald-400">₹{wallet.available.toLocaleString()}</div>
   361→                    </div>
   362→                    <div className={cn('rounded-lg p-3', totalPnl >= 0 ? 'bg-emerald-500/10' : 'bg-red-500/10')}>
   363→                      <div className={cn('text-[10px] uppercase', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>Total P&L</div>
   364→                      <div className={cn('text-lg font-bold font-mono mt-0.5', totalPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   365→                        {totalPnl >= 0 ? '+' : ''}₹{totalPnl.toLocaleString()}
   366→                      </div>
   367→                      <div className="text-[10px] text-muted-foreground">R: {wallet.realizedPnl.toLocaleString()} | U: {wallet.unrealizedPnl.toLocaleString()}</div>
   368→                    </div>
   369→                    {/* v2: Drawdown Card */}
   370→                    <div className={cn('rounded-lg p-3', (drawdown?.drawdownPct || 0) >= rules.maxDrawdownPct * 0.7 ? 'bg-red-500/10' : 'bg-orange-500/10')}>
   371→                      <div className="text-[10px] text-orange-400 uppercase flex items-center gap-1">
   372→                        <Skull className="h-3 w-3" /> Drawdown
   373→                      </div>
   374→                      <div className={cn('text-lg font-bold font-mono mt-0.5',
   375→                        (drawdown?.drawdownPct || 0) >= rules.maxDrawdownPct ? 'text-red-400' : 'text-orange-400'
   376→                      )}>
   377→                        {(drawdown?.drawdownPct || 0).toFixed(2)}%
   378→                      </div>
   379→                      <div className="text-[10px] text-muted-foreground">Limit: {rules.maxDrawdownPct}%</div>
   380→                    </div>
   381→                  </div>
   382→                )}
   383→              </CardContent>
   384→            </Card>
   385→
   386→            {/* Today's Summary + Engine State */}
   387→            <div className="space-y-4">
   388→              <Card className="border-border">
   389→                <CardContent className="p-4">
   390→                  <h3 className="text-sm font-semibold flex items-center gap-2 mb-3">
   391→                    <Activity className="h-4 w-4 text-primary" /> Today
   392→                  </h3>
   393→                  {scheduler && (
   394→                    <div className="space-y-2">
   395→                      <div className="flex justify-between text-xs">
   396→                        <span className="text-muted-foreground">Scans</span>
   397→                        <span className="font-mono font-bold">{scheduler.scanCount}</span>
   398→                      </div>
   399→                      <div className="flex justify-between text-xs">
   400→                        <span className="text-muted-foreground">Entries</span>
   401→                        <span className="font-mono font-bold text-emerald-400">{scheduler.todayEntries}</span>
   402→                      </div>
   403→                      <div className="flex justify-between text-xs">
   404→                        <span className="text-muted-foreground">Exits</span>
   405→                        <span className="font-mono font-bold text-red-400">{scheduler.todayExits}</span>
   406→                      </div>
   407→                      <div className="flex justify-between text-xs pt-2 border-t border-border/50">
   408→                        <span className="text-muted-foreground">Today P&L</span>
   409→                        <span className={cn('font-mono font-bold', scheduler.todayPnl >= 0 ? 'text-emerald-400' : 'text-red-400')}>
   410→                          {scheduler.todayPnl >= 0 ? '+' : ''}₹{Math.round(scheduler.todayPnl).toLocaleString()}
   411→                        </span>
   412→                      </div>
   413→                      <div className="flex justify-between text-xs">
   414→                        <span className="text-muted-foreground">Daily Limit</span>
   415→                        <span className={cn('font-mono',
   416→                          scheduler.todayPnl < 0 && Math.abs(scheduler.todayPnl) >= rules.dailyLossLimit * 0.8 ? 'text-red-400 font-bold' : 'text-muted-foreground'
   417→                        )}>
   418→                          ₹{rules.dailyLossLimit.toLocaleString()}
   419→                        </span>
   420→                      </div>
   421→                    </div>
