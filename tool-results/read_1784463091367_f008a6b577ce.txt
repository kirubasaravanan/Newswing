   157→const safeJson = async (res: Response | null) => {
   158→  if (!res) return null;
   159→  try { return await res.json(); } catch { return null; }
   160→};
   161→
   162→const fetchSafe = async (url: string) => {
   163→  const res = await fetch(url).catch(() => null);
   164→  return res ? await safeJson(res) : null;
   165→};
   166→
   167→const fmt = (n: number | null | undefined, decimals = 2) => {
   168→  if (n == null) return '—';
   169→  return n.toFixed(decimals);
   170→};
   171→
   172→const fmtRupee = (n: number | null | undefined) => {
   173→  if (n == null) return '—';
   174→  const abs = Math.abs(n);
   175→  if (abs >= 100000) return `₹${(n / 100000).toFixed(2)}L`;
   176→  if (abs >= 1000) return `₹${(n / 1000).toFixed(1)}K`;
   177→  return `₹${n.toFixed(0)}`;
   178→};

   760→                    const tpDist = Math.abs(sltp.takeProfit - entryP);

   772→                  <span className="text-foreground font-medium">Entry ({entryPremium || '—'})</span>
   773→                </div>
   774→              </div>
   775→            </CardContent>
   776→          </Card>
   777→        )}
   778→      </div>
   779→    </div>
