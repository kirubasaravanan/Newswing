   139→   139→function AlertsPanel() {
   140→   140→  const [alerts, setAlerts] = useState<any[]>([]);
   141→   141→  const [open, setOpen] = useState(false);
   142→   142→  const [form, setForm] = useState({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   143→   143→  const [checking, setChecking] = useState(false);
   144→   144→
   145→   145→  const fetchAlerts = useCallback(async () => {
   146→   146→    try {
   147→   147→      const res = await fetch('/api/portfolio/alerts');
   148→   148→      const data = await res.json();
   149→   149→      if (data.success) setAlerts(data.alerts);
   150→   150→    } catch { /* ignore */ }
   151→   151→  }, []);
   152→   152→
   153→   153→  useEffect(() => { fetchAlerts(); }, [fetchAlerts]);
   154→   154→
   155→   155→  const createAlert = async () => {
   156→   156→    if (!form.symbol || !form.targetPrice) return;
   157→   157→    try {
   158→   158→      const res = await fetch('/api/portfolio/alerts', {
   159→   159→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   160→   160→        body: JSON.stringify(form),
   161→   161→      });
   162→   162→      const data = await res.json();
   163→   163→      if (data.success) {
   164→   164→        toast.success(`Alert set: ${form.symbol} ${form.condition} ₹${form.targetPrice}`);
   165→   165→        setForm({ symbol: '', condition: 'ABOVE', targetPrice: '', notes: '' });
   166→   166→        setOpen(false);
   167→   167→        fetchAlerts();
   168→   168→      }
   169→   169→    } catch { toast.error('Failed to create alert'); }
   170→   170→  };
   171→   171→
   172→   172→  const deleteAlert = async (id: string) => {
   173→   173→    await fetch(`/api/portfolio/alerts?id=${id}`, { method: 'DELETE' });
   174→   174→    fetchAlerts();
   175→   175→  };
   176→   176→
   177→   177→  const checkAlerts = async () => {
   178→   178→    setChecking(true);
   179→   179→    try {
   180→   180→      const res = await fetch('/api/portfolio/alerts', { method: 'PUT' });
   181→   181→      const data = await res.json();
   182→   182→      if (data.success && data.triggered.length > 0) {
   183→   183→        for (const t of data.triggered) {
   184→   184→          toast.success(`🔔 ${t.symbol} hit ₹${t.triggeredPrice}!`, {
   185→   185→            description: `Target: ₹${t.targetPrice} ${t.condition}`,
   186→   186→            duration: 8000,
   187→   187→          });
   188→   188→        }
   189→   189→        fetchAlerts();
   190→   190→      } else {
   191→   191→        toast.info('No alerts triggered');
   192→   192→      }
   193→   193→    } catch { toast.error('Failed to check alerts'); }
   194→   194→    finally { setChecking(false); }
   195→   195→  };
   196→   196→
   197→   197→  return (
   198→   198→    <div>
   199→   199→      <div className="flex items-center justify-between mb-3">
   200→   200→        <h3 className="text-sm font-semibold flex items-center gap-2"><Bell className="h-4 w-4" /> Price Alerts</h3>
   201→   201→        <div className="flex gap-2">
   202→   202→          <Button variant="outline" size="sm" onClick={checkAlerts} disabled={checking} className="h-7 text-xs gap-1">
   203→   203→            {checking ? <Loader2 className="h-3 w-3 animate-spin" /> : <Target className="h-3 w-3" />} Check
   204→   204→          </Button>
   205→   205→          <Dialog open={open} onOpenChange={setOpen}>
   206→   206→            <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> New Alert</Button></DialogTrigger>
   207→   207→            <DialogContent className="sm:max-w-sm">
   208→   208→              <DialogHeader><DialogTitle>Create Price Alert</DialogTitle></DialogHeader>
   209→   209→              <div className="grid gap-3 py-2">
   210→   210→                <div><Label className="text-xs">Symbol</Label>
   211→   211→                  <Input value={form.symbol} onChange={e => setForm({ ...form, symbol: e.target.value.toUpperCase() })} placeholder="RELIANCE" className="h-8 text-xs" /></div>
   212→   212→                <div><Label className="text-xs">Condition</Label>
   213→   213→                  <Select value={form.condition} onValueChange={v => setForm({ ...form, condition: v })}>
   214→   214→                    <SelectTrigger className="h-8 text-xs"><SelectValue /></SelectTrigger>
   215→   215→                    <SelectContent><SelectItem value="ABOVE">Price goes ABOVE</SelectItem><SelectItem value="BELOW">Price goes BELOW</SelectItem></SelectContent>
   216→   216→                  </Select></div>
   217→   217→                <div><Label className="text-xs">Target Price (₹)</Label>
   218→   218→                  <Input type="number" value={form.targetPrice} onChange={e => setForm({ ...form, targetPrice: e.target.value })} placeholder="2500" className="h-8 text-xs" /></div>
   219→   219→              </div>
   220→   220→              <DialogFooter>
   221→   221→                <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
   222→   222→                <Button size="sm" onClick={createAlert}>Create</Button>
   223→   223→              </DialogFooter>
   224→   224→            </DialogContent>
   225→   225→          </Dialog>
   226→   226→        </div>
   227→   227→      </div>
   228→   228→      {alerts.length === 0 ? (
   229→   229→        <p className="text-xs text-muted-foreground text-center py-4">No alerts set. Create one to get notified when a price target is hit.</p>
   230→   230→      ) : (
   231→   231→        <ScrollArea className="max-h-[200px]">
   232→   232→          <div className="space-y-1.5 pr-2">
   233→   233→            {alerts.slice(0, 20).map(a => (
   234→   234→              <div key={a.id} className="flex items-center gap-2 text-xs py-1.5 border-b border-border/30">
   235→   235→                <div className={cn('h-2 w-2 rounded-full shrink-0', a.triggered ? 'bg-amber-400' : 'bg-blue-400')} />
   236→   236→                <span className="font-semibold">{a.symbol}</span>
   237→   237→                <Badge variant="outline" className="text-[9px] h-4">{a.condition}</Badge>
   238→   238→                <span className="font-mono">₹{a.targetPrice.toLocaleString()}</span>
   239→   239→                {a.triggered && <Badge className="text-[9px] h-4 bg-amber-500/20 text-amber-400">HIT ₹{a.triggeredPrice}</Badge>}
   240→   240→                <button onClick={() => deleteAlert(a.id)} className="ml-auto text-muted-foreground hover:text-red-400"><X className="h-3 w-3" /></button>
   241→   241→              </div>
   242→   242→            ))}
   243→   243→          </div>
   244→   244→        </ScrollArea>
   245→   245→      )}
   246→   246→    </div>
   247→   247→  );
   248→   248→}

   695→   695→
   696→   696→  const fetchAnalytics = useCallback(async () => {
   697→   697→    setLoading(true);
   698→   698→    try {
   699→   699→      const res = await fetch('/api/trades');
   700→   700→      const json = await res.json();
   701→   701→      const allTrades = json.trades || [];
   702→   702→      const closed = allTrades.filter((t: any) => t.status === 'CLOSED' && t.pnl != null);
   703→   703→
   704→   704→      const wins = closed.filter((t: any) => t.pnl > 0);
   705→   705→      const totalPnL = closed.reduce((s: number, t: any) => s + t.pnl, 0);
   706→   706→      const grossProfit = wins.reduce((s: number, t: any) => s + t.pnl, 0);
   707→   707→      const grossLoss = Math.abs(closed.filter((t: any) => t.pnl <= 0).reduce((s: number, t: any) => s + t.pnl, 0));
   708→   708→
   709→   709→      const monthMap = new Map<string, { pnl: number; trades: number }>();
   710→   710→      for (const t of closed) {
   711→   711→        const d = new Date(t.exitDate);
   712→   712→        const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   713→   713→        const prev = monthMap.get(key) || { pnl: 0, trades: 0 };
   714→   714→        prev.pnl += t.pnl;
   715→   715→        prev.trades += 1;
   716→   716→        monthMap.set(key, prev);
   717→   717→      }
   718→   718→      const byMonth = Array.from(monthMap.entries()).map(([month, v]) => ({
   719→   719→        month, pnl: Math.round(v.pnl), trades: v.trades,
   720→   720→      })).sort((a, b) => a.month.localeCompare(b.month));
   721→   721→
   722→   722→      const buckets = [
   723→   723→        { range: '< -5%', min: -Infinity, max: -5 }, { range: '-5% to -2%', min: -5, max: -2 },
   724→   724→        { range: '-2% to 0%', min: -2, max: 0 }, { range: '0% to 2%', min: 0, max: 2 },
   725→   725→        { range: '2% to 5%', min: 2, max: 5 }, { range: '> 5%', min: 5, max: Infinity },
   726→   726→      ];
   727→   727→      const pnlDistribution = buckets.map(b => ({
   728→   728→        range: b.range,
   729→   729→        count: closed.filter((t: any) => t.pnlPercent >= b.min && t.pnlPercent < b.max).length,
   730→   730→      }));
   731→   731→
   732→   732→      let peak = 0, running = 0, maxDD = 0;
   733→   733→      for (const t of closed.sort((a: any, b: any) => new Date(a.exitDate).getTime() - new Date(b.exitDate).getTime())) {
   734→   734→        running += t.pnl;
   735→   735→        peak = Math.max(peak, running);
   736→   736→        maxDD = Math.max(maxDD, (peak - running) / peak * 100);
   737→   737→      }

   739→   739→      setAnalyticsData({
   740→   740→        trades: closed, totalPnL,
   741→   741→        winRate: closed.length > 0 ? (wins.length / closed.length) * 100 : 0,

   743→   743→        avgLoss: closed.length - wins.length > 0 ? grossLoss / (closed.length - wins.length) : 0,
   744→   744→        profitFactor: grossLoss > 0 ? grossProfit / grossLoss : 0,
   745→   745→        maxDD, byMonth, pnlDistribution,
   746→   746→      });
   747→   747→    } catch { /* ignore */ }
   748→   748→    finally { setLoading(false); }
   749→   749→  }, []);

   761→   761→        <Button variant="outline" size="sm" onClick={fetchAnalytics} className="gap-2 h-8 text-xs"><RefreshCw className="h-3.5 w-3.5" /> Refresh</Button>
   762→   762→      </div>
   763→   763→
   764→   764→      <Tabs defaultValue="overview" className="w-full">
