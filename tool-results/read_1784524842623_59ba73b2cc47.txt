   106→   106→function get<T>(arr: T[], index: number, fallback: T): T {
   107→   107→  if (index < 0 || index >= arr.length) return fallback;
   108→   108→  return arr[index];
   109→   109→}

   111→   111→// ── Core Engine ────────────────────────────────────────────
   112→   112→
   113→   113→export function runScreening(
   114→   114→  symbol: string,
   115→   115→  candles: OHLCV[],
   116→   116→  niftyCandles: OHLCV[],
   117→   117→  config: ScreeningConfig = DEFAULT_CONFIG
   118→   118→): ScreeningResult | null {
   119→   119→  const len = candles.length;
   120→   120→  if (len < 250) return null; // Need enough data for SMA 200
   121→   121→
   122→   122→  const closes = candles.map(c => c.close);
   123→   123→  const highs = candles.map(c => c.high);
   124→   124→  const lows = candles.map(c => c.low);
   125→   125→  const volumes = candles.map(c => c.volume);
   126→   126→
   127→   127→  // ── Indicators ─────────────────────────────────────────
   128→   128→  const sma200Arr = SMA.calculate({ period: 200, values: closes });
   129→   129→  const ema20Arr = EMA.calculate({ period: 20, values: closes });
   130→   130→  const ema10Arr = EMA.calculate({ period: 10, values: closes });
   131→   131→  const rsi14Arr = RSI.calculate({ period: 14, values: closes });
   132→   132→  const atr14Arr = ATR.calculate({ period: 14, high: highs, low: lows, close: closes });
   133→   133→  const dmiResult = ADX.calculate({ period: 14, high: highs, low: lows, close: closes });
   134→   134→
   135→   135→  // Pad arrays to match candle length
   136→   136→  const sma200 = padArray(sma200Arr, len);
   137→   137→  const ema20 = padArray(ema20Arr, len);
   138→   138→  const ema10 = padArray(ema10Arr, len);
   139→   139→  const rsi14 = padArray(rsi14Arr, len);
   140→   140→  const atr14 = padArray(atr14Arr, len);
   141→   141→  const adxArr = dmiResult.map(d => d.adx);
   142→   142→  const diPlusArr = dmiResult.map(d => d.pdi);
   143→   143→  const diMinusArr = dmiResult.map(d => d.mdi);
   144→   144→  const adx = padArray(adxArr, len);
   145→   145→  const diPlus = padArray(diPlusArr, len);
   146→   146→  const diMinus = padArray(diMinusArr, len);
   147→   147→
   148→   148→  // ── Current bar (last bar) ─────────────────────────────
   149→   149→  const i = len - 1; // current bar index
   150→   150→  const i1 = len - 2; // previous bar
   151→   151→
   152→   152→  // Need at least 2 bars of valid indicator data
   153→   153→  if (!sma200[i] || !ema20[i] || !atr14[i] || !rsi14[i] || !adx[i]) return null;
   154→   154→
   155→   155→  const currentClose = closes[i];
   156→   156→  const currentOpen = candles[i].open;
   157→   157→  const currentHigh = highs[i];
   158→   158→  const currentLow = lows[i];
   159→   159→  const currentVolume = volumes[i];
   160→   160→  const prevHigh = get(highs, i1, currentHigh);
   161→   161→
   162→   162→  // ── Adaptive range (ATR-based lookback) ────────────────
   163→   163→  const atrSma20 = calcSMA(atr14.slice(0, i + 1).filter(v => v != null), 20);
   164→   164→  const currentAtr = atr14[i]!;
   165→   165→  const rangeLen = currentAtr > atrSma20 ? 30 : 60;
   166→   166→
   167→   167→  const structHigh = Math.max(...highs.slice(Math.max(0, i - rangeLen), i));
   168→   168→  const structLow = Math.min(...lows.slice(Math.max(0, i - rangeLen), i));
   169→   169→
   170→   170→  // ── ADX / Trending check ──────────────────────────────
   171→   171→  const prevAdx = get(adx, i1, 0);
   172→   172→  const adxRising = adx[i]! > prevAdx;
   173→   173→  const adxHealthy = adx[i]! > 15;
   174→   174→  const marketTrending = (adx[i]! > 18) || (adxRising && adxHealthy);
   175→   175→
   176→   176→  // ── Volatility check ───────────────────────────────────
   177→   177→  const volLo = marketTrending ? 0.6 : 0.8;
   178→   178→  const volHi = marketTrending ? 1.8 : 1.3;
   179→   179→  const validVolatility = currentAtr > (atrSma20 * volLo) && currentAtr < (atrSma20 * volHi);
   180→   180→
   181→   181→  // ── Macro / Nifty check ────────────────────────────────
   182→   182→  let regimeSafe = true;
   183→   183→  if (config.useMacro && niftyCandles.length >= 200) {
   184→   184→    const niftyCloses = niftyCandles.map(c => c.close);
   185→   185→    const nifty200Arr = SMA.calculate({ period: 200, values: niftyCloses });
   186→   186→    const nifty200 = nifty200Arr[nifty200Arr.length - 1];
   187→   187→    regimeSafe = nifty200 != null && currentClose > nifty200;
   188→   188→  }
   189→   189→
   190→   190→  // ── Liquidity check ────────────────────────────────────
   191→   191→  const turnoverCr = (currentClose * currentVolume) / 10000000;
   192→   192→  const turnoverMA20 = calcSMA(
   193→   193→    candles.slice(Math.max(0, i - 20), i).map(c => (c.close * c.volume) / 10000000),
   194→   194→    20
   195→   195→  );
   196→   196→  const liquid = turnoverMA20 >= config.minTurnoverCr;
   197→   197→
   198→   198→  // ── Volume MA ─────────────────────────────────────────
   199→   199→  const volMA20 = calcSMA(volumes.slice(Math.max(0, i - 20), i + 1), 20);
   200→   200→  const minLiquidityCheck = currentVolume > volMA20;
   201→   201→
   202→   202→  // ── Not extended check ─────────────────────────────────
   203→   203→  const notExtended = currentClose < (ema20[i]! * 1.06);
   204→   204→
   205→   205→  // ── Confluence Scores (exact Pine Script logic) ────────
   206→   206→
   207→   207→  // Score 1: Trend
   208→   208→  const scoreTrend = (currentClose > sma200[i]! && rsi14[i]! > 52) ? 1 : 0;
   209→   209→
   210→   210→  // Score 2: Pullback
   211→   211→  const lowest4 = Math.min(...lows.slice(Math.max(0, i - 4), i + 1));
   212→   212→  const scorePullback = (lowest4 < ema20[i]! * 1.02 && currentClose > ema20[i]!) ? 1 : 0;
   213→   213→
   214→   214→  // Score 3: Trigger
   215→   215→  const scoreTrigger = currentClose > prevHigh ? 1 : 0;
   216→   216→
   217→   217→  // Score 4: Volume
   218→   218→  const prevVolMA20 = calcSMA(volumes.slice(Math.max(0, i1 - 20), i1 + 1), 20);
   219→   219→  const scoreVolume = (currentVolume > volMA20 * 1.05 || get(volumes, i1, 0) > prevVolMA20 * 1.1) ? 1 : 0;
   220→   220→
   221→   221→  // Score 5: Relative Strength (vs Nifty)
   222→   222→  let scoreRS = 0;
   223→   223→  if (niftyCandles.length >= 20) {
   224→   224→    const niftyClose = niftyCandles[niftyCandles.length - 1]?.close ?? 0;
   225→   225→    if (niftyClose > 0) {
   226→   226→      // Calculate rolling RS for last 25 bars
   227→   227→      const rsWindow = Math.min(25, candles.length, niftyCandles.length);
   228→   228→      const rsValues: number[] = [];
   229→   229→      for (let j = len - rsWindow; j < len; j++) {
   230→   230→        const nj = niftyCandles.length - rsWindow + (j - (len - rsWindow));
   231→   231→        if (nj >= 0 && niftyCandles[nj]?.close > 0) {
   232→   232→          rsValues.push(closes[j] / niftyCandles[nj].close);
   233→   233→        }
   234→   234→      }
   235→   235→      const rsMA = calcSMA(rsValues, 20);
   236→   236→      const currentRS = closes[i] / niftyClose;
   237→   237→      const rs5ago = rsValues.length > 5 ? rsValues[rsValues.length - 6] : 0;
   238→   238→      scoreRS = (currentRS > rsMA || currentRS > rs5ago) ? 1 : 0;
   239→   239→    }
   240→   240→  }
   241→   241→
   242→   242→  // Score 6: Gap
   243→   243→  const prevClose = get(closes, i1, currentClose);
   244→   244→  const gapPct = Math.abs(currentOpen - prevClose) / prevClose * 100;
   245→   245→  const scoreGap = gapPct < 3.5 ? 1 : 0;
   246→   246→
   247→   247→  const totalScore = scoreTrend + scorePullback + scoreTrigger + scoreVolume + scoreRS + scoreGap;
   248→   248→
   249→   249→  // ── Regime Clear check ─────────────────────────────────
   250→   250→  const regimeClear = regimeSafe && liquid && minLiquidityCheck && marketTrending && validVolatility && notExtended;
   251→   251→
   252→   252→  // ── Entry / SL / TP calculation ────────────────────────
   253→   253→  const theoreticalEP = Math.max(currentOpen, prevHigh);
   254→   254→  const stopLow5 = Math.min(...lows.slice(Math.max(0, i - 5), i));
   255→   255→  const rawSL = stopLow5 - (get(atr14, i1, currentAtr) * 0.8);
   256→   256→  const riskPerShare = theoreticalEP - rawSL;
   257→   257→
   258→   258→  if (riskPerShare <= 0) return null;
   259→   259→
   260→   260→  const structRange = structHigh - structLow;
   261→   261→  const expectedReward = structRange;
   262→   262→  const rr = expectedReward / riskPerShare;
   263→   263→
   264→   264→  if (rr < config.minRR && totalScore < 6) return null;
   265→   265→
   266→   266→  // ── Position Sizing ────────────────────────────────────
   267→   267→  const tp1 = theoreticalEP + (riskPerShare * 1.5);
   268→   268→
   269→   269→  // A+ sizing
   270→   270→  const riskAmtA = config.liveCapital * (config.riskPct / 100);
   271→   271→  const qtyRiskA = Math.floor(riskAmtA / (riskPerShare * 1)); // pointvalue = 1 for NSE cash
   272→   272→  const qtyCapA = Math.floor((config.liveCapital / config.maxSlots) / theoreticalEP);
   273→   273→  const qtyA = Math.max(0, Math.min(qtyRiskA, qtyCapA));
   274→   274→
   275→   275→  // B sizing (half risk, 60% capital)
   276→   276→  const riskAmtB = config.liveCapital * ((config.riskPct * 0.5) / 100);
   277→   277→  const qtyRiskB = Math.floor(riskAmtB / (riskPerShare * 1));
   278→   278→  const qtyCapB = Math.floor(((config.liveCapital / config.maxSlots) * 0.6) / theoreticalEP);
   279→   279→  const qtyB = Math.max(0, Math.min(qtyRiskB, qtyCapB));
   280→   280→
   281→   281→  // ── Final assembly ─────────────────────────────────────
   282→   282→  if (totalScore < config.minScore) return null;
   283→   283→  if (!regimeClear) return null;
   284→   284→
   285→   285→  // Strong momentum check (gap < 5% for score 6)
   286→   286→  const strongMomentum = totalScore === 6 && gapPct < 5.0;
   287→   287→  const withinGap = currentOpen <= (prevHigh * 1.03);
   288→   288→  if (!withinGap && !strongMomentum) return null;
   289→   289→
   290→   290→  return {
   291→   291→    symbol,
   292→   292→    date: candles[i].date,
   293→   293→    score: totalScore,
   294→   294→    setupType: totalScore === 6 ? 'A+' : 'B',
   295→   295→    entryPrice: Math.round(theoreticalEP * 100) / 100,
   296→   296→    stopLoss: Math.round(rawSL * 100) / 100,
   297→   297→    targetPrice: Math.round(tp1 * 100) / 100,
   298→   298→    riskReward: Math.round(rr * 100) / 100,
   299→   299→    atr: Math.round(currentAtr * 100) / 100,
   300→   300→    rsi: Math.round(rsi14[i]! * 100) / 100,
   301→   301→    scores: {
   302→   302→      scoreTrend,
   303→   303→      scorePullback,
   304→   304→      scoreTrigger,
   305→   305→      scoreVolume,
   306→   306→      scoreRS,
   307→   307→      scoreGap,
   308→   308→      totalScore,
   309→   309→    },
   310→   310→    checks: {
   311→   311→      trendAbove: scoreTrend === 1,
   312→   312→      pullbackOk: scorePullback === 1,
   313→   313→      triggerOk: scoreTrigger === 1,
   314→   314→      volumeOk: scoreVolume === 1,
   315→   315→      rsOk: scoreRS === 1,
   316→   316→      gapOk: scoreGap === 1,
   317→   317→      regimeSafe,
   318→   318→      liquid,
   319→   319→      trending: marketTrending,
   320→   320→      validVol: validVolatility,
   321→   321→      notExtended,
   322→   322→    },
   323→   323→    sizing: {
   324→   324→      qtyA,
   325→   325→      qtyB,
   326→   326→      riskPerShareA: Math.round(riskPerShare * 100) / 100,
   327→   327→      riskPerShareB: Math.round(riskPerShare * 100) / 100,
   328→   328→      riskAmtA: Math.round(riskAmtA),
   329→   329→      riskAmtB: Math.round(riskAmtB),
   330→   330→    },
   331→   331→    indicators: {
   332→   332→      sma200: Math.round(sma200[i]! * 100) / 100,
   333→   333→      ema20: Math.round(ema20[i]! * 100) / 100,
   334→   334→      ema10: Math.round(get(ema10, i, 0) * 100) / 100,
   335→   335→      adx: Math.round(adx[i]! * 100) / 100,
   336→   336→      diPlus: Math.round(get(diPlus, i, 0) * 100) / 100,
   337→   337→      diMinus: Math.round(get(diMinus, i, 0) * 100) / 100,
   338→   338→    },
   339→   339→  };
   340→   340→}

   376→   376→
   377→   377→export function runBacktest(
   378→   378→  symbol: string,
   379→   379→  candles: OHLCV[],
   380→   380→  niftyCandles: OHLCV[],
   381→   381→  config: ScreeningConfig = DEFAULT_CONFIG
   382→   382→): BacktestResult {
   383→   383→  const { maxSlots, maxOpenTrades, maxHoldBars, riskPct, liveCapital } = config;
   384→   384→  const trades: BacktestTradeResult[] = [];
   385→   385→  const equityCurve: { date: string; equity: number }[] = [];
   386→   386→  
   387→   387→  let capital = liveCapital;
   388→   388→  let peakEquity = capital;
   389→   389→  let maxDD = 0;
   390→   390→  let lastTradeBar = -999;
   391→   391→
   392→   392→  // Track open positions for pyramiding
   393→   393→  interface OpenPosition {
   394→   394→    entryBar: number;
   395→   395→    entryPrice: number;
   396→   396→    qty: number;
   397→   397→    stopLoss: number;
   398→   398→    tp1: number;
   399→   399→    score: number;
   400→   400→    setupType: string;
   401→   401→    partialTaken: boolean;
   402→   402→    trailActive: boolean;
   403→   403→    totalQty: number;
   404→   404→    avgPrice: number;
   405→   405→    blendedSL: number;
   406→   406→  }
   407→   407→  let openPositions: OpenPosition[] = [];
   408→   408→
   409→   409→  // Pre-compute all indicators for the full dataset
   410→   410→  const closes = candles.map(c => c.close);
   411→   411→  const highs = candles.map(c => c.high);
   412→   412→  const lows = candles.map(c => c.low);
   413→   413→  const volumes = candles.map(c => c.volume);
   414→   414→
   415→   415→  const sma200Full = padArray(SMA.calculate({ period: 200, values: closes }), candles.length);
   416→   416→  const ema20Full = padArray(EMA.calculate({ period: 20, values: closes }), candles.length);
   417→   417→  const ema10Full = padArray(EMA.calculate({ period: 10, values: closes }), candles.length);
   418→   418→  const rsi14Full = padArray(RSI.calculate({ period: 14, values: closes }), candles.length);
   419→   419→  const atr14Full = padArray(ATR.calculate({ period: 14, high: highs, low: lows, close: closes }), candles.length);
   420→   420→  const adxResultFull = ADX.calculate({ period: 14, high: highs, low: lows, close: closes });
   421→   421→  const adxFull = padArray(adxResultFull.map(d => d.adx), candles.length);
   422→   422→
   423→   423→  // Nifty data
   424→   424→  let nifty200Full: (number | null)[] = [];
   425→   425→  if (niftyCandles.length >= 200) {
   426→   426→    const nCloses = niftyCandles.map(c => c.close);
   427→   427→    nifty200Full = padArray(SMA.calculate({ period: 200, values: nCloses }), niftyCandles.length);
   428→   428→  }
   429→   429→
   430→   430→  // Walk through each bar
   431→   431→  for (let barIdx = 201; barIdx < candles.length; barIdx++) {
   432→   432→    const bar = candles[barIdx];
   433→   433→    
   434→   434→    // Check exits for all open positions
   435→   435→    const closedIndices: number[] = [];
   436→   436→    for (let p = 0; p < openPositions.length; p++) {
   437→   437→      const pos = openPositions[p];
   438→   438→      const barsHeld = barIdx - pos.entryBar;
   439→   439→      
   440→   440→      // Partial TP check
   441→   441→      if (!pos.partialTaken && bar.high >= pos.tp1) {
   442→   442→        const closeQty = Math.ceil(pos.totalQty * 0.3);
   443→   443→        const pnl = closeQty * (pos.tp1 - pos.avgPrice);
   444→   444→        capital += pnl + closeQty * pos.avgPrice;
   445→   445→        pos.totalQty -= closeQty;
   446→   446→        pos.trailActive = true;
   447→   447→        pos.partialTaken = true;
   448→   448→      }
   449→   449→
   450→   450→      // Trailing stop
   451→   451→      if (pos.trailActive) {
   452→   452→        const trailLow5 = Math.min(...lows.slice(Math.max(0, barIdx - 5), barIdx + 1));
   453→   453→        const dynamicTrail = Math.max(
   454→   454→          (ema10Full[barIdx] ?? pos.avgPrice) - (atr14Full[barIdx] ?? 0),
   455→   455→          trailLow5
   456→   456→        );
   457→   457→        pos.blendedSL = Math.max(pos.blendedSL, pos.avgPrice, dynamicTrail);
   458→   458→      }
   459→   459→
   460→   460→      // SL hit
   461→   461→      if (bar.low <= pos.blendedSL && pos.totalQty > 0) {
   462→   462→        const pnl = pos.totalQty * (pos.blendedSL - pos.avgPrice);
   463→   463→        capital += pnl + pos.totalQty * pos.avgPrice;
   464→   464→        trades.push({
   465→   465→          symbol,
   466→   466→          entryDate: candles[pos.entryBar].date,
   467→   467→          entryPrice: pos.avgPrice,
   468→   468→          exitDate: bar.date,
   469→   469→          exitPrice: pos.blendedSL,
   470→   470→          qty: pos.totalQty,
   471→   471→          pnl: Math.round(pnl * 100) / 100,
   472→   472→          pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   473→   473→          score: pos.score,
   474→   474→          setupType: pos.setupType,
   475→   475→          exitReason: pos.trailActive ? 'TRAIL_STOP' : 'SL_HIT',
   476→   476→        });
   477→   477→        closedIndices.push(p);
   478→   478→        continue;
   479→   479→      }
   480→   480→
   481→   481→      // Dead capital exit
   482→   482→      if (barsHeld > maxHoldBars && !pos.trailActive && pos.totalQty > 0) {
   483→   483→        const pnl = pos.totalQty * (bar.close - pos.avgPrice);
   484→   484→        capital += pnl + pos.totalQty * pos.avgPrice;
   485→   485→        trades.push({
   486→   486→          symbol,
   487→   487→          entryDate: candles[pos.entryBar].date,
   488→   488→          entryPrice: pos.avgPrice,
   489→   489→          exitDate: bar.date,
   490→   490→          exitPrice: bar.close,
   491→   491→          qty: pos.totalQty,
   492→   492→          pnl: Math.round(pnl * 100) / 100,
   493→   493→          pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   494→   494→          score: pos.score,
   495→   495→          setupType: pos.setupType,
   496→   496→          exitReason: 'DEAD_CAPITAL',
   497→   497→        });
   498→   498→        closedIndices.push(p);
   499→   499→      }
   500→   500→    }
   501→   501→    // Remove closed positions (and fully exited ones)
   502→   502→    openPositions = openPositions.filter((_, idx) => !closedIndices.includes(idx)).filter(p => p.totalQty > 0);
   503→   503→
   504→   504→    // Track equity
   505→   505→    peakEquity = Math.max(peakEquity, capital);
   506→   506→    const dd = ((peakEquity - capital) / peakEquity) * 100;
   507→   507→    maxDD = Math.max(maxDD, dd);
   508→   508→    equityCurve.push({ date: bar.date, equity: Math.round(capital * 100) / 100 });
   509→   509→
   510→   510→    // Check for new entry (simplified - using previous bar's signal)
   511→   511→    if (barIdx < 2 || barIdx - lastTradeBar <= config.cooldownBars) continue;
   512→   512→    if (openPositions.length >= maxOpenTrades) continue;
   513→   513→
   514→   514→    const i = barIdx;
   515→   515→    const i1 = barIdx - 1;
   516→   516→    
   517→   517→    // Quick signal check on previous bar
   518→   518→    const prevSMA200 = sma200Full[i1];
   519→   519→    const prevEMA20 = ema20Full[i1];
   520→   520→    const prevRSI = rsi14Full[i1];
   521→   521→    const prevATR = atr14Full[i1];
   522→   522→    const prevADX = adxFull[i1];
   523→   523→    const prevCandle = candles[i1];
   524→   524→
   525→   525→    if (!prevSMA200 || !prevEMA20 || !prevRSI || !prevATR || !prevADX) continue;
   526→   526→
   527→   527→    // Simplified confluence check (mirroring the screening logic)
   528→   528→    const prevADX1 = adxFull[i1 - 1] ?? 0;
   529→   529→    const trending = prevADX > 18 || (prevADX > prevADX1 && prevADX > 15);
   530→   530→    const atrSma20 = calcSMA(atr14Full.slice(Math.max(0, i - 20), i).filter(v => v != null), 20);
   531→   531→    const validVol = prevATR > atrSma20 * 0.6 && prevATR < atrSma20 * 1.8;
   532→   532→    const notExt = prevCandle.close < prevEMA20 * 1.06;
   533→   533→
   534→   534→    // Nifty check
   535→   535→    let regimeSafe = true;
   536→   536→    if (niftyCandles.length >= 200) {
   537→   537→      const niftyIdx = Math.min(niftyCandles.length - 1, barIdx);
   538→   538→      const nifty200 = nifty200Full[niftyIdx];
   539→   539→      if (nifty200) regimeSafe = niftyCandles[niftyIdx].close > nifty200;
   540→   540→    }
   541→   541→
   542→   542→    // Only require 2 of 4 regime filters (instead of all 4)
   543→   543→    const regimePassCount = [trending, validVol, notExt, regimeSafe].filter(Boolean).length;
   544→   544→    if (regimePassCount < 2) continue;
   545→   545→
   546→   546→    // Score calculation on prev bar
   547→   547→    let score = 0;
   548→   548→    if (prevCandle.close > prevSMA200 && prevRSI > 52) score++;
   549→   549→    const low4 = Math.min(...lows.slice(Math.max(0, i1 - 4), i1 + 1));
   550→   550→    if (low4 < prevEMA20 * 1.02 && prevCandle.close > prevEMA20) score++;
   551→   551→    if (prevCandle.close > candles[i1 - 1]?.high) score++;
   552→   552→    const vm = calcSMA(volumes.slice(Math.max(0, i1 - 20), i1 + 1), 20);
   553→   553→    if (prevCandle.volume > vm * 1.05) score++;
   554→   554→    
   555→   555→    const gapPct = Math.abs(prevCandle.open - candles[i1 - 1]?.close) / (candles[i1 - 1]?.close || 1) * 100;
   556→   556→    if (gapPct < 3.5) score++;
   557→   557→
   558→   558→    // Score 6: Relative Strength (vs Nifty)
   559→   559→    if (niftyCandles.length >= 25) {
   560→   560→      const niftyIdx = Math.min(niftyCandles.length - 1, i1);
   561→   561→      const niftyClose = niftyCandles[niftyIdx]?.close ?? 0;
   562→   562→      if (niftyClose > 0) {
   563→   563→        const rsWindow = Math.min(25, i1 + 1, niftyCandles.length);
   564→   564→        const rsStart = Math.max(0, i1 - rsWindow + 1);
   565→   565→        const niftyStart = Math.max(0, niftyIdx - rsWindow + 1);
   566→   566→        const rsValues: number[] = [];
   567→   567→        for (let j = 0; j < rsWindow; j++) {
   568→   568→          const cj = rsStart + j;
   569→   569→          const nj = niftyStart + j;
   570→   570→          if (closes[cj] && niftyCandles[nj]?.close) {
   571→   571→            rsValues.push(closes[cj] / niftyCandles[nj].close);
   572→   572→          }
   573→   573→        }
   574→   574→        if (rsValues.length > 5) {
   575→   575→          const rsMA = calcSMA(rsValues, 20);
   576→   576→          const currentRS = rsValues[rsValues.length - 1];
   577→   577→          const rs5ago = rsValues[rsValues.length - 6];
   578→   578→          if (currentRS > rsMA || currentRS > rs5ago) score++;
   579→   579→        }
   580→   580→      }
   581→   581→    }
   582→   582→
   583→   583→    if (score >= config.minScore) {
   584→   584→      const theoreticalEP = Math.max(bar.open, prevCandle.high);
   585→   585→      const stopLow5 = Math.min(...lows.slice(Math.max(0, i - 5), i));
   586→   586→      const rawSL = stopLow5 - (prevATR * 0.8);
   587→   587→      const riskPerShare = theoreticalEP - rawSL;
   588→   588→
   589→   589→      if (riskPerShare <= 0) continue;
   590→   590→
   591→   591→      const structRange = Math.max(...highs.slice(Math.max(0, i - 60), i)) - Math.min(...lows.slice(Math.max(0, i - 60), i));
   592→   592→      const rr = structRange / riskPerShare;
   593→   593→      if (rr < config.minRR && score < 6) continue;
   594→   594→
   595→   595→      // Size calculation
   596→   596→      const ddPenalty = Math.max(0.25, Math.min(1.0, 1.0 - (maxDD / 20.0)));
   597→   597→      const baseRisk = score === 6 ? riskPct : riskPct * 0.5;
   598→   598→      const appliedRisk = baseRisk * ddPenalty;
   599→   599→      const maxRisk = capital * (appliedRisk / 100);
   600→   600→      
   601→   601→      const qtyRisk = Math.floor(maxRisk / riskPerShare);
   602→   602→      const slotMultiplier = score === 6 ? 1.0 : 0.6;
   603→   603→      const targetAlloc = (capital / maxSlots) * slotMultiplier;
   604→   604→      const qtyCap = Math.floor(targetAlloc / theoreticalEP);
   605→   605→      const qty = Math.max(1, Math.min(qtyRisk, qtyCap));
   606→   606→
   607→   607→      const tp1 = theoreticalEP + (riskPerShare * 1.5);
   608→   608→
   609→   609→      if (capital < qty * theoreticalEP) continue;
   610→   610→
   611→   611→      openPositions.push({
   612→   612→        entryBar: barIdx,
   613→   613→        entryPrice: theoreticalEP,
   614→   614→        qty,
   615→   615→        stopLoss: rawSL,
   616→   616→        tp1,
   617→   617→        score,
   618→   618→        setupType: score === 6 ? 'A+' : 'B',
   619→   619→        partialTaken: false,
   620→   620→        trailActive: false,
   621→   621→        totalQty: qty,
   622→   622→        avgPrice: theoreticalEP,
   623→   623→        blendedSL: rawSL,
   624→   624→      });
   625→   625→
   626→   626→      capital -= qty * theoreticalEP;
   627→   627→      lastTradeBar = barIdx;
   628→   628→    }
   629→   629→  }
   630→   630→
   631→   631→  // Close remaining open positions at last bar price
   632→   632→  for (const pos of openPositions) {
   633→   633→    const lastPrice = candles[candles.length - 1].close;
   634→   634→    const pnl = pos.totalQty * (lastPrice - pos.avgPrice);
   635→   635→    capital += pnl + pos.totalQty * pos.avgPrice;
   636→   636→    trades.push({
   637→   637→      symbol,
   638→   638→      entryDate: candles[pos.entryBar].date,
   639→   639→      entryPrice: pos.avgPrice,
   640→   640→      exitDate: candles[candles.length - 1].date,
   641→   641→      exitPrice: lastPrice,
   642→   642→      qty: pos.totalQty,
   643→   643→      pnl: Math.round(pnl * 100) / 100,
   644→   644→      pnlPercent: Math.round((pnl / (pos.totalQty * pos.avgPrice)) * 10000) / 100,
   645→   645→      score: pos.score,
   646→   646→      setupType: pos.setupType,
   647→   647→      exitReason: 'BACKTEST_END',
   648→   648→    });
   649→   649→  }
   650→   650→
   651→   651→  // Calculate stats
   652→   652→  const wins = trades.filter(t => t.pnl > 0);
   653→   653→  const losses = trades.filter(t => t.pnl <= 0);
   654→   654→  const grossProfit = wins.reduce((s, t) => s + t.pnl, 0);
   655→   655→  const grossLoss = Math.abs(losses.reduce((s, t) => s + t.pnl, 0));
   656→   656→
   657→   657→  // Sharpe ratio (simplified)
   658→   658→  const returns = trades.map(t => t.pnlPercent);
   659→   659→  const avgReturn = returns.length > 0 ? returns.reduce((a, b) => a + b, 0) / returns.length : 0;
   660→   660→  const stdReturn = returns.length > 1
   661→   661→    ? Math.sqrt(returns.reduce((s, r) => s + Math.pow(r - avgReturn, 2), 0) / (returns.length - 1))
   662→   662→    : 1;
   663→   663→  const sharpeRatio = stdReturn > 0 ? (avgReturn / stdReturn) * Math.sqrt(252) : 0;
   664→   664→
   665→   665→  return {
   666→   666→    trades,
   667→   667→    equityCurve,
   668→   668→    stats: {
   669→   669→      totalTrades: trades.length,
   670→   670→      winTrades: wins.length,
   671→   671→      lossTrades: losses.length,
   672→   672→      winRate: trades.length > 0 ? Math.round((wins.length / trades.length) * 10000) / 100 : 0,
   673→   673→      profitFactor: grossLoss > 0 ? Math.round((grossProfit / grossLoss) * 100) / 100 : 0,
   674→   674→      maxDrawdown: Math.round(maxDD * 100) / 100,
   675→   675→      avgWin: wins.length > 0 ? Math.round((grossProfit / wins.length) * 100) / 100 : 0,
   676→   676→      avgLoss: losses.length > 0 ? Math.round((grossLoss / losses.length) * 100) / 100 : 0,
   677→   677→      bestTrade: trades.length > 0 ? Math.round(Math.max(...trades.map(t => t.pnl)) * 100) / 100 : 0,
   678→   678→      worstTrade: trades.length > 0 ? Math.round(Math.min(...trades.map(t => t.pnl)) * 100) / 100 : 0,
   679→   679→      finalCapital: Math.round(capital * 100) / 100,
   680→   680→      sharpeRatio: Math.round(sharpeRatio * 100) / 100,
   681→   681→    },
   682→   682→  };
   683→   683→}

   687→   687→function padArray(arr: (number | undefined)[], targetLen: number): (number | null)[] {
   688→   688→  const padding = targetLen - arr.length;
   689→   689→  const padded: (number | null)[] = new Array(padding).fill(null);
   690→   690→  return [...padded, ...arr.map(v => v ?? null)];
   691→   691→}

   693→   693→function calcSMA(values: (number | null | undefined)[], period: number): number {
   694→   694→  const valid = values.filter(v => v != null) as number[];
   695→   695→  const slice = valid.slice(-period);
   696→   696→  if (slice.length === 0) return 0;
   697→   697→  return slice.reduce((a, b) => a + b, 0) / slice.length;
   698→   698→}
