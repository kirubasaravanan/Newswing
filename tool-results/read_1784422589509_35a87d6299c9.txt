   321→   321→      if (data.success && data.summary?.wallet) {
   322→   322→        setWallet(data.summary.wallet);
   323→   323→        setCapital(String(data.summary.wallet.totalCapital));

   423→   423→function DividendPanel() {
   424→   424→  const [data, setData] = useState<any>(null);
   425→   425→  const [open, setOpen] = useState(false);
   426→   426→  const [form, setForm] = useState({ symbol: '', exDate: '', dividendPerShare: '', qty: '1', stockName: '' });
   427→   427→
   428→   428→  const fetchDividends = useCallback(async () => {
   429→   429→    try {
   430→   430→      const res = await fetch('/api/portfolio/dividends');
   431→   431→      const d = await res.json();
   432→   432→      if (d.success) setData(d);
   433→   433→    } catch { /* ignore */ }
   434→   434→  }, []);
   435→   435→  useEffect(() => { fetchDividends(); }, [fetchDividends]);
   436→   436→
   437→   437→  const addDividend = async () => {
   438→   438→    if (!form.symbol || !form.exDate || !form.dividendPerShare) return;
   439→   439→    try {
   440→   440→      const dps = parseFloat(form.dividendPerShare);
   441→   441→      const qty = parseInt(form.qty) || 1;
   442→   442→      const res = await fetch('/api/portfolio/dividends', {
   443→   443→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   444→   444→        body: JSON.stringify({ ...form, totalAmount: dps * qty }),
   445→   445→      });
   446→   446→      if ((await res.json()).success) {
   447→   447→        toast.success(`Dividend recorded: ${form.symbol}`);
   448→   448→        setForm({ symbol: '', exDate: '', dividendPerShare: '', qty: '1', stockName: '' });
   449→   449→        setOpen(false);
   450→   450→        fetchDividends();
   451→   451→      }
   452→   452→    } catch { toast.error('Failed to record dividend'); }
   453→   453→  };
   454→   454→
   455→   455→  const deleteDiv = async (id: string) => {
   456→   456→    await fetch(`/api/portfolio/dividends?id=${id}`, { method: 'DELETE' });
   457→   457→    fetchDividends();
   458→   458→  };
   459→   459→
   460→   460→  return (
   461→   461→    <div>
   462→   462→      <div className="flex items-center justify-between mb-3">
   463→   463→        <h3 className="text-sm font-semibold flex items-center gap-2"><IndianRupee className="h-4 w-4" /> Dividend Tracker</h3>
   464→   464→        <Dialog open={open} onOpenChange={setOpen}>
   465→   465→          <DialogTrigger asChild><Button size="sm" className="h-7 text-xs gap-1"><Plus className="h-3 w-3" /> Record Dividend</Button></DialogTrigger>
   466→   466→          <DialogContent className="sm:max-w-sm">
   467→   467→            <DialogHeader><DialogTitle>Record Dividend</DialogTitle></DialogHeader>
   468→   468→            <div className="grid gap-3 py-2">
   469→   469→              <div><Label className="text-xs">Symbol</Label><Input value={form.symbol} onChange={e => setForm({ ...form, symbol: e.target.value.toUpperCase() })} placeholder="ITC" className="h-8 text-xs" /></div>
   470→   470→              <div className="grid grid-cols-2 gap-2">
   471→   471→                <div><Label className="text-xs">Ex-Date</Label><Input type="date" value={form.exDate} onChange={e => setForm({ ...form, exDate: e.target.value })} className="h-8 text-xs" /></div>
   472→   472→                <div><Label className="text-xs">DPS (₹)</Label><Input type="number" step="0.01" value={form.dividendPerShare} onChange={e => setForm({ ...form, dividendPerShare: e.target.value })} placeholder="6.75" className="h-8 text-xs" /></div>
   473→   473→              </div>
   474→   474→              <div><Label className="text-xs">Qty Held</Label><Input type="number" value={form.qty} onChange={e => setForm({ ...form, qty: e.target.value })} className="h-8 text-xs" /></div>
   475→   475→            </div>
   476→   476→            <DialogFooter>
   477→   477→              <DialogClose asChild><Button variant="ghost" size="sm">Cancel</Button></DialogClose>
   478→   478→              <Button size="sm" onClick={addDividend}>Record</Button>
   479→   479→            </DialogFooter>
   480→   480→          </DialogContent>
   481→   481→        </Dialog>
   482→   482→      </div>
   483→   483→
   484→   484→      {data?.summary && data.summary.totalDividends > 0 && (
   485→   485→        <div className="grid grid-cols-2 gap-3 mb-3">
   486→   486→          <div className="rounded-lg bg-emerald-500/10 p-3">
   487→   487→            <div className="text-[10px] text-emerald-400 uppercase">Total Dividends</div>
   488→   488→            <div className="text-lg font-bold font-mono text-emerald-400">₹{data.summary.totalDividends.toLocaleString()}</div>
   489→   489→          </div>
   490→   490→          <div className="rounded-lg bg-secondary/50 p-3">
   491→   491→            <div className="text-[10px] text-muted-foreground uppercase">Stocks</div>
   492→   492→            <div className="text-lg font-bold font-mono">{data.summary.uniqueStocks}</div>
   493→   493→          </div>
   494→   494→        </div>
   495→   495→      )}
   496→   496→
   497→   497→      {(!data || data.dividends.length === 0) ? (
   498→   498→        <p className="text-xs text-muted-foreground text-center py-4">No dividends recorded yet.</p>
   499→   499→      ) : (
   500→   500→        <ScrollArea className="max-h-[220px]">
   501→   501→          <div className="space-y-1.5 pr-2">
   502→   502→            {data.dividends.slice(0, 20).map((d: any) => (
   503→   503→              <div key={d.id} className="flex items-center gap-2 text-xs py-1.5 border-b border-border/30">
   504→   504→                <span className="font-semibold shrink-0">{d.symbol}</span>
   505→   505→                <Badge variant="outline" className="text-[9px] h-4 text-emerald-400">₹{d.dividendPerShare}/share</Badge>
   506→   506→                <span className="font-mono text-emerald-400">₹{Math.round(d.totalAmount).toLocaleString()}</span>
   507→   507→                <span className="text-muted-foreground ml-auto">{new Date(d.exDate).toLocaleDateString()}</span>
   508→   508→                <button onClick={() => deleteDiv(d.id)} className="text-muted-foreground hover:text-red-400"><Trash2 className="h-3 w-3" /></button>
   509→   509→              </div>
   510→   510→            ))}
   511→   511→          </div>
   512→   512→        </ScrollArea>
   513→   513→      )}
   514→   514→    </div>
   515→   515→  );
   516→   516→}
